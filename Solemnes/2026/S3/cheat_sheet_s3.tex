\begin{cheatbox}{NumPy: arreglos y modelo}
\begin{lstlisting}
x = np.array([x0, x1, x2])
y = np.array([y0, y1, y2])

x_fino = np.linspace(xmin, xmax, 200)

\begin{tabularx}{\linewidth}{@{}lY@{}}

\end{cheatbox}

\begin{cheatbox}{Raices con \texttt{np.roots}}
\begin{lstlisting}
# Resolver a*x**2 + b*x + c = 0
raices = np.roots([a, b, c])

# Resolver a*x**2 + b*x + c = umbral
raices = np.roots([a, b, c - umbral])
\end{lstlisting}
\begin{tabularx}{\linewidth}{@{}lY@{}}
\entry{np.roots([...])}{calcula raíces de un polinomio}
\entry{[a, b, c]}{entrega coeficientes desde mayor a menor grado}
\entry{c - umbral}{mueve el umbral al lado izquierdo de la ecuación}
\entry{raices}{puede contener valores reales o complejos}
\end{tabularx}
\end{cheatbox}

\begin{cheatbox}{Filtrar raíces}
\begin{lstlisting}
soluciones = []

for r in raices:
    es_real = abs(r.imag) < tol
    esta_en_rango = xmin <= r.real <= xmax

    if es_real and esta_en_rango:
        soluciones.append(r.real)
\end{lstlisting}
\begin{tabularx}{\linewidth}{@{}lY@{}}
\entry{for r in raices:}{recorre una raíz a la vez}
\entry{r.imag}{parte imaginaria}
\entry{r.real}{parte real}
\entry{abs(r.imag) < tol}{verifica que sea casi real}
\entry{lista.append(valor)}{agrega una solución}
\end{tabularx}
\end{cheatbox}
