\begin{frame}[fragile]{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Instrumentar Bubble Sort}

\begin{minted}{python}
def bubble_sort(lista):
    n = len(lista)
    for i in range(n - 1):
        for j in range(n - 1 - i):
            if lista[j] > lista[j + 1]:
                lista[j], lista[j + 1] = \
                    lista[j + 1], lista[j]
    return lista
\end{minted}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 4: \hfill \textcolor{green}{$\checkmark$} \\ Suma recursiva}
\begin{minted}{python}
def suma_recursiva(lista):
    # Caso base
    if len(lista) == 0:
        return 0

    # Caso recursivo
    return lista[0] + suma_recursiva(lista[1:])

print(suma_recursiva([4, 7, 2]))
\end{minted}

  \begin{block}{Seguimiento}
    \[
      S([4,7,2])=4+S([7,2])
      =4+7+S([2])
      =4+7+2+S([])
      =13.
    \]
  \end{block}

  Sin caso base, las llamadas no terminan y Python genera un \texttt{RecursionError}.
\end{frame}

\begin{frame}[fragile]{Solución 5: \hfill \textcolor{green}{$\checkmark$} \\ Bubble Sort instrumentado}
\begin{minted}{python}
def bubble_sort_instrumentado(lista):
    n = len(lista)
    comparaciones = 0
    intercambios = 0

    for i in range(n - 1):
        for j in range(n - 1 - i):
            comparaciones += 1

            if lista[j] > lista[j + 1]:
                lista[j], lista[j + 1] = \
                    lista[j + 1], lista[j]
                intercambios += 1

    return lista, comparaciones, intercambios
\end{minted}
\end{frame}

\section{Ejercicios con \texttt{roots} y \texttt{polyfit}}

\begin{frame}{¿Qué herramienta corresponde?}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{np.roots}}
        \begin{itemize}
          \item Tenemos una ecuación polinómica.
          \item Conocemos sus coeficientes.

        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

\begin{frame}{Ejercicio 9: \hfill \textcolor{red}{$\clubsuit$} \\ Ajuste cuadrático y umbral}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 7: \hfill \textcolor{green}{$\checkmark$} \\ Altura de un objeto}
\begin{minted}{python}
coef = [-4.9, 16, 2]
raices = np.roots(coef)

raices_reales = []
for raiz in raices:
    if abs(raiz.imag) < 1e-10:
        raices_reales.append(raiz.real)

tiempos_fisicos = []
for tiempo in raices_reales:
    if tiempo >= 0:
        tiempos_fisicos.append(tiempo)

print("Raíces:", raices)
print("Tiempo de impacto:", tiempos_fisicos[0])
\end{minted}

  La raíz positiva representa el instante posterior al lanzamiento en que el objeto llega al suelo.
\end{frame}

\begin{frame}[fragile]{Solución 8: \hfill \textcolor{green}{$\checkmark$} \\ Ajuste lineal y \(R^2\)}
\begin{columns}
\begin{column}{0.52\textwidth}
\begin{minted}{python}
v, x0 = np.polyfit(t, x, deg=1)
x_fit = v*t + x0

residuos = x - x_fit
SS_res = np.sum(residuos**2)
SS_tot = np.sum(
    (x - np.mean(x))**2
)
R2 = 1 - SS_res/SS_tot

print(f"v = {v:.3f}")
print(f"x0 = {x0:.3f}")
print(f"R2 = {R2:.4f}")
\end{minted}
\end{column}

\begin{minted}{python}

plt.show()
\end{minted}
\end{column}
\end{columns}
