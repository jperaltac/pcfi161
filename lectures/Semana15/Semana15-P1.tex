\section{Repaso de Pandas}

% 7
\begin{frame}{¿Qué es un DataFrame?}
  \begin{block}{Idea}
    Un DataFrame es una tabla bidimensional con filas, columnas, nombres y tipos de datos.
  \end{block}

  \begin{itemize}
    \item Cada columna representa una variable.
    \item Cada fila representa una observación.
    \item Las columnas pueden contener números, texto o valores lógicos.
    \item Los valores faltantes suelen representarse con \texttt{NaN}.
  \end{itemize}

  \begin{alertblock}{Antes de calcular}
    Debemos conocer la estructura y la calidad de los datos.
  \end{alertblock}
\end{frame}

% 8
\begin{frame}[fragile]{Inspección inicial}
\begin{minted}{python}
print(df.head())
print(df.shape)
print(df.columns)
print(df.dtypes)
print(df.info())
\end{minted}

  \begin{itemize}
    \item \texttt{head()}: muestra las primeras filas.
    \item \texttt{shape}: entrega \texttt{(filas, columnas)}.
    \item \texttt{columns}: muestra los nombres disponibles.
    \item \texttt{dtypes}: informa el tipo de cada columna.
  \end{itemize}
\end{frame}

% 9
\begin{frame}[fragile]{Seleccionar columnas}
  Una columna se selecciona con un nombre; varias columnas, con una lista de nombres.

\begin{minted}{python}
temperatura = df["temperatura"]

subtabla = df[
    ["nombre", "temperatura", "distancia"]
]
\end{minted}

  \begin{alertblock}{Error frecuente}
    Para varias columnas se necesitan dos pares de corchetes: uno para seleccionar y otro para construir la lista.
  \end{alertblock}
\end{frame}

% 10
\begin{frame}[fragile]{Filtrar filas}
\begin{minted}{python}
cercanas = df[df["distancia"] < 20]

seleccion = df[
    (df["temperatura"] > 6000) &
    (df["distancia"] < 20)
]
\end{minted}

  \begin{itemize}
    \item Cada comparación produce valores \texttt{True} o \texttt{False}.
    \item Pandas conserva las filas donde la condición es verdadera.
    \item En filtros múltiples usamos \texttt{\&} y paréntesis.
  \end{itemize}
\end{frame}

% 11
\begin{frame}[fragile]{Datos faltantes}
\begin{minted}{python}
print(df.isna().sum())

df_limpio = df.dropna(
    subset=["temperatura"]
).copy()

df["temperatura"] = df["temperatura"].fillna(
    df["temperatura"].median()
)
\end{minted}

  \begin{block}{Decisión}
    Eliminar y rellenar no son operaciones equivalentes. La elección depende del significado de los datos.
  \end{block}
\end{frame}

  \begin{columns}
    \begin{column}{0.48\textwidth}

        \begin{itemize}
          \item Media
          \item Mediana
        \end{itemize}
      \end{block}

      La mediana es menos sensible a valores extremos.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Dispersión}
        \begin{itemize}
          \item Desviación estándar
          \item Mínimo y máximo
          \item Cuantiles
        \end{itemize}
      \end{block}

% 13
\begin{frame}[fragile]{Agrupar y resumir}
\begin{minted}{python}
resumen = df.groupby("clase").agg(
    temperatura_media=("temperatura", "mean"),
    magnitud_media=("magnitud", "mean"),
    cantidad=("nombre", "count")
)

\end{minted}

  \begin{block}{Interpretación}
    \texttt{groupby} separa las filas por categoría y \texttt{agg} calcula estadísticas dentro de cada grupo.
  \end{block}
\end{frame}

\section{Repaso de Algoritmos}

% 16
\begin{frame}{¿Qué es un algoritmo?}
  \begin{block}{Definición práctica}
    Un algoritmo es una secuencia finita y ordenada de instrucciones para resolver un problema.
  \end{block}

  Un algoritmo debe:
  \begin{itemize}
    \item recibir datos de entrada;
    \item realizar operaciones claramente definidas;
    \item terminar;
    \item producir un resultado;
    \item funcionar para más de un caso particular.
  \end{itemize}
\end{frame}

% 17
\begin{frame}{Búsqueda lineal y búsqueda binaria}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{Lineal}
        \begin{itemize}
          \item Revisa uno por uno.
          \item Funciona en listas desordenadas.
          \item Costo aproximado: \(\mathcal{O}(n)\).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Binaria}
        \begin{itemize}
          \item Descarta la mitad en cada paso.
          \item Requiere una lista ordenada.
          \item Costo aproximado: \(\mathcal{O}(\log n)\).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% 18
\begin{frame}{Condición para la búsqueda binaria}
  Considere la lista:
  \[
    [2,\ 5,\ 8,\ 12,\ 20,\ 31,\ 45].
  \]

  Para buscar \(20\):
  \begin{enumerate}
    \item comparar con el elemento central;
    \item decidir izquierda o derecha;
    \item repetir sobre la mitad restante.
  \end{enumerate}

  \begin{alertblock}{Importante}
    Si la lista no está ordenada, la decisión de descartar una mitad deja de ser válida.
  \end{alertblock}
\end{frame}

% 19
\begin{frame}{Estructura de la recursividad}
  Una función recursiva contiene:

  \begin{block}{Caso base}
    Situación sencilla que puede resolverse directamente y detiene las llamadas.
  \end{block}

  \begin{block}{Caso recursivo}
    La función se llama a sí misma con una versión más pequeña del problema.
  \end{block}

  \begin{alertblock}{Riesgo}
    Sin un caso base alcanzable, Python termina con \texttt{RecursionError}.
  \end{alertblock}
\end{frame}

% 20
\begin{frame}[fragile]{Seguimiento de una llamada recursiva}
\begin{minted}{python}
def suma_recursiva(lista):
    if len(lista) == 0:
        return 0
    return lista[0] + suma_recursiva(lista[1:])
\end{minted}

  Para \texttt{[4, 7, 2]}:
  \[
    S([4,7,2])=4+S([7,2])
    =4+7+S([2])
    =4+7+2+S([])
    =13.
  \]
\end{frame}

% 21
\begin{frame}{Bubble Sort}
  \begin{block}{Idea}
    Comparar elementos vecinos e intercambiarlos cuando están en el orden incorrecto.
  \end{block}

  \begin{itemize}
    \item En cada pasada, un elemento grande avanza hacia el final.
    \item Utiliza dos bucles anidados.
    \item Es fácil de comprender e implementar.
    \item Para listas grandes realiza muchas comparaciones.
  \end{itemize}

  \[
    \text{Costo típico: }\mathcal{O}(n^2).
  \]
\end{frame}

% 22
\begin{frame}{Merge Sort}
  \begin{block}{Divide y conquista}
    \begin{enumerate}
      \item Dividir la lista en dos mitades.
      \item Ordenar recursivamente cada mitad.
      \item Mezclar las dos mitades ordenadas.
    \end{enumerate}
  \end{block}

  \begin{itemize}
    \item El caso base es una lista con cero o un elemento.
    \item La función \texttt{merge} combina listas ya ordenadas.
    \item Su costo típico es \(\mathcal{O}(n\log n)\).
  \end{itemize}
\end{frame}

% 23
\begin{frame}{Complejidad y escalamiento}
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Algoritmo & Complejidad & Al duplicar \(n\) \\
      \midrule
      Búsqueda lineal & \(\mathcal{O}(n)\) & trabajo aproximado \(\times 2\) \\
      Búsqueda binaria & \(\mathcal{O}(\log n)\) & aumenta muy poco \\
      Bubble Sort & \(\mathcal{O}(n^2)\) & trabajo aproximado \(\times 4\) \\
      Merge Sort & \(\mathcal{O}(n\log n)\) & algo más que \(\times 2\) \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{alertblock}{Interpretación}
    La complejidad describe una tendencia de crecimiento, no un tiempo exacto en segundos.
  \end{alertblock}
\end{frame}

% 24
\begin{frame}{Medir el trabajo de un algoritmo}
  Podemos estudiar un algoritmo de dos formas:

  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{Tiempo}
        Medir cuánto tarda con \texttt{time.time()}.

        Depende del computador y de otros procesos.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Operaciones}
        Contar comparaciones, intercambios o llamadas.

        Describe directamente el trabajo realizado.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  Al probar varios tamaños podemos graficar costo versus \(n\).
\end{frame}

\section{Repaso de \texttt{roots} y \texttt{polyfit}}

% 25
\begin{frame}{Resolver ecuaciones con \texttt{np.roots}}
  \texttt{np.roots} encuentra los valores donde un polinomio es cero.

  \[
    p(x)=a_nx^n+\cdots+a_2x^2+a_1x+a_0.
  \]

  La entrada es:
  \[
    \texttt{[a\_n, ..., a\_2, a\_1, a\_0]}.
  \]

  \begin{alertblock}{Orden obligatorio}
    Los coeficientes se escriben desde la potencia mayor hasta el término independiente.
  \end{alertblock}
\end{frame}

% 26
\begin{frame}[fragile]{Raíces reales y soluciones físicas}
\begin{minted}{python}
coef = [-4.9, 16, 2]
raices = np.roots(coef)

for raiz in raices:
    if abs(raiz.imag) < 1e-10:
        tiempo = raiz.real
        if tiempo >= 0:
            print(tiempo)
\end{minted}

  \begin{itemize}
    \item Una raíz compleja puede ser matemáticamente correcta.
    \item Una raíz real negativa puede no tener sentido físico.
    \item El contexto determina cuáles soluciones se aceptan.
  \end{itemize}
\end{frame}

% 27
\begin{frame}{Ajustar datos con \texttt{np.polyfit}}
  Si tenemos mediciones \((x_i,y_i)\), buscamos un polinomio que represente su tendencia.

  \begin{block}{Ajuste lineal}
    \[
      y=mx+b,
      \qquad
      \texttt{m, b = np.polyfit(x, y, deg=1)}.
    \]
  \end{block}

  \begin{block}{Ajuste cuadrático}
    \[
      y=ax^2+bx+c,
      \qquad
      \texttt{a, b, c = np.polyfit(x, y, deg=2)}.
    \]
  \end{block}
\end{frame}

% 28
\begin{frame}[fragile]{Residuos y coeficiente \(R^2\)}
\begin{minted}{python}
y_fit = m*x + b
residuos = y - y_fit

SS_res = np.sum(residuos**2)
SS_tot = np.sum((y - np.mean(y))**2)
R2 = 1 - SS_res/SS_tot
\end{minted}

  \begin{itemize}
    \item El residuo es la diferencia entre dato y modelo.
    \item \(R^2\) cercano a 1 indica que el modelo explica gran parte de la variación.
    \item Un valor alto no reemplaza la inspección del gráfico.
  \end{itemize}
\end{frame}

% 29
\begin{frame}{Combinar \texttt{polyfit} y \texttt{roots}}
  Podemos ajustar datos y luego resolver una pregunta sobre el modelo.

  \begin{block}{Ejemplo de flujo}
    \begin{enumerate}
      \item Ajustar \(y=ax^2+bx+c\).
      \item Preguntar cuándo el modelo alcanza \(y=20\).
      \item Reescribir:
      \[
        ax^2+bx+(c-20)=0.
      \]
      \item Aplicar \texttt{np.roots([a, b, c - 20])}.
      \item Filtrar soluciones según el contexto.
    \end{enumerate}
  \end{block}
\end{frame}
