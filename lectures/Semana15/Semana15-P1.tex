\section{Ejercicio 1: Movimiento Rectilíneo Uniforme (np.polyfit)}

% ------------------------------------------------------------------------
% Slide: Ejercicio 1 - Enunciado
% ------------------------------------------------------------------------

\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Movimiento Rectilíneo Uniforme}
  \begin{block}{Contexto Físico}
    Un carrito se mueve sobre un riel de aire con velocidad constante. Se registra su posición en función del tiempo.
    
    \textbf{Modelo matemático:}
    \[
    x(t) = x_0 + v \cdot t
    \]
    donde: $x_0$ es la posición inicial (m), $v$ es la velocidad (m/s).
  \end{block}
  
  \begin{block}{Datos Experimentales Proporcionados}
    \begin{itemize}
      \item \texttt{t\_mru}: tiempos de medición (0, 1, 2, ..., 10 segundos)
      \item \texttt{x\_obs}: posiciones observadas con ruido experimental
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Ejercicio 1: Tareas a Realizar \hfill \textcolor{red}{$\clubsuit$}}
  \begin{enumerate}
    \item \textbf{Ajuste lineal con np.polyfit:}
    \begin{itemize}
      \item Usar \texttt{np.polyfit(t\_mru, x\_obs, deg=1)}
      \item Extraer velocidad ($v$) y posición inicial ($x_0$)
      \item Calcular posiciones ajustadas: $x_{\text{est}} = x_0 + v \cdot t$
    \end{itemize}
    
    \item \textbf{Cálculo de $R^2$:}
    \begin{itemize}
      \item Calcular $SS_{\text{res}} = \sum (x_{\text{obs}} - x_{\text{est}})^2$
      \item Calcular $SS_{\text{tot}} = \sum (x_{\text{obs}} - \overline{x}_{\text{obs}})^2$
      \item Obtener $R^2 = 1 - \frac{SS_{\text{res}}}{SS_{\text{tot}}}$
    \end{itemize}
    
    \item \textbf{Visualización:} Gráfico con datos observados (scatter) y ajuste (línea)
    
    \item \textbf{Reporte:} Imprimir $x_0$, $v$ y $R^2$ con formato adecuado
  \end{enumerate}
\end{frame}

import numpy as np
import matplotlib.pyplot as plt

# Tarea 1: Ajuste lineal
coef = np.polyfit(t_mru, x_obs, deg=1)
v_estimado = coef[0]
x0_estimado = coef[1]
x_estimado = x0_estimado + v_estimado * t_mru

# Tarea 2: Cálculo de R²
SS_res = np.sum((x_obs - x_estimado)**2)
SS_tot = np.sum((x_obs - np.mean(x_obs))**2)
R2 = 1 - (SS_res / SS_tot)

\section{Ejercicio 2: Ley de Gravitación Universal (curve\_fit)}

% ------------------------------------------------------------------------
% Slide: Ejercicio 2 - Enunciado

\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Ley de Gravitación Universal}
  \begin{block}{Contexto Físico}
    Se mide la fuerza gravitacional entre dos masas a diferentes distancias para verificar la ley de gravitación universal.
    
    \textbf{Modelo matemático:}
    \[
    F(r) = \frac{G M m}{r^2}
    \]
    donde: $G$ es la constante gravitacional, $M$ y $m$ son las masas, $r$ es la distancia.
    
    Para simplificar: $F(r) = \frac{K}{r^2}$ donde $K = GMm$
  \end{block}
  
  \begin{block}{Datos Experimentales Proporcionados}
    \begin{itemize}
      \item \texttt{r\_grav}: distancias (1 a 10 metros)
      \item \texttt{F\_obs}: fuerzas observadas con incertidumbres
      \item \texttt{sigma\_F}: incertidumbres en las mediciones
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Ejercicio 2: Tareas a Realizar \hfill \textcolor{red}{$\clubsuit$}}
  \begin{enumerate}
    \item \textbf{Definir modelo:} \texttt{def modelo\_gravitacion(r, K)}
    
    \item \textbf{Ajuste no lineal:}
    \begin{itemize}
      \item Valor inicial: \texttt{p0 = [80]}
      \item Usar \texttt{curve\_fit} con ponderación por \texttt{sigma\_F}
      \item Extraer parámetro óptimo y covarianza
    \end{itemize}
    
    \item \textbf{Incertidumbres:} Calcular \texttt{perr = np.sqrt(np.diag(pcov))}
    
    \item \textbf{Cálculo de $\chi^2$ reducido:}
    \begin{itemize}
      \item Residuos ponderados: $r_i = \frac{F_{\text{obs},i} - F_{\text{fit},i}}{\sigma_{F,i}}$
      \item $\chi^2 = \sum r_i^2$, $\nu = n - 1$, $\chi^2_{\text{red}} = \chi^2 / \nu$
    \end{itemize}
    
    \item \textbf{Visualización:} Gráfico con barras de error y curva ajustada
    
    \item \textbf{Reporte:} Imprimir $K \pm \sigma_K$ y calidad del ajuste
  \end{enumerate}
\end{frame}

from scipy.optimize import curve_fit

# Datos proporcionados
#r_grav 
#F_obs  
#sigma_F 

# Tarea 1: Definir modelo
def modelo_gravitacion(r, K):
    return K / r**2

# Tarea 2: Ajuste no lineal
p0 = [80]
popt, pcov = curve_fit(modelo_gravitacion, 
                       r_grav, F_obs, p0=p0, 
                       sigma=sigma_F, 
                       absolute_sigma=True)
F_fit = modelo_gravitacion(r_grav, *popt)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[fontsize=\tiny,firstnumber=29]{python}
# Tarea 3: Incertidumbres
perr = np.sqrt(np.diag(pcov))
sigma_K = perr[0]

# Tarea 4: χ² reducido
residuos = (F_obs - F_fit) / sigma_F
chi2 = np.sum(residuos**2)
nu = len(r_grav) - 1
chi2_red = chi2 / nu

print(f"χ² = {chi2:.4f}")
print(f"ν = {nu}")
print(f"χ²_red = {chi2_red:.4f}")

# Tarea 5: Visualización
r_fino = np.linspace(1, 10, 200)
F_fino = modelo_gravitacion(r_fino, *popt)

plt.figure(figsize=(8, 5))
plt.errorbar(r_grav, F_obs, yerr=sigma_F,fmt='o', label='Datos experimentales')
plt.plot(r_fino, F_fino, 'r-', linewidth=2,label=f'Ajuste: K={popt[0]:.1f}±{sigma_K:.1f}')
plt.xlabel('Distancia (m)', fontsize=12)
plt.ylabel('Fuerza (N)', fontsize=12)
plt.title('Ley de Gravitación Universal', fontsize=14)
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

# Tarea 6: Reporte
print(f"\nK = ({popt[0]:.2f} ± {sigma_K:.2f}) N·m²")
calidad = "Bueno" if chi2_red < 2 else "Regular"
print(f"Calidad del ajuste (χ²_red): {calidad}")
\end{minted}
\end{column}

\section{Ejercicio 3: Análisis del Algoritmo Selection Sort}

% ------------------------------------------------------------------------
% Slide: Explicación del Algoritmo Selection Sort
% ------------------------------------------------------------------------

\begin{frame}{¿Cómo funciona Selection Sort?}
  \begin{block}{Idea Principal}
    Selection Sort ordena una lista \textbf{seleccionando} repetidamente el elemento más pequeño de la parte no ordenada y colocándolo al inicio.
  \end{block}
  
  \begin{block}{Proceso Paso a Paso}
    \begin{enumerate}
      \item Buscar el elemento \textbf{mínimo} en toda la lista
      \item Intercambiarlo con el elemento en la \textbf{primera posición}
      \item Buscar el mínimo en el resto de la lista (desde posición 2)
      \item Intercambiarlo con el elemento en la \textbf{segunda posición}
      \item Repetir hasta que toda la lista esté ordenada
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Visualización del Proceso de Selection Sort}

  \begin{alertblock}{Ejemplo Visual}
    Lista inicial: \texttt{[64, 25, 12, 22, 11]}
    \begin{itemize}
      \item Iteración 1: Encontrar mín(64,25,12,22,11)=\\ 11 $\rightarrow$ \texttt{[\textbf{11}, 25, 12, 22, 64]}
      \item Iteración 2: Encontrar mín(25,12,22,64)=\\ 12 $\rightarrow$ \texttt{[11, \textbf{12}, 25, 22, 64]}
      \item Iteración 3: Encontrar mín(25,22,64)=\\ 22 $\rightarrow$ \texttt{[11, 12, \textbf{22}, 25, 64]}
      \item Iteración 4: Encontrar mín(25,64)=\\ 25 $\rightarrow$ \texttt{[11, 12, 22, \textbf{25}, 64]}
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Análisis del Algoritmo Selection Sort}
  \begin{block}{Contexto Computacional}
    Selection Sort es otro algoritmo de ordenamiento simple con complejidad $O(n^2)$. Encuentra el elemento mínimo y lo intercambia con la posición actual.
    
    \textbf{Algoritmo proporcionado:}
\begin{minted}[fontsize=\tiny]{python}
def selection_sort(lst):
    n = len(lst)
    for i in range(n-1):
        min_idx = i
        for j in range(i+1, n):
            if lst[j] < lst[min_idx]:
                min_idx = j
        lst[i], lst[min_idx] = lst[min_idx], lst[i]
    return lst
\end{minted}
  \end{block}
\end{frame}

# Tarea 1: Instrumentación
def selection_sort_instrumentado(lst):
    n = len(lst)
    comparaciones = 0
    intercambios = 0
    
    for i in range(n-1):
        min_idx = i
        for j in range(i+1, n):
            comparaciones += 1
            if lst[j] < lst[min_idx]:
                min_idx = j
        
        if min_idx != i:
            lst[i], lst[min_idx] = \
                lst[min_idx], lst[i]
            intercambios += 1
    
    return lst, comparaciones, intercambios
