\begin{frame}{Limitaciones de las Estructuras Nativas de Python}
  \begin{itemize}
    \item \textbf{Rendimiento}: Las listas de Python no están optimizadas para cálculos numéricos masivos
    \item \textbf{Sintaxis verbose}: Operaciones matemáticas requieren bucles explícitos
      \begin{itemize}
        \item Ejemplo: Multiplicar cada elemento de una lista por 2 → \texttt{[x*2 for x in lista]}
      \end{itemize}
    \item \textbf{Funcionalidad limitada}: Sin soporte nativo para álgebra lineal, estadística avanzada
    \item \textbf{Heterogeneidad}: Las listas pueden contener diferentes tipos de datos, lo que impide optimizaciones
  \end{itemize}
  
  \begin{alertblock}{Problema en Ciencia y Física}
    Los cálculos científicos requieren alto rendimiento y operaciones optimizadas con grandes conjuntos de datos numéricos
  \end{alertblock}
\end{frame}

\begin{frame}{¿Qué es NumPy?}
  \begin{itemize}
    \item \textbf{NumPy} = \textit{Numerical Python}
    \item \textbf{Biblioteca fundamental} para computación científica en Python

    \item Pilares principales:

        \item \textbf{ndarray}: Estructura de datos multidimensional y homogénea

        \item \textbf{Herramientas matemáticas}: Álgebra lineal, transformaciones, estadística

  \end{itemize}

  \begin{alertblock}{Importancia en Física Computacional}
    NumPy es la base de prácticamente todo el ecosistema científico de Python: SciPy, Matplotlib, Pandas, scikit-learn y TensorFlow se construyen sobre NumPy

\begin{frame}[fragile]{Broadcasting}
  \begin{itemize}
    \item Mecanismo que NumPy utiliza para manejar operaciones entre arrays de diferentes formas.

  \end{itemize}

  \end{itemize}

                     [30, 31, 32, 30, 29]]) # Ciudad 3

\end{minted}
\end{frame}

\begin{frame}[fragile]{Slicing Lógico en NumPy}
  \begin{itemize}
    \item El \textbf{slicing lógico} permite modificar elementos basándose en condiciones
    \item Sintaxis: \texttt{array[condición] = nuevo\_valor}
  \end{itemize}

\begin{minted}[fontsize=\tiny]{python}
import numpy as np
# Crear array de ejemplo
arr = np.array([1, 3, 5, 7, 9, 2, 4, 6, 8])
print("Array original:", arr)
# Crear una copia para no modificar el original
arr_mod = arr.copy()
# Aplicar condiciones lógicas
arr_mod[arr_mod <= 5] = 0    # Valores ≤ 5 → 0
arr_mod[arr_mod > 5] = 1     # Valores > 5 → 1
print("Array modificado:", arr_mod)
# Resultado: [0 0 0 1 1 0 0 1 1]
\end{minted}
  \begin{alertblock}{¿Cómo funciona?}
    \begin{itemize}
      \item \texttt{arr\_mod <= 5} crea un array booleano: \texttt{[True, True, True, ...]}
      \item NumPy usa este array como "máscara" para seleccionar elementos
      \item Solo los elementos donde la condición es \texttt{True} se modifican
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Concatenación de Arrays}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np

# Crear arrays
a = np.array([[1, 2], [3, 4]])
b = np.array([[5, 6], [7, 8]])

print("Array a:\n", a)
print("Array b:\n", b)

# Concatenar horizontalmente (axis=1)
h_concat = np.concatenate((a, b), axis=1)
print("Concatenación horizontal:\n", h_concat)
# [[1 2 5 6]
#  [3 4 7 8]]

# Concatenar verticalmente (axis=0)
v_concat = np.concatenate((a, b), axis=0)
print("Concatenación vertical:\n", v_concat)
# [[1 2]
#  [3 4]
#  [5 6]
#  [7 8]]
\end{minted}

\begin{itemize}
  \item \texttt{concatenate} une arrays a lo largo de un eje específico.
  \item \texttt{axis=0}: une a lo largo de las filas (verticalmente).
  \item \texttt{axis=1}: une a lo largo de las columnas (horizontalmente).
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Stack - Apilamiento de Arrays}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np

# Crear arrays 1D
a = np.array([1, 2, 3])
b = np.array([4, 5, 6])

# Stack vertical (crear un array 2D)
v_stack = np.vstack((a, b))
print("vstack:\n", v_stack)
# [[1 2 3]
#  [4 5 6]]

# Stack horizontal
h_stack = np.hstack((a, b))
print("hstack:", h_stack)
# [1 2 3 4 5 6]

# Crear una matriz 3D con arrays 2D
c = np.array([[1, 2], [3, 4]])
d = np.array([[5, 6], [7, 8]])
depth_stack = np.dstack((c, d))
print("dstack:\n", depth_stack)
\end{minted}

\begin{itemize}
  \item \texttt{vstack}: apila arrays verticalmente (añade filas).
  \item \texttt{hstack}: apila arrays horizontalmente (añade columnas).
  \item \texttt{dstack}: apila arrays en "profundidad" (3D).
  \item También existe \texttt{np.stack} para control más preciso.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------------
\begin{frame}[fragile]{Transposición y Manipulación de Ejes}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np

# Crear matriz 2D
mat = np.array([[1, 2, 3],
                [4, 5, 6]])
print("Original:\n", mat)  # 2x3

# Transponer matriz (intercambiar filas y columnas)
trans = mat.T
print("Transpuesta:\n", trans)  # 3x2

# Crear un array 3D
arr_3d = np.arange(24).reshape(2, 3, 4)
print("Array 3D shape:", arr_3d.shape)  # (2, 3, 4)

# Transponer todos los ejes (0, 1, 2) -> (2, 1, 0)
trans_3d = arr_3d.transpose(2, 1, 0)
print("Transpuesto 3D shape:", trans_3d.shape)  # (4, 3, 2)

# Intercambiar solo algunos ejes
swap_axes = np.swapaxes(arr_3d, 0, 2)
print("Swapped shape:", swap_axes.shape)  # (4, 3, 2)
\end{minted}

\begin{itemize}
  \item \texttt{.T} es equivalente a \texttt{transpose()} para 2D.
  \item \texttt{transpose()} permite especificar el nuevo orden de ejes.
  \item \texttt{swapaxes()} intercambia dos ejes específicos.
\end{itemize}
\end{frame}

# Aproximación de pi usando la serie de Leibniz
import numpy as np

n = 100000  # término superior
k = np.arange(n+1)  # [0..n]

# (-1)^k = np.power(-1, k)
# denominador = (2*k + 1)
terminos = ((-1)**k) / (2*k + 1)
aprox_pi = 4 * np.sum(terminos)
print("Aproximación de pi con n =", n, ":", aprox_pi)

# Comparar con valor real
import math
print("Diferencia:", abs(math.pi - aprox_pi))
\end{minted}
