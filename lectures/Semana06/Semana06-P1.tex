\begin{frame}{Limitaciones de las Estructuras Nativas de Python}
  \begin{itemize}
    \item \textbf{Rendimiento}: Las listas de Python no están optimizadas para cálculos numéricos masivos
    \item \textbf{Sintaxis verbose}: Operaciones matemáticas requieren bucles explícitos
      \begin{itemize}
        \item Ejemplo: Multiplicar cada elemento de una lista por 2 → \texttt{[x*2 for x in lista]}
      \end{itemize}
    \item \textbf{Funcionalidad limitada}: Sin soporte nativo para álgebra lineal, estadística avanzada
    \item \textbf{Heterogeneidad}: Las listas pueden contener diferentes tipos de datos, lo que impide optimizaciones
  \end{itemize}
  
  \begin{alertblock}{Problema en Ciencia y Física}
    Los cálculos científicos requieren alto rendimiento y operaciones optimizadas con grandes conjuntos de datos numéricos
  \end{alertblock}
\end{frame}

\section{Motivación: NumPy}

% ------------------------------------------------------------------------
% Slide 5: ¿Qué es NumPy?
% ------------------------------------------------------------------------
\begin{frame}{¿Qué es NumPy?}
  \begin{itemize}
    \item \textbf{NumPy} = \textbf{Num}erical \textbf{Py}thon.
    \item Ofrece:
      \begin{itemize}
        \item Estructura central: \textbf{ndarray} (arreglo multidimensional).
        \item Funciones matemáticas optimizadas para operar sobre estos arreglos.
        \item Alto rendimiento (implementaciones en C bajo la interfaz de Python).
      \end{itemize}
    \item \textbf{Base de muchas librerías} científicas (p.e. SciPy, Pandas, etc.).
    \item \textbf{Estándar de facto} en computación científica con Python.
  \end{itemize}
  
  \begin{block}{Historia Breve}
    Creada en 2005, evolucionando de bibliotecas anteriores (Numeric, Numarray), diseñada específicamente para cálculos científicos eficientes.
  \end{block}
\end{frame}

% ------------------------------------------------------------------------
% Slide 6: Ventajas Frente a Listas de Python
% ------------------------------------------------------------------------
\begin{frame}{Ventajas frente a Listas Nativas de Python}
  \begin{itemize}
    \item \textbf{Vectorización}: escribir operaciones como \texttt{arr * 2} en lugar de bucles manuales.
    \item \textbf{Rendimiento}: implementaciones de bajo nivel (C/Fortran) para cálculos numéricos.
    \item \textbf{Herramientas avanzadas}: slicing sofisticado, broadcasting, manipulación de ejes (dimensiones).
    \item \textbf{Compatibilidad} con otras librerías (Matplotlib, SciPy, etc.).
  \end{itemize}
  
  \begin{block}{Aplicaciones en Física}
    \begin{itemize}
      \item Simulaciones numéricas (p.e. dinámica de partículas)
      \item Análisis de datos experimentales
      \item Procesamiento de señales e imágenes astronómicas
      \item Resolución numérica de ecuaciones diferenciales
    \end{itemize}
  \end{block}
\end{frame}

% ------------------------------------------------------------------------
% Slide 7: Comparativa de Rendimiento
% ------------------------------------------------------------------------
\begin{frame}{Comparativa de Rendimiento: NumPy vs. Listas}
  \begin{itemize}
    \item \textbf{Ejemplo}: Suma de todos los elementos en una secuencia de un millón de números
    \item Con \textbf{listas de Python}: Aproximadamente 10-100 veces más lento
    \item Con \textbf{NumPy}: Operaciones optimizadas a nivel de hardware
  \end{itemize}
  
  \begin{alertblock}{Importancia del Rendimiento}
    En problemas físicos reales (simulaciones, análisis de datos experimentales), la diferencia de rendimiento puede significar horas o incluso días de tiempo de cómputo.
  \end{alertblock}
  
  \textbf{Hoy aprenderemos}: Cómo aprovechar esta potencia para nuestros problemas científicos.
\end{frame}

\section{Manipulación de Listas en Python}

% ------------------------------------------------------------------------
% Slide: Creación y Acceso a Listas
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Creación y Acceso a Listas}
  \begin{itemize}
    \item Las listas son \textbf{colecciones ordenadas y mutables}
    \item Creación y acceso:
  \end{itemize}
\begin{minted}{python}
# Creación de listas
numeros = [10, 20, 30, 40, 50]
nombres = ["Ana", "Carlos", "Elena"]
mixta = [1, "dos", 3.0, True]

# Acceso por índice (desde 0)
print(numeros[0])     # 10
print(numeros[-1])    # 50 (último elemento)

# Slicing [inicio:fin:paso]
print(numeros[1:4])   # [20, 30, 40]
print(numeros[::2])   # [10, 30, 50] (saltos de 2)
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide: Métodos para Agregar y Quitar Elementos
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Métodos para Agregar y Quitar Elementos}
\begin{minted}[fontsize=\tiny]{python}
frutas = ["manzana", "naranja", "plátano"]

# Agregar elementos
frutas.append("uva")           # Al final
print(frutas)  # ['manzana', 'naranja', 'plátano', 'uva']

frutas.insert(1, "pera")       # En posición específica
print(frutas)  # ['manzana', 'pera', 'naranja', 'plátano', 'uva']

# Eliminar elementos
eliminado = frutas.pop()       # Elimina y devuelve el último
print("Eliminado:", eliminado) # Eliminado: uva
print(frutas)  # ['manzana', 'pera', 'naranja', 'plátano']

frutas.remove("pera")          # Elimina por valor
print(frutas)  # ['manzana', 'naranja', 'plátano']

del frutas[0]                  # Elimina por índice
print(frutas)  # ['naranja', 'plátano']
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide: Unir y Separar Listas
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Unir y Separar Listas}
\begin{minted}[fontsize=\tiny]{python}
# Unir listas
lista1 = [1, 2, 3]
lista2 = [4, 5, 6]

# Concatenación con +
combinada = lista1 + lista2
print(combinada)  # [1, 2, 3, 4, 5, 6]

# Método extend
lista1.extend(lista2)
print(lista1)     # [1, 2, 3, 4, 5, 6]

# Multiplicar listas
repetida = [0] * 4
print(repetida)   # [0, 0, 0, 0]

# Dividir cadena en lista
texto = "Python es poderoso"
palabras = texto.split()
print(palabras)   # ['Python', 'es', 'poderoso']

# Unir lista en cadena
unida = " | ".join(palabras)
print(unida)      # 'Python | es | poderoso'
\end{minted}
\end{frame}

% Slide: Métodos Adicionales para Listas

\begin{frame}[fragile]{Métodos Adicionales para Listas}
\begin{minted}[fontsize=\tiny]{python}
numeros = [3, 1, 4, 1, 5, 9, 2]

# Ordenar lista (modifica la original)
numeros.sort()
print(numeros)  # [1, 1, 2, 3, 4, 5, 9]

# Invertir lista
numeros.reverse()
print(numeros)  # [9, 5, 4, 3, 2, 1, 1]

# Contar ocurrencias
print(numeros.count(1))  # 2

# Encontrar índice (primera aparición)
print(numeros.index(5))  # 1

# Copiar una lista
lista_copia = numeros.copy()  # o list(numeros) o numeros[:]
print(lista_copia)     # [9, 5, 4, 3, 2, 1, 1]

# Verificar si existe un elemento
print(5 in numeros)  # True
print(7 in numeros)  # False
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide: Listas por Comprensión
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Listas por Comprensión}
  \begin{itemize}
    \item Forma concisa de crear listas basadas en listas existentes
    \item Estructura: \texttt{[expresión for item in iterable if condición]}
  \end{itemize}
\begin{minted}[fontsize=\tiny]{python}
# Lista del cuadrado de números del 0 al 9
cuadrados = [x**2 for x in range(10)]
print(cuadrados)  # [0, 1, 4, 9, 16, 25, 36, 49, 64, 81]

# Lista de números pares del 0 al 9
pares = [x for x in range(10) if x % 2 == 0]
print(pares)      # [0, 2, 4, 6, 8]

# Combinar elementos de dos listas
nombres = ["Ana", "Luis"]
apellidos = ["Gómez", "Pérez"]
nombres_completos = [n + " " + a for n in nombres for a in apellidos]
print(nombres_completos)  
# ['Ana Gómez', 'Ana Pérez', 'Luis Gómez', 'Luis Pérez']
\end{minted}
\end{frame}

\begin{frame}[fragile]{Caso Práctico con Listas: Análisis de Datos}
\begin{minted}{python}
# Temperaturas diarias de una semana (°C)
temperaturas = [22.5, 23.1, 21.8, 24.2, 25.0, 22.9, 21.5]

# Análisis básico manual
suma = sum(temperaturas)
prom = suma / len(temperaturas)
temp_max = max(temperaturas)
temp_min = min(temperaturas)

print(f"Promedio: {prom:.1f}°C")
print(f"Máxima: {temp_max}°C, Mínima: {temp_min}°C")
print(f"Rango: {temp_max - temp_min:.1f}°C")

# Identificar días con temperaturas > 23°C
dias_calurosos = [i+1 for i, t in enumerate(temperaturas) if t > 23]
print(f"Días calurosos: {dias_calurosos}")  # [4, 5]
\end{minted}
\begin{itemize}
  \item Observa cómo podemos combinar \texttt{enumerate}, listas por comprensión e \texttt{if} para análisis.

# Creamos el arreglo y calculamos estadísticas
import numpy as np
arr = np.arange(1, 11)  # [1..10]

arr_mod[arr_mod > 5] = 1
print("Array modificado:", arr_mod)

# Suma de vector a matriz usando broadcasting
import numpy as np
mat = np.array([[1,2,3],
                [4,5,6],

resultado = mat + fila  # Broadcasting

# [[11 22 33]
#  [14 25 36]
#  [17 28 39]]
\end{minted}
	extbf{Discusión:} Broadcasting permite sumar el vector a cada fila sin bucles explícitos.

\begin{frame}[fragile]{Solución 3 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Aproximación de $\pi$}
\begin{minted}{python}
# Aproximación de pi usando la serie de Leibniz
import numpy as np

n = 100000  # término superior
k = np.arange(n+1)  # [0..n]

# (-1)^k = np.power(-1, k)
# denominador = (2*k + 1)
terminos = ((-1)**k) / (2*k + 1)
aprox_pi = 4 * np.sum(terminos)
print("Aproximación de pi con n =", n, ":", aprox_pi)

# Comparar con valor real
import math
print("Diferencia:", abs(math.pi - aprox_pi))
\end{minted}
	\textbf{Discusión:} La vectorización permite calcular sumas grandes de manera eficiente y rápida.
\end{frame}
