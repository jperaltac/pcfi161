      \item Crear una lista de 1000 números usando \texttt{range(1, 1001)}.
      \item Calcular el cuadrado de cada elemento usando listas por comprensión.
      \item Repetir el mismo cálculo usando un array de NumPy.
      \item Comparar los resultados y discutir las diferencias.
    \end{itemize}
  \end{block}
  
  \textbf{Trabajemos este ejercicio JUNTOS paso a paso:}
  \begin{enumerate}
    \item Crear datos con ambos métodos
    \item Aplicar la operación de elevar al cuadrado
    \item Verificar que ambos dan el mismo resultado
    \item Analizar ventajas de cada enfoque
  \end{enumerate}
\end{frame}

% ------------------------------------------------------------------------
% Slide 7: Solución Ejercicio Guiado 1
% ------------------------------------------------------------------------
\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 1: \hfill \textcolor{green}{$\checkmark$} \\ Comparación Listas vs NumPy}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np

# Método 1: Listas de Python
lista_numeros = list(range(1, 1001))
cuadrados_lista = [x**2 for x in lista_numeros]
print(f"Primeros 5 cuadrados (lista): {cuadrados_lista[:5]}")
print(f"Suma total (lista): {sum(cuadrados_lista)}")

# Método 2: NumPy
array_numeros = np.arange(1, 1001)
cuadrados_array = array_numeros**2
print(f"Primeros 5 cuadrados (NumPy): {cuadrados_array[:5]}")
print(f"Suma total (NumPy): {np.sum(cuadrados_array)}")

# Verificación
son_iguales = cuadrados_lista == cuadrados_array.tolist()
print(f"¿Los resultados son iguales? {all(son_iguales)}")

# Análisis estadístico con NumPy
print(f"Media: {np.mean(cuadrados_array)}")
print(f"Desviación estándar: {np.std(cuadrados_array)}")
\end{minted}
\textbf{Discusión:} NumPy es más conciso para operaciones matemáticas y ofrece funciones estadísticas integradas.
\end{frame}
\end{solutionblock}

      \item Calcular estadísticas básicas: media, mínimo, máximo, desviación estándar.
      \item Identificar las horas con temperaturas extremas (> media + std, < media - std).
      \item Mostrar un resumen completo del análisis.
    \end{itemize}
  \end{block}
  
  \textbf{Conceptos:} Simulación de datos, análisis estadístico, indexación condicional.
  \\

% ------------------------------------------------------------------------
% Slide 9: Solución Ejercicio Guiado 2
% ------------------------------------------------------------------------
\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 2: \hfill \textcolor{green}{$\checkmark$} \\ Análisis de Datos de Temperatura}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np

# Simular datos de temperatura (24 horas)
np.random.seed(42)  # Para reproducibilidad
temperaturas = np.random.normal(20, 3, 24)
horas = np.arange(0, 24)

# Estadísticas básicas
media = np.mean(temperaturas)
minimo = np.min(temperaturas)
maximo = np.max(temperaturas)
std_dev = np.std(temperaturas)

print(f"=== ANÁLISIS DE TEMPERATURAS (24 HORAS) ===")
print(f"Media: {media:.2f}°C")
print(f"Mínimo: {minimo:.2f}°C (hora {np.argmin(temperaturas)})")
print(f"Máximo: {maximo:.2f}°C (hora {np.argmax(temperaturas)})")
print(f"Desviación estándar: {std_dev:.2f}°C")

# Identificar temperaturas extremas
temp_altas = temperaturas > (media + std_dev)
temp_bajas = temperaturas < (media - std_dev)

print(f"\n=== TEMPERATURAS EXTREMAS ===")
print(f"Horas con temp. alta: {horas[temp_altas]}")
print(f"Valores altos: {temperaturas[temp_altas]:.2f}")
print(f"Horas con temp. baja: {horas[temp_bajas]}")
print(f"Valores bajos: {temperaturas[temp_bajas]:.2f}")
\end{minted}
\textbf{Discusión:} El uso de indexación booleana permite filtrar datos eficientemente.
\end{frame}
\end{solutionblock}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 5 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Análisis de Caída Libre con NumPy}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np
# Constantes físicas
y0 = 100  # altura inicial (m)
v0 = 0    # velocidad inicial (m/s)
g = 9.8   # aceleración gravitacional (m/s^2)
# Crear array de tiempos (0 a 5 segundos, 50 puntos)
tiempos = np.linspace(0, 5, 50)
# Calcular posición: y = y0 + v0*t - 0.5*g*t^2
posiciones = y0 + v0*tiempos - 0.5*g*tiempos**2
# Calcular velocidad: v = v0 - g*t
velocidades = v0 - g*tiempos
# Encontrar cuándo el objeto llega al suelo (y ≈ 0)
indices_suelo = np.where(posiciones <= 0)[0]
if len(indices_suelo) > 0:
    tiempo_impacto = tiempos[indices_suelo[0]]
    velocidad_impacto = velocidades[indices_suelo[0]]
    print(f"Tiempo de impacto: {tiempo_impacto:.2f} s")
    print(f"Velocidad al impactar: {velocidad_impacto:.2f} m/s")
else:
    print("El objeto no llega al suelo en el intervalo analizado")
# Velocidad máxima (en magnitud)
velocidad_max = np.max(np.abs(velocidades))
tiempo_vel_max = tiempos[np.argmax(np.abs(velocidades))]
print(f"Velocidad máxima: {velocidad_max:.2f} m/s a los {tiempo_vel_max:.2f} s")
# Posición a los 3 segundos (aproximadamente)
indice_3s = np.argmin(np.abs(tiempos - 3))
posicion_3s = posiciones[indice_3s]
print(f"Posición a los {tiempos[indice_3s]:.2f} s: {posicion_3s:.2f} m")
\end{minted}
\textbf{Discusión:} NumPy permite simular fácilmente fenómenos físicos y analizar resultados sin necesidad de bucles.
\end{frame}
\end{solutionblock}
