\begin{frame}[fragile]{Solución 1 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Ecuación de Movimiento en 1D}
\begin{minted}{python}
# Solicitar datos al usuario con unidades claras
x0 = float(input("Posición inicial x0 (m): "))
v0 = float(input("Velocidad inicial v0 (m/s): "))
a = float(input("Aceleración a (m/s²): "))
t = float(input("Tiempo t (s): "))

# Aplicar la ecuación cinemática
x_final = x0 + v0 * t + 0.5 * a * (t**2)

# Mostrar resultado con formato claro
print(f"La posición final es: {x_final:.2f} m")
\end{minted}

\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Promedio y Varianza de Mediciones}
\begin{minted}[fontsize=\tiny]{python}
# le solicitare distancias al usuario 

x1 = float(input('Ingrese la primera distancia en metros:  '))
x2 = float(input('Ingrese la segunda distancia en metros:  '))
x3 = float(input('Ingrese la tercera distancia en metros:  '))

#calculo promedio sobre las distancias 
promedio = (x1 + x2 + x3) / 3 

#calculo varianza muestral .. expandiendo suma
varianza = 1 / (3 - 2) * ((x1 - promedio) **2  + (x2 - promedio) ** 2 + (x3 - promedio) ** 2)

#muestro resultados en pantalla 
print(f' El promedio sobre las variables de distancia es {promedio}')
print(f' La varianza muestral sobre las variables es {varianza}')
\end{minted}
\textbf{Discusión:} Uso de varibales para calcular otra,  calculo de varianza, formato de decimales.
\end{frame}

    print(f"Tabla de multiplicar del {numero}:")
    print("-" * 25)

    # Generar tabla usando bucle for
    for i in range(1, 11):
        resultado = numero * i
