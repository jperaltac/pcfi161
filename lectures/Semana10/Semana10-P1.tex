% ------------------------------------------------------------------------
% Slide 7: Ejercicio 1 - Órbita Parabólica
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Órbita Parabólica con NumPy y Matplotlib}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Una trayectoria parabólica se describe con: \(r(\theta) = \frac{p}{1 + \cos(\theta)}\)
      \item Datos: \(p = 2.0\) UA (parámetro orbital)
      \item Crear array de ángulos \(\theta\) de 0 a \(\pi\) con 360 puntos
      \item Calcular \(r(\theta)\) y convertir a coordenadas cartesianas:
        \begin{itemize}
          \item \(x = r \cos(\theta)\)
          \item \(y = r \sin(\theta)\)
        \end{itemize}
      \item Graficar la trayectoria parabólica
      \item Calcular y mostrar el valor máximo de \(r\) con 3 decimales
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} Arrays NumPy, operaciones vectorizadas, conversión polar-cartesiana, gráficos.
\end{frame}

% ------------------------------------------------------------------------
% Slide 8: Solución 1 - Órbita Parabólica
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 1 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Órbita Parabólica}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Parámetro de la órbita parabólica
p = 2.0  # UA

# 1. Crear array de ángulos theta
theta = np.linspace(0, np.pi, 360)

# 2. Calcular radio r(theta) - ecuación parabólica
r = p / (1 + np.cos(theta))

# 3. Convertir a coordenadas cartesianas
x = r * np.cos(theta)
y = r * np.sin(theta)
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide 9: Solución 1 (continuación) - Graficación
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 1 (continuación): \hfill \textcolor{green}{$\checkmark$} \\ Graficación y Análisis}
\begin{minted}{python}
# 4. Graficar la trayectoria parabólica
plt.figure(figsize=(10, 8))
plt.plot(x, y, 'r-', linewidth=2, label='Trayectoria parabólica')
plt.axis('equal')
plt.grid(True, alpha=0.3)
plt.xlabel('x [UA]')
plt.ylabel('y [UA]')
plt.title(f'Trayectoria Parabólica (p={p} UA)')
plt.legend()
plt.show()

# 5. Calcular valor máximo de r
r_max = np.max(r)
print(f"Valor máximo de r: {r_max:.3f} UA")
\end{minted}
\textbf{Discusión:} La órbita parabólica representa una trayectoria de escape. El valor máximo de \(r\) ocurre en \(\theta = \pi\).
\end{frame}

\begin{frame}[fragile]{Generación de Datos: \hfill \textcolor{blue}{$\star$} \\ Creación del archivo galaxies\_data.csv}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}[fontsize=\tiny]{python}
import pandas as pd
import random

# Establecer semilla para reproducibilidad
random.seed(42)

# Definir tipos de galaxias
tipos_galaxias = ['Elíptica', 'Espiral', 
                  'Irregular']

# Generar datos aleatorios
data = {
    'galaxy_type': [random.choice(tipos_galaxias) 
                    for _ in range(100)],
    'distance_Mpc': [random.uniform(10.0, 150.0) 
                     for _ in range(100)],
    'luminosity': [random.uniform(0.5, 5.0) 
                   for _ in range(100)]
}
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[fontsize=\tiny,firstnumber=16]{python}
# Crear DataFrame
df = pd.DataFrame(data)

# Redondear valores para mejor presentación
df['distance_Mpc'] = df['distance_Mpc'].round(2)
df['luminosity'] = df['luminosity'].round(3)

# Mostrar primeras filas
print(df.head(10))

# Guardar a archivo CSV
df.to_csv('galaxies_data.csv', index=False)

print("\n¡Archivo galaxies_data.csv creado!")
print(f"Total de galaxias: {len(df)}")
\end{minted}
\end{column}

% Slide 11: Solución 2 - Análisis con pandas
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Análisis con pandas}
\begin{minted}{python}
import pandas as pd
import matplotlib.pyplot as plt

# 1. Cargar datos
df = pd.read_csv("galaxies_data.csv")

# 2. Mostrar primeras filas
print(df.head())
print("="*50)

# 3. Agrupación y análisis estadístico
resultado = df.groupby('galaxy_type').agg({
    'distance_Mpc': ['mean', 'std'],
    'luminosity': 'max'
}).round(2)

print(resultado)
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide 12: Solución 2 (continuación) - Ordenamiento y Gráfico

\begin{minted}{python}
# 4. Ordenar por distancia media (descendente)
resultado_ordenado = resultado.sort_values(('distance_Mpc', 'mean'), ascending=False)

print(resultado_ordenado)

# 5. Scatter plot
plt.figure(figsize=(10, 6))
plt.scatter(df['distance_Mpc'], df['luminosity'],alpha=0.6, edgecolors='black')

plt.ylabel('Luminosidad relativa')
plt.title('Relación Distancia-Luminosidad de Galaxias')
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\textbf{Discusión:} El análisis por tipo de galaxia revela patrones de distribución espacial y luminosidad.
\end{frame}

\begin{frame}{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Verificación de la Ley de Hubble}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Datos observacionales de 5 galaxias:
        \begin{itemize}
          \item \texttt{d = np.array([10, 25, 50, 75, 100])} [Mpc] - distancias
          \item \texttt{v = np.array([700, 1750, 3500, 5250, 7000])} [km/s] - velocidades
        \end{itemize}
      \item Crear scatter plot de \(d\) vs \(v\)
      \item Calcular la constante de Hubble \(H_0 = v/d\) para cada galaxia
      \item Mostrar los 5 valores de \(H_0\) con 2 decimales
      \item Calcular y mostrar media y desviación estándar de \(H_0\)
      \item Agregar etiquetas y título descriptivo al gráfico
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} Arrays NumPy, operaciones elemento a elemento, estadísticas, scatter plot.
  \\
  \textbf{Física relevante:} La Ley de Hubble establece que \(v = H_0 \cdot d\), evidencia de expansión del universo.
\end{frame}

% ------------------------------------------------------------------------
% Slide 14: Solución 3 - Ley de Hubble
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 3 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Verificación de la Ley de Hubble}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Datos de galaxias
d = np.array([10, 25, 50, 75, 100])  # Distancia [Mpc]
v = np.array([700, 1750, 3500, 5250, 7000])  # Velocidad [km/s]

# 1. Scatter plot
plt.figure(figsize=(10, 6))
plt.scatter(d, v, c='red', s=100, alpha=0.7, 
           edgecolors='black', linewidth=1.5)
plt.xlabel('Distancia d [Mpc]')
plt.ylabel('Velocidad de recesión v [km/s]')
plt.title('Ley de Hubble: Relación Distancia-Velocidad')
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide 15: Solución 3 (continuación) - Cálculo de H0
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 3 (continuación): \hfill \textcolor{green}{$\checkmark$} \\ Cálculo de la Constante de Hubble}
\begin{minted}{python}
# 2. Calcular constante de Hubble H0 = v/d
H0 = v / d

# 3. Mostrar valores con 2 decimales
print("Valores de H0 [km/s/Mpc]:")
print(H0.round(2))

# 4. Estadísticas de H0
media_H0 = np.mean(H0)
std_H0 = np.std(H0)

print(f"\nMedia de H0: {media_H0:.2f} km/s/Mpc")
print(f"Desviación estándar: {std_H0:.2f} km/s/Mpc")
print(f"\nLos valores de H0 son consistentes, confirmando")
print(f"la proporcionalidad entre distancia y velocidad.")
\end{minted}
\textbf{Discusión:} Si \(H_0\) es aproximadamente constante, confirma la Ley de Hubble. Los valores cercanos a 70 km/s/Mpc coinciden con mediciones cosmológicas modernas.
\end{frame}

\begin{frame}{Errores Comunes a Evitar}
  \begin{block}{NumPy}
    \begin{itemize}
      \item Olvidar usar funciones de NumPy (\texttt{np.cos} vs \texttt{cos})
      \item Confundir operaciones elemento a elemento con álgebra matricial
      \item No verificar dimensiones al crear arrays
    \end{itemize}
  \end{block}
  
  \begin{block}{pandas}
    \begin{itemize}
      \item Sintaxis incorrecta en \texttt{groupby().agg()}
      \item Olvidar entre paréntesis en \texttt{sort\_values()}
      \item No especificar nombres correctos de columnas (sensibles a mayúsculas)
    \end{itemize}
  \end{block}
  
  \begin{block}{Matplotlib}
    \begin{itemize}
      \item Olvidar \texttt{plt.show()} al final
      \item No incluir etiquetas de ejes y título
      \item Confundir \texttt{plot} con \texttt{scatter}
    \end{itemize}
  \end{block}
