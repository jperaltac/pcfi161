      \item Datos: \(p = 2.0\) UA (parámetro orbital)
      \item Crear array de ángulos \(\theta\) de 0 a \(\pi\) con 360 puntos
      \item Calcular \(r(\theta)\) y convertir a coordenadas cartesianas:
        \begin{itemize}
          \item \(x = r \cos(\theta)\)
          \item \(y = r \sin(\theta)\)
        \end{itemize}
      \item Graficar la trayectoria parabólica
      \item Calcular y mostrar el valor máximo de \(r\) con 3 decimales
    \end{itemize}
  \end{block}

import matplotlib.pyplot as plt

theta = np.linspace(0, np.pi, 360)

# 2. Calcular radio r(theta) - ecuación parabólica
r = p / (1 + np.cos(theta))

x = r * np.cos(theta)
y = r * np.sin(theta)

% ------------------------------------------------------------------------

% ------------------------------------------------------------------------
\begin{solutionblock}
\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Perfil Térmico}
\begin{Verbatim}[fontsize=\tiny]
import numpy as np
import matplotlib.pyplot as plt

# 1. Definir una función para el modelo
def temperatura_barra(x):
    T = 22 + 55*np.exp(-x/4)*np.cos(np.pi*x/10)
    return T

# 2. Crear posiciones sobre la barra
x = np.linspace(0, 10, 200)  # cm

# 3. Calcular temperatura en cada posicion
T = temperatura_barra(x)

# 4. Estadísticas básicas
T_max = np.max(T)
T_min = np.min(T)
T_prom = np.mean(T)

indice_max = np.argmax(T)
x_max = x[indice_max]

print(f"Temperatura maxima: {T_max:.2f} C en x={x_max:.2f} cm")
print(f"Temperatura minima: {T_min:.2f} C")
print(f"Temperatura promedio: {T_prom:.2f} C")

if T_max > 70:
    print("Advertencia: la barra supera 70 C")
else:
    print("La barra no supera 70 C")
\end{Verbatim}
\end{frame}
\end{solutionblock}

\begin{frame}[shrink=5]{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Curva de Luz de una Estrella Variable}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Crear un array \texttt{t} con 300 valores entre 0 y 12 horas.
      \item Modelar el brillo con:
        \[
          B(t)=10+0.45\sin\left(\frac{2\pi t}{4}\right)+0.12\cos\left(\frac{2\pi t}{2}\right)
        \]
      \item Graficar brillo versus tiempo.
      \item Calcular brillo máximo, brillo mínimo y amplitud \((B_{\max}-B_{\min})/2\).
      \item Contar cuántos puntos quedan sobre el brillo promedio.
      \item Crear una función \texttt{brillo\_estrella(t)} para calcular \(B(t)\).
      \item Usar un condicional para clasificar la variabilidad como baja o alta.

% ------------------------------------------------------------------------
% Slide 14: Solución 3 - Curva de luz
% ------------------------------------------------------------------------
\begin{solutionblock}
\begin{frame}[fragile]{Solución 3 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Curva de Luz}
\begin{Verbatim}[fontsize=\tiny]
import numpy as np
import matplotlib.pyplot as plt

t = np.linspace(0, 12, 300)

def brillo_estrella(t):
    B = 10 + 0.45*np.sin(2*np.pi*t/4)
    B = B + 0.12*np.cos(2*np.pi*t/2)
    return B

B = brillo_estrella(t)

B_max = np.max(B)
B_min = np.min(B)
amplitud = (B_max - B_min) / 2
promedio = np.mean(B)
puntos_sobre_promedio = np.sum(B > promedio)

print(f"Brillo maximo: {B_max:.3f}")
print(f"Brillo minimo: {B_min:.3f}")
print(f"Amplitud: {amplitud:.3f}")
print(f"Puntos sobre el promedio: {puntos_sobre_promedio}")

if amplitud > 0.5:
    print("Variabilidad alta")
else:
    print("Variabilidad baja o moderada")

plt.figure(figsize=(9, 5))
plt.plot(t, B, color="purple", label="brillo")
plt.axhline(promedio, color="black", linestyle="--", label="promedio")
plt.xlabel("Tiempo [h]")
plt.ylabel("Brillo relativo")
\end{Verbatim}
\end{frame}
\end{solutionblock}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 5 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Péndulo Simple}
\begin{Verbatim}[fontsize=\tiny]
import numpy as np
import matplotlib.pyplot as plt

g = 9.8
L = np.linspace(0.20, 1.80, 9)

def periodo_pendulo(L, g):
    T = 2*np.pi*np.sqrt(L/g)
    return T

T_teo = periodo_pendulo(L, g)

diferencias = np.array([0.03, -0.02, 0.01, 0.04, -0.01,
                        0.02, -0.03, 0.01, 0.03])
T_obs = T_teo + diferencias

residuos = T_obs - T_teo
promedio_abs = np.mean(np.abs(residuos))

print("Longitudes:", L.round(2))
print("Periodo teorico:", T_teo.round(3))
print("Periodo observado:", T_obs.round(3))
print("Residuos:", residuos.round(3))
print(f"Promedio absoluto de residuos: {promedio_abs:.3f} s")

if promedio_abs < 0.05:
    print("Modelo aceptable para estas mediciones")
else:
    print("Revisar mediciones o modelo")
\end{Verbatim}
\end{frame}
\end{solutionblock}

  \begin{block}{Enunciado}

      \item Modelar la intensidad como:

      \item Graficar intensidad versus longitud de onda.

% ------------------------------------------------------------------------

% ------------------------------------------------------------------------
\begin{solutionblock}
\begin{frame}[fragile]{Solución 6 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Línea de Emisión}
\begin{Verbatim}[fontsize=\tiny]
import numpy as np
import matplotlib.pyplot as plt

lam = np.linspace(650, 670, 400)
centro = 656.3
sigma = 1.4

def linea_emision(lam, centro, sigma):
    I = 1 + 4*np.exp(-((lam - centro)**2) / (2*sigma**2))
    return I

I = linea_emision(lam, centro, sigma)

indice_max = np.argmax(I)
lam_max = lam[indice_max]
I_max = I[indice_max]

mitad_max = I_max / 2
puntos_sobre_mitad = np.sum(I > mitad_max)

print(f"Longitud de onda del maximo: {lam_max:.2f} nm")
print(f"Intensidad maxima: {I_max:.2f}")
print(f"Puntos sobre la mitad del maximo: {puntos_sobre_mitad}")

if abs(lam_max - centro) < 0.1:
    print("Maximo consistente con el centro esperado")
else:
    print("Maximo desplazado respecto al centro esperado")
\end{Verbatim}
\end{frame}
\end{solutionblock}

  \begin{block}{Durante la resolución}
    \begin{itemize}
      \item Usa \textbf{nombres descriptivos} de variables.
      \item Encapsula modelos repetidos en una \textbf{función} con \texttt{def}.
      \item Usa \textbf{condicionales} para reportar decisiones o clasificaciones.
      \item Incluye \textbf{comentarios breves} cuando el cálculo tenga varios pasos.
      \item \textbf{Verifica} dimensiones con \texttt{shape} cuando uses matrices.
      \item \textbf{Prueba} el código por partes antes de continuar.
    \end{itemize}
  \end{block}
\end{frame}

% ------------------------------------------------------------------------
% Slide 26: Errores Comunes a Evitar
% ------------------------------------------------------------------------
\begin{frame}[shrink=8]{Errores Comunes a Evitar}
  \begin{block}{NumPy}
    \begin{itemize}
      \item Usar \texttt{math.sin()} o \texttt{math.cos()} sobre arrays en vez de \texttt{np.sin()} o \texttt{np.cos()}.
      \item Confundir el índice de un valor con el valor mismo.
      \item Olvidar que las máscaras booleanas deben tener el mismo tamaño que el array filtrado.
      \item Usar \texttt{reshape()} con un número total de elementos incompatible.
    \end{itemize}
  \end{block}

  \begin{block}{Matplotlib}
    \begin{itemize}
      \item Olvidar \texttt{plt.show()} al final.
      \item Entregar gráficos sin etiquetas de ejes, unidades o título.
      \item No usar \texttt{plt.axis("equal")} en trayectorias 2D.
      \item Poner muchas curvas sin \texttt{legend()}.
    \end{itemize}
  \end{block}
\end{frame}
