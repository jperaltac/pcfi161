\section{Dos Problemas Frecuentes}

\begin{frame}{¿Qué problema queremos resolver?}
  \begin{block}{Dos situaciones distintas}
    \begin{enumerate}
      \item \textbf{Resolver una ecuación:} encontrar los valores de \(x\) donde un polinomio vale cero.
      \item \textbf{Ajustar datos:} encontrar una recta o curva polinómica que represente mediciones con ruido.
    \end{enumerate}
  \end{block}

  \begin{alertblock}{Herramientas de hoy}
    \begin{itemize}
      \item \texttt{np.roots}: resuelve ecuaciones polinómicas.
      \item \texttt{np.polyfit}: ajusta un polinomio a datos.
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{Mapa mental: \texttt{roots} vs \texttt{polyfit}}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{np.roots}}
        \begin{itemize}
          \item Entrada: coeficientes del polinomio.
          \item Salida: raíces.
          \item Pregunta típica: ¿cuándo \(p(x)=0\)?
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{np.polyfit}}
        \begin{itemize}
          \item Entrada: datos \(x\), datos \(y\), grado.
          \item Salida: coeficientes del ajuste.
          \item Pregunta típica: ¿qué modelo representa mis datos?
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  \begin{alertblock}{Idea clave}
    \texttt{roots} trabaja con una ecuación ya conocida. \texttt{polyfit} construye un modelo desde datos.
  \end{alertblock}
\end{frame}

\section{Resolver Polinomios con \texttt{np.roots}}

\begin{frame}{Polinomios como listas de coeficientes}
  Para usar \texttt{np.roots}, los coeficientes se escriben desde la potencia mayor hasta el término independiente.

  \begin{block}{Ejemplo}
    \[
      x^3 - 6x^2 + 11x - 6 = 0
    \]
    \[
      \texttt{coef = [1, -6, 11, -6]}
    \]
  \end{block}

  \begin{itemize}
    \item Si falta una potencia, se coloca coeficiente cero.
    \item \(x^4 - 5x^2 + 4 = 0\) se escribe como \texttt{[1, 0, -5, 0, 4]}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Sintaxis básica de \texttt{np.roots}}
\begin{minted}{python}
import numpy as np

# x^3 - 6x^2 + 11x - 6 = 0
coef = [1, -6, 11, -6]

raices = np.roots(coef)
print(raices)
\end{minted}

  \begin{block}{Interpretación}
    La salida contiene todos los valores de \(x\) que hacen que el polinomio sea cero.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Raíces reales y raíces complejas}
  \texttt{np.roots} puede devolver números complejos. En problemas físicos muchas veces interesa filtrar las raíces reales.

\begin{minted}{python}
raices_reales = []

for raiz in raices:
    if abs(raiz.imag) < 1e-10:
        raices_reales.append(raiz.real)

print(raices_reales)
\end{minted}

  \begin{alertblock}{Criterio práctico}
    Usamos \texttt{abs(raiz.imag) < 1e-10} porque los cálculos numéricos pueden dejar partes imaginarias muy pequeñas.
  \end{alertblock}
\end{frame}

\section{Ajustar Datos con \texttt{np.polyfit}}

\begin{frame}{Ajuste lineal}
  Si esperamos una relación aproximadamente lineal,
  \[
    y = mx + b,
  \]
  podemos estimar la pendiente \(m\) y el intercepto \(b\) desde datos experimentales.

  \begin{block}{Sintaxis}
    \[
      \texttt{m, b = np.polyfit(x, y, deg=1)}
    \]
  \end{block}

  \begin{itemize}
    \item \(m\): pendiente del ajuste.
    \item \(b\): intercepto del ajuste.
    \item \texttt{deg=1}: polinomio de grado 1, es decir, recta.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Flujo mínimo de ajuste}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

m, b = np.polyfit(x, y, deg=1)
y_fit = m*x + b

plt.scatter(x, y, label="datos")
plt.plot(x, y_fit, "r-", label="ajuste")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\end{frame}

\begin{frame}[fragile]{Calidad del ajuste: residuos y \(R^2\)}
  Después de ajustar, conviene evaluar si la recta realmente representa los datos.

\begin{minted}{python}
residuos = y - y_fit

SS_res = np.sum(residuos**2)
SS_tot = np.sum((y - np.mean(y))**2)
R2 = 1 - SS_res/SS_tot

print(f"R2 = {R2:.4f}")
\end{minted}

  \begin{block}{Interpretación}
    \(R^2\) cercano a 1 indica que el modelo explica gran parte de la variación de los datos.
  \end{block}
\end{frame}

\section{Ejercicios Guiados}

\begin{frame}{Ejercicio Guiado 1: \hfill \textcolor{red}{$\clubsuit$} \\ Tiempo de vuelo de un proyectil}
  \begin{block}{Enunciado}
    La altura de un proyectil está dada por
    \[
      h(t) = -4.9t^2 + 18t + 1.5.
    \]
    \begin{itemize}
      \item Resolver \(h(t)=0\) usando \texttt{np.roots}.
      \item Filtrar raíces reales.
      \item Identificar cuál raíz tiene sentido físico.
      \item Interpretar el resultado como tiempo de impacto con el suelo.
    \end{itemize}
  \end{block}

  \textbf{Conceptos}: coeficientes, raíces reales, descarte de raíces no físicas.
\end{frame}

\begin{frame}{Ejercicio Guiado 2: \hfill \textcolor{red}{$\clubsuit$} \\ Calibración de un sensor}
  \begin{block}{Enunciado}
    Un sensor entrega voltaje \(V\) para distintas temperaturas \(T\). Se miden:
    \[
      T = [0, 10, 20, 30, 40, 50]\,^\circ\mathrm{C}
    \]
    \[
      V = [0.52, 0.71, 0.93, 1.11, 1.32, 1.51]\,\mathrm{V}
    \]
    \begin{itemize}
      \item Ajustar \(V = mT + b\) con \texttt{np.polyfit}.
      \item Graficar datos y recta ajustada.
      \item Calcular residuos y \(R^2\).
      \item Interpretar pendiente e intercepto.
    \end{itemize}
  \end{block}
\end{frame}

\begin{solutionblock}
\section{Soluciones de Referencia}

\begin{frame}[fragile]{Solución Guiada 1: \hfill \textcolor{green}{$\checkmark$} \\ Tiempo de vuelo}
\begin{minted}{python}
import numpy as np

# h(t) = -4.9t^2 + 18t + 1.5
coef = [-4.9, 18, 1.5]
raices = np.roots(coef)

raices_reales = []
for r in raices:
    if abs(r.imag) < 1e-10:
        raices_reales.append(r.real)

print("Raíces reales:", raices_reales)

tiempos_fisicos = []
for t in raices_reales:
    if t >= 0:
        tiempos_fisicos.append(t)

print(f"Tiempo de impacto = {tiempos_fisicos[0]:.3f} s")
\end{minted}

  \textbf{Discusión}: La raíz negativa aparece matemáticamente, pero no representa un tiempo posterior al lanzamiento.
\end{frame}

\begin{frame}[fragile]{Solución Guiada 2: \hfill \textcolor{green}{$\checkmark$} \\ Calibración lineal}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

T = np.array([0, 10, 20, 30, 40, 50])
V = np.array([0.52, 0.71, 0.93,
              1.11, 1.32, 1.51])

m, b = np.polyfit(T, V, deg=1)
V_fit = m*T + b

residuos = V - V_fit
SS_res = np.sum(residuos**2)
SS_tot = np.sum((V - np.mean(V))**2)
R2 = 1 - SS_res/SS_tot
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=15]{python}
plt.scatter(T, V, label="datos")
plt.plot(T, V_fit, "r-",
         label="ajuste lineal")
plt.xlabel("Temperatura (°C)")
plt.ylabel("Voltaje (V)")
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

print(f"m = {m:.4f} V/°C")
print(f"b = {b:.4f} V")
print(f"R2 = {R2:.4f}")
\end{minted}
\end{column}
\end{columns}

  \textbf{Discusión}: La pendiente indica sensibilidad del sensor en V/°C.
\end{frame}
\end{solutionblock}

\section{Cierre}

\begin{frame}{Comparación final}
  \begin{block}{\texttt{np.roots}}
    \begin{itemize}
      \item Se usa cuando ya tenemos una ecuación polinómica.
      \item El resultado son valores de la variable independiente.
      \item Requiere interpretar raíces reales, complejas y físicamente válidas.
    \end{itemize}
  \end{block}

  \begin{block}{\texttt{np.polyfit}}
    \begin{itemize}
      \item Se usa cuando tenemos datos experimentales.
      \item El resultado son coeficientes de un modelo.
      \item Requiere evaluar ajuste con gráfica, residuos y \(R^2\).
    \end{itemize}
  \end{block}
\end{frame}
