\section{Repaso Conceptual}

\begin{frame}{Recordatorio: Ajuste Lineal con \texttt{np.polyfit}}
  \begin{block}{Sintaxis básica}
    \texttt{coefs = np.polyfit(x, y, deg=1)} \\
    Retorna coeficientes \([m, b]\) para \(y = mx + b\)
  \end{block}
  
  \begin{itemize}
    \item \textbf{Ventajas:}
      \begin{itemize}
        \item Simple y directo para polinomios
        \item Soporta ponderación con parámetro \texttt{w}
      \end{itemize}
    \item \textbf{Limitaciones:}
      \begin{itemize}
        \item Solo modelos polinomiales
        \item No retorna matriz de covarianza directamente
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Recordatorio: Las Tres Métricas Clave}
  \begin{enumerate}
    \item \textbf{Residuos (\(r_i = y_i - \hat{y}_i\))}:
      \begin{itemize}
        \item Diferencia entre datos y modelo
        \item Deben distribuirse aleatoriamente alrededor de cero
      \end{itemize}
    
    \item \textbf{Coeficiente de determinación (\(R^2 = 1 - \frac{SS_{\text{res}}}{SS_{\text{tot}}}\))}:
      \begin{itemize}
        \item Fracción de varianza explicada por el modelo
        \item \(R^2 \approx 1\): excelente ajuste
      \end{itemize}
    
    \item \textbf{Chi cuadrado reducido (\(\chi^2_{\text{red}} = \frac{1}{\nu}\sum\frac{r_i^2}{\sigma_i^2}\))}:
      \begin{itemize}
        \item Evalúa consistencia con incertidumbres
        \item \(\chi^2_{\text{red}} \approx 1\): ajuste óptimo
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Recordatorio: \texttt{curve\_fit} y Matriz de Covarianza}
  \begin{block}{Sintaxis}
    \texttt{popt, pcov = curve\_fit(func, x, y, sigma=..., absolute\_sigma=True)}
  \end{block}
  
  \begin{itemize}
    \item \textbf{Ventajas sobre polyfit:}
      \begin{itemize}
        \item Funciona con cualquier función (no solo polinomios)
        \item Retorna matriz de covarianza \texttt{pcov}
        \item Permite extraer incertidumbres: \(\sigma_i = \sqrt{\text{pcov}[i,i]}\)
      \end{itemize}
    \item \textbf{Parámetros importantes:}
      \begin{itemize}
        \item \texttt{sigma}: incertidumbres en \(y\)
        \item \texttt{absolute\_sigma=True}: usar valores absolutos
        \item \texttt{p0}: valores iniciales para parámetros
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Péndulo Simple - Periodo vs Longitud}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Un péndulo simple tiene periodo \(T = 2\pi\sqrt{\frac{L}{g}}\).
      \item Para \(g = 9.8\,\text{m/s}^2\), se miden periodos para diferentes longitudes:
        \begin{itemize}
          \item \(L = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]\) m
          \item Generar \(T\) con ruido: \texttt{sigma = 0.05} s
        \end{itemize}
      \item Ajustar el modelo \(T = a\sqrt{L}\) usando \texttt{curve\_fit}.
      \item Calcular \(R^2\) y \(\chi^2_{\text{red}}\).
      \item Comparar \(a\) estimado con el valor teórico \(2\pi/\sqrt{g} \approx 2.007\).
    \end{itemize}
  \end{block}
  
  \textbf{Conceptos}: Modelo no lineal, ajuste con \texttt{curve\_fit}, validación con teoría física.
  \\
  \textbf{Física relevante}: Ley del péndulo simple, relación \(T \propto \sqrt{L}\).
\end{frame}

\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
from scipy.optimize import curve_fit

# Generar datos
g = 9.8  # m/s^2
L = np.array([0.5, 1.0, 1.5, 
              2.0, 2.5, 3.0])
a_true = 2*np.pi/np.sqrt(g)
T_true = a_true * np.sqrt(L)
sigma = 0.05  # s

np.random.seed(42)
T_obs = T_true + np.random.normal(
    0, sigma, size=L.size)

# Definir modelo
def pendulo_model(L, a):
    return a * np.sqrt(L)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=17]{python}
# Ajustar
popt, pcov = curve_fit(
    pendulo_model, L, T_obs,
    sigma=np.full(L.size, sigma),
    absolute_sigma=True)
a_est = popt[0]
sigma_a = np.sqrt(pcov[0, 0])

# Reportar
print(f"a teórico = {a_true:.4f}")
print(f"a estimado = {a_est:.4f} "
      f"± {sigma_a:.4f}")
print(f"Diferencia = "
      f"{abs(a_est-a_true):.4f}")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 1: \hfill \textcolor{green}{$\checkmark$} \\ Péndulo Simple (Parte 2: Métricas)}
\begin{minted}{python}
# Calcular predicciones
T_fit = pendulo_model(L, a_est)

# Calcular R^2
residuos = T_obs - T_fit
SS_res = np.sum(residuos**2)
SS_tot = np.sum((T_obs - np.mean(T_obs))**2)
R2 = 1 - SS_res/SS_tot

# Calcular chi^2 reducido
chi2 = np.sum((residuos/sigma)**2)
nu = L.size - 1  # 1 parámetro ajustado
chi2_red = chi2/nu

print(f"\nMétricas de calidad:")
print(f"R² = {R2:.4f}")
print(f"χ²_red = {chi2_red:.2f}")

if 0.5 < chi2_red < 2.0:
    print("✓ Ajuste consistente con incertidumbres")
\end{minted}
\textbf{Discusión}: \(R^2 \approx 1\) indica excelente ajuste. \(\chi^2_{\text{red}} \approx 1\) confirma que el ruido es consistente con \(\sigma = 0.05\) s.
\end{frame}

p0 = [25, 75, 0.1]
popt, pcov = curve_fit(
    enfriamiento, t, T_obs,
    p0=p0,
    sigma=np.full(t.size, 
                  sigma_T),

T_amb, T0, k = popt

\begin{frame}{Errores Comunes en Ajuste de Datos}
  \begin{block}{Errores frecuentes}
    \begin{itemize}
      \item \textbf{Olvidar \texttt{absolute\_sigma=True}}: Las incertidumbres quedan mal escaladas
      \item \textbf{No verificar residuos}: Patrones sistemáticos indican modelo inadecuado
      \item \textbf{Interpretar mal \(\chi^2_{\text{red}}\)}: Valor muy bajo puede indicar sobreestimación de \(\sigma\)
      \item \textbf{Ignorar la matriz de covarianza}: Perder información sobre correlación entre parámetros
      \item \textbf{No ponderar cuando \(\sigma_i\) varía}: Sesgo hacia puntos con mayor incertidumbre
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Consejo}
    Siempre realizar análisis completo: ajuste → métricas → residuos → interpretación física.
