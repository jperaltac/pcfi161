\section{Elegir la Herramienta}

\begin{frame}{Antes de programar}
  \begin{block}{Pregunta 1}
    ¿El enunciado me entrega una ecuación polinómica y pide encontrar cuándo vale cero?

    \[
      \Rightarrow \texttt{np.roots}
    \]
  \end{block}

  \begin{block}{Pregunta 2}
    ¿El enunciado me entrega datos medidos y pide estimar una relación entre variables?

    \[
      \Rightarrow \texttt{np.polyfit}
    \]
  \end{block}

  \begin{alertblock}{Cuidado}
    Resolver una ecuación y ajustar datos son tareas diferentes, aunque ambas usen polinomios.
  \end{alertblock}
\end{frame}

\section{Ejercicios}

\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Altura de un cohete}
  \begin{block}{Enunciado}
    La altura de un cohete pequeño se modela como
    \[
      h(t) = -4.9t^2 + 32t + 2.
    \]
    \begin{itemize}
      \item Use \texttt{np.roots} para resolver \(h(t)=0\).
      \item Filtre las raíces reales.
      \item Quédese solo con los tiempos positivos.
      \item Reporte el tiempo de caída al suelo.
    \end{itemize}
  \end{block}

  \textbf{Idea}: no todas las raíces matemáticas son físicamente válidas.
\end{frame}

\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Equilibrios de una fuerza}
  \begin{block}{Enunciado}
    Una fuerza unidimensional se aproxima por
    \[
      F(x) = x^3 - 4x.
    \]
    Los puntos de equilibrio cumplen \(F(x)=0\).
    \begin{itemize}
      \item Encuentre las raíces reales con \texttt{np.roots}.
      \item Calcule \(F'(x)=3x^2-4\) en cada raíz.
      \item Clasifique cualitativamente cada equilibrio:
        \begin{itemize}
          \item \(F'(x)<0\): estable para esta convención.
          \item \(F'(x)>0\): inestable para esta convención.
        \end{itemize}
    \end{itemize}
  \end{block}

  \textbf{Idea}: resolver y luego interpretar con una regla adicional.
\end{frame}

\begin{frame}{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Cruce de dos modelos}
  \begin{block}{Enunciado}
    Dos modelos de energía dependen de \(x\):
    \[
      E_1(x)=x^2+2x+1,\qquad E_2(x)=3x+5.
    \]
    \begin{itemize}
      \item Plantee \(E_1(x)-E_2(x)=0\).
      \item Use \texttt{np.roots} para encontrar los puntos de cruce.
      \item Evalúe \(E_1\) y \(E_2\) en las raíces para verificar.
    \end{itemize}
  \end{block}

  \textbf{Idea}: transformar un problema de intersección en una ecuación polinómica.
\end{frame}

\begin{frame}{Ejercicio 4: \hfill \textcolor{red}{$\clubsuit$} \\ Calibración de temperatura}
  \begin{block}{Enunciado}
    Un sensor mide voltaje \(V\) para distintas temperaturas \(T\):
    \[
      T=[0,10,20,30,40,50,60]
    \]
    \[
      V=[0.49,0.70,0.91,1.10,1.31,1.50,1.72].
    \]
    \begin{itemize}
      \item Ajuste \(V=mT+b\) con \texttt{np.polyfit}.
      \item Grafique datos y recta.
      \item Calcule residuos y \(R^2\).
      \item Interprete \(m\).
    \end{itemize}
  \end{block}

  \textbf{Idea}: ajuste lineal e interpretación de pendiente.
\end{frame}

\begin{frame}{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Movimiento con aceleración}
  \begin{block}{Enunciado}
    Se registra la posición de un objeto acelerado:
    \[
      t=[0,1,2,3,4,5]
    \]
    \[
      x=[1.1,3.0,7.2,13.1,21.0,31.4].
    \]
    \begin{itemize}
      \item Ajuste un polinomio de grado 2 con \texttt{np.polyfit}.
      \item Use el modelo \(x(t)=at^2+bt+c\).
      \item Estime la aceleración usando \(a_{\mathrm{fis}}=2a\).
      \item Grafique datos y curva ajustada.
    \end{itemize}
  \end{block}

  \textbf{Idea}: los coeficientes del ajuste pueden tener interpretación física.
\end{frame}

\begin{frame}{Ejercicio 6: \hfill \textcolor{red}{$\clubsuit$} \\ Enfriamiento linealizado}
  \begin{block}{Enunciado}
    Un objeto se enfría según
    \[
      T(t)-T_{\mathrm{amb}} = A e^{-kt}.
    \]
    Si \(T_{\mathrm{amb}}=20^\circ\mathrm{C}\), se miden:
    \[
      t=[0,2,4,6,8,10],\quad T=[80,62,49,39,32,27].
    \]
    \begin{itemize}
      \item Calcule \(y=\ln(T-T_{\mathrm{amb}})\).
      \item Ajuste \(y=mt+b\) con \texttt{np.polyfit}.
      \item Estime \(k=-m\).
      \item Grafique los datos linealizados y el ajuste.
    \end{itemize}
  \end{block}

  \textbf{Idea}: convertir un modelo exponencial en un ajuste lineal.
\end{frame}

\begin{frame}{Ejercicio 7: \hfill \textcolor{red}{$\clubsuit$} \\ Umbral de seguridad}
  \begin{block}{Enunciado}
    Una variable de respuesta \(R\) depende de una dosis \(d\). Se mide:
    \[
      d=[0,1,2,3,4,5,6]
    \]
    \[
      R=[1.2,2.1,3.9,6.8,10.9,16.1,22.0].
    \]
    \begin{itemize}
      \item Ajuste \(R(d)\) con un polinomio de grado 2.
      \item Encuentre con \texttt{np.roots} cuándo el ajuste alcanza \(R=12\).
      \item Filtre soluciones reales dentro del rango medido.
      \item Grafique datos, ajuste y línea horizontal \(R=12\).
    \end{itemize}
  \end{block}

  \textbf{Idea}: combinar ajuste de datos y resolución de ecuaciones.
\end{frame}

\section{Soluciones de Referencia}

\begin{frame}[fragile]{Solución 1: \hfill \textcolor{green}{$\checkmark$} \\ Altura de un cohete}
\begin{minted}{python}
import numpy as np

coef = [-4.9, 32, 2]
raices = np.roots(coef)

tiempos = []
for r in raices:
    if abs(r.imag) < 1e-10 and r.real >= 0:
        tiempos.append(r.real)

print("Raíces:", raices)
print(f"Tiempo físico de impacto = {tiempos[0]:.3f} s")
\end{minted}

  \textbf{Discusión}: se descarta el tiempo negativo porque corresponde a una extrapolación antes del lanzamiento.
\end{frame}

\begin{frame}[fragile]{Solución 2: \hfill \textcolor{green}{$\checkmark$} \\ Equilibrios de una fuerza}
\begin{minted}{python}
import numpy as np

# F(x) = x^3 - 4x
coef = [1, 0, -4, 0]
raices = np.roots(coef)

for r in raices:
    if abs(r.imag) < 1e-10:
        x = r.real
        dF = 3*x**2 - 4
        tipo = "estable" if dF < 0 else "inestable"
        print(f"x = {x:.3f}, F'(x) = {dF:.3f}: {tipo}")
\end{minted}

  \textbf{Discusión}: además de resolver \(F(x)=0\), se usa la derivada para clasificar.
\end{frame}

\begin{frame}[fragile]{Solución 3: \hfill \textcolor{green}{$\checkmark$} \\ Cruce de dos modelos}
\begin{minted}{python}
import numpy as np

def E1(x):
    return x**2 + 2*x + 1

def E2(x):
    return 3*x + 5

# E1 - E2 = x^2 - x - 4
coef = [1, -1, -4]
raices = np.roots(coef)

for r in raices:
    if abs(r.imag) < 1e-10:
        x = r.real
        print(f"x = {x:.3f}")
        print(f"E1 = {E1(x):.3f}, E2 = {E2(x):.3f}")
\end{minted}
\end{frame}

d_fino = np.linspace(0, 6, 100)

plt.scatter(d, R, label="datos")

\section{Cierre}

\begin{frame}{Errores comunes}
  \begin{block}{Con \texttt{np.roots}}
    \begin{itemize}
      \item Escribir los coeficientes en orden incorrecto.
      \item Olvidar coeficientes cero para potencias ausentes.
      \item No filtrar raíces complejas o no físicas.
    \end{itemize}
  \end{block}

  \begin{block}{Con \texttt{np.polyfit}}
    \begin{itemize}
      \item Confundir el orden de salida de los coeficientes.
      \item Graficar solo los datos y no el ajuste.
      \item No revisar residuos ni \(R^2\).
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Lista de verificación}
  \begin{block}{Para resolver ecuaciones}
    \checkmark\ Identificar el polinomio igualado a cero\\
    \checkmark\ Construir la lista de coeficientes\\
    \checkmark\ Usar \texttt{np.roots}\\
    \checkmark\ Filtrar e interpretar raíces
  \end{block}

  \begin{block}{Para ajustar datos}
    \checkmark\ Construir arreglos \texttt{x} e \texttt{y}\\
    \checkmark\ Elegir el grado del polinomio\\
    \checkmark\ Usar \texttt{np.polyfit}\\
    \checkmark\ Graficar, calcular residuos e interpretar
  \end{block}
\end{frame}
