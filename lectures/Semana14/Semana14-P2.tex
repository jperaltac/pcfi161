import numpy as np
from scipy.optimize import curve_fit
import matplotlib.pyplot as plt

# Datos
x = np.array([0.02, 0.05, 0.08,

# Modelo
def hooke(x, k):
    return k * x

# Ajustar
popt, _ = curve_fit(hooke, x, F_obs)
k_est = popt[0]
F_fit = hooke(x, k_est)

# R²
R2 = 1 - np.sum((F_obs - F_fit)**2) / \
         np.sum((F_obs - np.mean(F_obs))**2)

print(f"k = {k_est:.2f} N/m")
print(f"R² = {R2:.4f}")
\end{minted}

import numpy as np
from scipy.optimize import curve_fit

# Datos
T = np.array([300, 400, 500, 600, 
              700, 800, 900, 1000])
c_real = 5.67e-10
sigma_P = (0.5 + 0.001 * T) * 1e-6

# Generar P
np.random.seed(20)
P_true = c_real * T**4
P_obs = P_true + np.random.normal(
    0, 1, size=T.size) * sigma_P

# Modelo
def stefan(T, c):
    return c * T**4
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
# Ajustar con ponderación
popt, pcov = curve_fit(
    stefan, T, P_obs,
    sigma=sigma_P,
    absolute_sigma=True)
c_est = popt[0]
sigma_c = np.sqrt(pcov[0, 0])

# Chi² reducido
P_fit = stefan(T, c_est)
chi2 = np.sum(((P_obs - P_fit) / 
               sigma_P)**2)
chi2_red = chi2 / (T.size - 1)

print(f"c = {c_est:.2e} ± "
      f"{sigma_c:.2e} W/K⁴")
print(f"χ²_red = {chi2_red:.2f}")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 4: \hfill \textcolor{green}{$\checkmark$} \\ Circuito RC (Parte 1)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
from scipy.optimize import curve_fit

# Datos
t = np.array(range(11), dtype=float)
V0_true, tau_true = 12.0, 3.0
sigma_V = 0.3 + 0.02 * t

# Generar V
np.random.seed(25)
V_true = V0_true * np.exp(-t/tau_true)
V_obs = V_true + np.random.normal(
    0, 1, size=t.size) * sigma_V

# Modelo
def rc_decay(t, V0, tau):
    return V0 * np.exp(-t/tau)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
# Ajuste SIN ponderación
popt_sin, _ = curve_fit(
    rc_decay, t, V_obs)
V0_sin, tau_sin = popt_sin

# Ajuste CON ponderación
popt_con, pcov_con = curve_fit(
    rc_decay, t, V_obs,
    sigma=sigma_V,
    absolute_sigma=True)
V0_con, tau_con = popt_con
sigma_tau = np.sqrt(pcov_con[1, 1])

print("Sin ponderación:")
print(f"  τ = {tau_sin:.3f} s")
print("Con ponderación:")
print(f"  τ = {tau_con:.3f} ± "
      f"{sigma_tau:.3f} s")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 4: \hfill \textcolor{green}{$\checkmark$} \\ Circuito RC (Parte 2: Gráficos)}

# Chi² reducido
V_fit_con = rc_decay(t, V0_con, tau_con)

# Gráficos

GM_est, a_est = popt
perr = np.sqrt(np.diag(pcov))

print(f"GM = {GM_est:.3f} ± "

\begin{frame}[fragile]{Solución 6: \hfill \textcolor{green}{$\checkmark$} \\ Selección de Modelo (Parte 1)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
from scipy.optimize import curve_fit

# Datos
t = np.linspace(0, 10, 50)
A, gamma, omega, phi = 5.0, 0.3, 2.0, 0.5
sigma_x = 0.3

# Generar datos
np.random.seed(35)
x_true = A * np.exp(-gamma * t) * \
         np.cos(omega * t + phi)
x_obs = x_true + np.random.normal(
    0, sigma_x, size=t.size)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=13]{python}
# Modelo 1: completo
def modelo1(t, A, gamma, omega, phi):
    return A * np.exp(-gamma * t) * \
           np.cos(omega * t + phi)

# Modelo 2: simple
def modelo2(t, A, gamma):
    return A * np.exp(-gamma * t)

# Ajustar ambos
popt1, _ = curve_fit(
    modelo1, t, x_obs, 
    p0=[5, 0.3, 2, 0.5])
popt2, _ = curve_fit(
    modelo2, t, x_obs, p0=[5, 0.3])
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 6: \hfill \textcolor{green}{$\checkmark$} \\ Selección de Modelo (Parte 2)}
\begin{minted}{python}
# Calcular chi² para cada modelo
x_fit1 = modelo1(t, *popt1)
x_fit2 = modelo2(t, *popt2)

chi2_1 = np.sum(((x_obs - x_fit1) / sigma_x)**2)
chi2_2 = np.sum(((x_obs - x_fit2) / sigma_x)**2)

# Calcular AIC
n = t.size
k1, k2 = 4, 2  # número de parámetros
AIC1 = 2*k1 + n*np.log(chi2_1/n)
AIC2 = 2*k2 + n*np.log(chi2_2/n)

delta_AIC = AIC2 - AIC1

print(f"Modelo 1 (completo): AIC = {AIC1:.2f}, χ² = {chi2_1:.2f}")
print(f"Modelo 2 (simple): AIC = {AIC2:.2f}, χ² = {chi2_2:.2f}")
print(f"ΔAIC = {delta_AIC:.2f}")

if delta_AIC > 10:
    print("→ Modelo 1 es significativamente mejor")
elif delta_AIC < -10:
    print("→ Modelo 2 es significativamente mejor")
else:
    print("→ Evidencia no concluyente")
\end{minted}
\textbf{Discusión}: AIC penaliza complejidad, favorece modelo que equilibra ajuste y parsimonia.
\end{frame}

\begin{frame}{Errores Comunes en Ajuste Avanzado}
  \begin{block}{Errores frecuentes}
    \begin{itemize}
      \item \textbf{No usar \texttt{absolute\_sigma=True}}: incertidumbres mal escaladas
      \item \textbf{Olvidar valores iniciales (\texttt{p0})}: modelos complejos no convergen
      \item \textbf{Ignorar análisis de residuos}: perder patrones sistemáticos
      \item \textbf{Interpretar mal \(\chi^2_{\text{red}} \gg 1\)}: puede indicar subestimación de \(\sigma\)
      \item \textbf{Sobre-ajustar sin criterio objetivo}: AIC ayuda a evitarlo
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Consejo}
    Siempre: ajuste → métricas → residuos → interpretación física → decisión informada.
