\section{Recordatorio de NumPy}
\begin{frame}[fragile]{Crear arreglos}
\textbf{Herramientas mínimas para hoy.}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np

a = np.array([1, 2, 3, 4])
x = np.linspace(0, 10, 100)
n = np.arange(0, 20, 2)
\end{minted}
\begin{itemize}
  \item \texttt{np.array}: transforma datos en arreglo.
  \item \texttt{np.linspace}: puntos igualmente espaciados.
  \item \texttt{np.arange}: secuencia con paso fijo.
\end{itemize}
\end{frame}

% 5 ──────────────────────────────────────────────────────────
\begin{frame}[fragile]{Operaciones vectorizadas}
\textbf{Idea clave.} Operar sobre todo el arreglo sin escribir un \texttt{for}.
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
t = np.linspace(0, 5, 50)
x0 = 2.0
v0 = 3.0
a  = -1.5

x = x0 + v0*t + 0.5*a*t**2

print(x.mean(), x.min(), x.max())
\end{minted}
\textit{Punto docente.} Las operaciones se aplican elemento a elemento.
\end{frame}

\section{Recordatorio de Matplotlib}

\section{Ejercicios guiados}
\begin{frame}{Ejercicio guiado 1: Movimiento 1D}
\begin{block}{Enunciado}
\begin{itemize}
  \item Crea un arreglo \texttt{t} con 100 valores entre 0 y 8 segundos.
  \item Usa \(x(t)=x_0+v_0t+\frac{1}{2}at^2\), con \(x_0=0\), \(v_0=12\) y \(a=-9.8\).
  \item Grafica posición versus tiempo.
  \item Calcula la altura máxima usando NumPy.
\end{itemize}
\end{block}
\textbf{Conceptos:} \texttt{linspace}, operaciones vectorizadas, \texttt{max}, \texttt{plot}.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 1: \hfill \textcolor{green}{$\checkmark$} \\ Movimiento 1D}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
t = np.linspace(0, 8, 100)
x0 = 0
v0 = 12
a = -9.8
x = x0 + v0*t + 0.5*a*t**2
altura_maxima = np.max(x)
print("Altura maxima:", altura_maxima)
plt.plot(t, x)
plt.xlabel("tiempo [s]")
plt.ylabel("posicion [m]")
plt.title("Movimiento 1D")
plt.grid(True)
plt.show()
\end{minted}
\textbf{Discusión:} El arreglo \texttt{t} contiene todos los tiempos y la formula se aplica a todos sus valores al mismo tiempo.
\end{frame}
\end{solutionblock}

\begin{frame}{Ejercicio guiado 2: Trayectoria 2D}
\begin{block}{Enunciado}
\begin{itemize}
  \item Crea un arreglo \texttt{theta} entre 0 y \(2\pi\).
  \item Define \(r=2+\cos(\theta)\).
  \item Calcula \(x=r\cos(\theta)\) e \(y=r\sin(\theta)\).
  \item Grafica la trayectoria y usa \texttt{plt.axis("equal")}.
\end{itemize}
\end{block}
\textbf{Conceptos:} coordenadas polares, \texttt{np.cos}, \texttt{np.sin}, ejes proporcionales.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 2: \hfill \textcolor{green}{$\checkmark$} \\ Trayectoria 2D}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
theta = np.linspace(0, 2*np.pi, 200)
r = 2 + np.cos(theta)
x = r*np.cos(theta)
y = r*np.sin(theta)
plt.plot(x, y)
plt.xlabel("x")
plt.ylabel("y")
plt.title("Trayectoria 2D")
plt.axis("equal")
plt.grid(True)
plt.show()
\end{minted}
\textbf{Discusión:} Primero se calcula el radio y luego se transforma cada punto desde coordenadas polares a coordenadas cartesianas.
\end{frame}
\end{solutionblock}

\begin{frame}{Ejercicio guiado 3: Datos con ruido}
\begin{block}{Enunciado}
\begin{itemize}
  \item Genera 200 mediciones alrededor de 5.0 usando \texttt{np.random.normal}.
  \item Calcula media y desviación estándar.
  \item Dibuja un histograma de las mediciones.
  \item Marca la media con una línea vertical.
\end{itemize}
\end{block}
\textbf{Conceptos:} números aleatorios, estadística descriptiva, histogramas.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 3: \hfill \textcolor{green}{$\checkmark$} \\ Datos con ruido}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
np.random.seed(1)
mediciones = np.random.normal(5.0, 0.3, 200)
media = np.mean(mediciones)
desviacion = np.std(mediciones)
print("Media:", media)
print("Desviacion estandar:", desviacion)
plt.hist(mediciones, bins=15, edgecolor="black")
plt.axvline(media, color="red", label="media")
plt.xlabel("medicion")
plt.ylabel("frecuencia")
plt.title("Mediciones con ruido")
plt.legend()
plt.show()
\end{minted}
\textbf{Discusión:} La media indica el valor central de las mediciones y la desviación estándar indica cuánto se dispersan.
\end{frame}
\end{solutionblock}

\begin{frame}{Ejercicio guiado 4: Relación física}
\begin{block}{Enunciado}
\begin{itemize}
  \item Crea arreglos para distancia \(d\) y velocidad \(v\).
  \item Calcula \(H=v/d\) para cada dato.
  \item Grafica \(v\) versus \(d\) usando \texttt{scatter}.
  \item Calcula el promedio de \(H\).
\end{itemize}
\end{block}
\textbf{Conceptos:} división elemento a elemento, \texttt{scatter}, promedio.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución Guiado 4: \hfill \textcolor{green}{$\checkmark$} \\ Relación física}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
d = np.array([2, 4, 6, 8, 10])
v = np.array([140, 260, 430, 540, 700])
H = v/d
promedio_H = np.mean(H)
print("Valores de H:", H)
print("Promedio de H:", promedio_H)
plt.scatter(d, v)
plt.xlabel("distancia")
plt.ylabel("velocidad")
plt.title("Velocidad versus distancia")
plt.grid(True)
plt.show()
\end{minted}
\textbf{Discusión:} Al dividir dos arreglos del mismo tamaño, NumPy divide cada velocidad por su distancia correspondiente.
\end{frame}
\end{solutionblock}

\section{Ejercicios tipo Solemne}
\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Altura de un proyectil}
\begin{block}{Enunciado}
\begin{itemize}
  \item Construye \(t\) entre 0 y 4 segundos con 80 puntos.
  \item Calcula \(y(t)=1.5+18t-4.9t^2\).
  \item Grafica \(y\) versus \(t\), con etiquetas y grilla.
  \item Informa la altura máxima y el tiempo donde ocurre.
\end{itemize}
\end{block}
\textbf{Conceptos:} arreglo temporal, vectorización, \texttt{np.max}, \texttt{np.argmax}.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 1 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Altura de un proyectil}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
t = np.linspace(0, 4, 80)
y = 1.5 + 18*t - 4.9*t**2
altura_maxima = np.max(y)
indice_maximo = np.argmax(y)
tiempo_maximo = t[indice_maximo]
print("Altura maxima:", altura_maxima)
print("Tiempo de altura maxima:", tiempo_maximo)
plt.plot(t, y)
plt.xlabel("tiempo [s]")
plt.ylabel("altura [m]")
plt.title("Altura de un proyectil")
plt.grid(True)
plt.show()
\end{minted}
\textbf{Discusión:} \texttt{np.argmax} entrega la posición del valor máximo; con esa posición se obtiene el tiempo correspondiente.
\end{frame}
\end{solutionblock}

\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Órbita circular perturbada}
\begin{block}{Enunciado}
\begin{itemize}
  \item Crea \(\theta\) entre 0 y \(2\pi\) con 360 puntos.
  \item Define \(r=1+0.15\sin(5\theta)\).
  \item Calcula \(x=r\cos(\theta)\) e \(y=r\sin(\theta)\).
  \item Grafica la órbita usando \texttt{axis("equal")}.
\end{itemize}
\end{block}
\textbf{Conceptos:} trigonometría con NumPy, trayectoria 2D, proporción de ejes.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Órbita circular perturbada}
\begin{minted}[fontsize=\scriptsize,breaklines,breakanywhere,baselinestretch=0.78]{python}
import numpy as np
import matplotlib.pyplot as plt
theta = np.linspace(0, 2*np.pi, 360)
r = 1 + 0.15*np.sin(5*theta)
x = r*np.cos(theta)
y = r*np.sin(theta)
plt.plot(x, y)
plt.xlabel("x")
plt.ylabel("y")
plt.title("Orbita circular perturbada")
plt.axis("equal")
plt.grid(True)
plt.show()
\end{minted}
\textbf{Discusión:} La parte \texttt{0.15*np.sin(5*theta)} modifica suavemente el radio y produce la perturbación de la órbita.
\end{frame}
\end{solutionblock}

\begin{block}{Enunciado}
\begin{itemize}
  \item Simula 300 mediciones de una longitud promedio \(L=2.4\) m.
  \item Usa una dispersión de \(0.08\) m.
  \item Calcula media, desviación estándar, mínimo y máximo.
  \item Dibuja un histograma e interpreta la dispersión.

\begin{solutionblock}

import numpy as np

print("Desviacion estandar:", desviacion)

plt.ylabel("frecuencia")
plt.title("Mediciones de longitud")
plt.show()

\end{frame}
\end{solutionblock}

\section{Errores comunes}
\begin{frame}{Errores comunes a evitar}
\begin{itemize}
  \item Usar \texttt{math.sin} o \texttt{math.cos} sobre arreglos en vez de \texttt{np.sin} o \texttt{np.cos}.
  \item Olvidar \texttt{plt.show()} al final del gráfico.
  \item No usar \texttt{plt.axis("equal")} en trayectorias 2D.
  \item Mezclar listas y arrays sin revisar el tipo de dato.
  \item Entregar gráficos sin título, ejes o unidades.
\end{itemize}
\end{frame}
