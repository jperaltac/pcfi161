\section{Ejercicios Prácticos Integrados}

% ------------------------------------------------------------------------
% Ejercicio 1: Inspección
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Lectura e inspección inicial}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Lee el archivo con \texttt{pd.read\_csv("https://raw.githubusercontent.com/mwaskom/seaborn-data/master/planets.csv")}.
      \item Muestra \texttt{df.head()}, \texttt{df.shape}, \texttt{df.columns} y \texttt{df.info()}.
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} Lectura de CSV e inspección inicial.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 1 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Lectura e inspección inicial}
\begin{minted}{python}
import pandas as pd

df = pd.read_csv("https://raw.githubusercontent.com/mwaskom/seaborn-data/master/planets.csv")

print(df.head())
print("shape:", df.shape)
print("columnas:", df.columns.tolist())
print("="*40)
df.info()
\end{minted}
\textbf{Discusión:} Antes de analizar cualquier dataset conviene ver su estructura general y sus columnas disponibles.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 2: Limpieza e imputación
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Selección de columnas y descripción}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Selecciona un subconjunto de columnas relevantes.
      \item Muestra \texttt{df.describe()} para las columnas numéricas del dataset.
      \item Indica cuáles columnas son numéricas y cuáles no.
    \end{itemize}
  \end{block}
  \textbf{Objetivo}: Reconocer tipos de columnas y obtener un resumen estadístico básico.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Selección de columnas y descripción}
\begin{minted}{python}
subset = df.iloc[:, :4]
print(subset.head())

print(df.describe())

num_cols = df.select_dtypes(include='number').columns.tolist()
non_num_cols = df.select_dtypes(exclude='number').columns.tolist()
print("Numéricas:", num_cols)
print("No numéricas:", non_num_cols)
\end{minted}
\textbf{Discusión:} Distinguir tipos de columnas ayuda a decidir qué operaciones tienen sentido sobre cada variable.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 3: Derivaciones físicas
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Filtrado simple y múltiple}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Filtra el DataFrame con una condición simple sobre una columna numérica.
      \item Luego aplica un filtrado múltiple con dos o tres condiciones usando \texttt{\&}.
      \item Muestra el subconjunto resultante.
    \end{itemize}
  \end{block}
  \textbf{Conceptos}: Máscaras booleanas y selección.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 3 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Filtrado simple y múltiple}
\begin{minted}{python}
filtro_simple = df[df['temperatura'] > 300]
print(filtro_simple.head())

filtro_multiple = df[
    (df['temperatura'] > 300) &
    (df['masa'] > 1.0)
]
print(filtro_multiple)
\end{minted}
\textbf{Discusión:} El filtrado es una de las operaciones más frecuentes cuando se trabaja con tablas de datos.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 4: Filtrado físico
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 4: \hfill \textcolor{red}{$\clubsuit$} \\ Datos faltantes}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Cuenta valores faltantes por columna con \texttt{isnull()}.
      \item Crea un DataFrame limpio con \texttt{dropna()}.
      \item Compara cuántas filas había antes y después de limpiar.
    \end{itemize}
  \end{block}
  \textbf{Objetivo}: Identificar y eliminar filas incompletas.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 4 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Datos faltantes}
\begin{minted}{python}
print(df.isnull().sum())

df_clean = df.dropna()

print("Filas originales:", len(df))
print("Filas sin nulos:", len(df_clean))
\end{minted}
\textbf{Discusión:} Primero se detectan los nulos y luego se decide si conviene eliminarlos o tratarlos de otra forma.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 5: Agrupación y tasas
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Eliminar columnas no necesarias}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Elimina una o más columnas que no sean necesarias para el análisis inmediato.
      \item Guarda el resultado en un nuevo DataFrame.
      \item Muestra las columnas que quedaron.
    \end{itemize}
  \end{block}
  \textbf{Conceptos}: Uso de \texttt{drop(columns=[...])}.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 5 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Eliminar columnas no necesarias}
\begin{minted}{python}
df_reducido = df.drop(columns=['comentario'])
print(df_reducido.columns.tolist())
\end{minted}
\textbf{Discusión:} Quitar columnas irrelevantes deja un DataFrame más claro y manejable para el análisis.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 6: Visualización
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 6: \hfill \textcolor{red}{$\clubsuit$} \\ Agrupación básica}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Agrupa el DataFrame por una columna categórica.
      \item Calcula la media de una o dos columnas numéricas para cada grupo.
      \item Muestra el resultado.
    \end{itemize}
  \end{block}
  \textbf{Objetivo}: Resumir información por grupos usando pandas.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 6 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Agrupación básica}
\begin{minted}{python}
resumen = df.groupby('categoria')[['temperatura', 'masa']].mean()
print(resumen)
\end{minted}
\textbf{Discusión:} \texttt{groupby().mean()} permite resumir un dataset de forma compacta y muy útil.
\end{frame}
\end{solutionblock}

% ------------------------------------------------------------------------
% Ejercicio 7: Exportación
% ------------------------------------------------------------------------
\begin{frame}{Ejercicio 7: \hfill \textcolor{red}{$\clubsuit$} \\ Visualización y exportación}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Dibuja un histograma simple de una columna numérica.
      \item Crea un gráfico de barras con conteos por categoría.
      \item Exporta un DataFrame limpio o filtrado a un archivo CSV.
    \end{itemize}
  \end{block}
  \textbf{Objetivo}: Integrar visualización rápida y persistencia de resultados.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile]{Solución 7 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Visualización y exportación}
\begin{minted}{python}
import matplotlib.pyplot as plt

df['temperatura'].hist(bins=20)
plt.title('Histograma de temperatura')
plt.xlabel('Temperatura')
plt.ylabel('Frecuencia')
plt.show()

df['categoria'].value_counts().plot(kind='bar')
plt.title('Conteo por categoría')
plt.ylabel('Cantidad')
plt.show()

df_clean.to_csv('experimento_limpio.csv', index=False)
\end{minted}
\textbf{Discusión:} Se combinan dos tareas frecuentes del análisis exploratorio: visualizar rápidamente y guardar resultados.
\end{frame}
\end{solutionblock}
