\begin{frame}{Tema: Exoplanetas detectados por tránsito}

\begin{frame}[fragile]{Generación del dataset sintético: \hfill \textcolor{green}{$\checkmark$} \\ Exoplanetas por tránsito}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import pandas as pd

rng = np.random.default_rng(42)
n = 1200

# Estrellas
star_teff = np.clip(
    rng.normal(5700, 800, n), 3500, 7500)  # K
star_radius_Rsun = np.clip(rng.lognormal(mean=np.log(1.0), sigma=0.25, size=n), 0.6, 2.5)  # R_sun
star_metallicity = rng.normal(0.0, 0.2, n)  # [Fe/H] dex
star_mass_Msun = np.clip(rng.lognormal(np.log(1.0), 0.20, n), 0.5, 1.6)  # M_sun

# Planetas
# Periodo ~ lognormal (mediana ~20 d)
period_days = rng.lognormal(
    mean=np.log(20), sigma=0.9, size=n
)

# Radio planetario con cola pesada (pareto + esc.)
rp_base = 0.8 + rng.pareto(1.6, n)  # >0.8
planet_radius_Re = np.clip(rp_base, 0.5, 15.0)

# Geometría de tránsito simple
b = rng.uniform(0, 1, n)  # impacto
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=24]{python}
# Ley de Kepler en unidades astronómicas (aprox):
# a[AU] ≈ (P/365.25)^(2/3) * M_star^(1/3)
a_AU = (period_days/365.25)**(2/3) * (star_mass_Msun**(1/3))

# Profundidad de tránsito (ppm) ~ (Rp/R*)^2
Rsun_to_Re = 109.1
depth_ppm = ((planet_radius_Re / (star_radius_Rsun * Rsun_to_Re))**2) * 1e6

# SNR aprox: depth/sigma * sqrt(Ntr), sigma ~ lognormal
sigma_ppm = rng.lognormal(mean=np.log(300), sigma=0.4, size=n)
Ntr = np.maximum(1, (3*365.25)/period_days)  # 3 años misión
snr = depth_ppm / sigma_ppm * np.sqrt(Ntr)

# Dataset base
df = pd.DataFrame({
    "id": [f"EP{idx:04d}" for idx in range(1, n+1)],
    "star_teff": star_teff,
    "star_radius_Rsun": star_radius_Rsun,
    "star_metallicity": star_metallicity,
    "star_mass_Msun": star_mass_Msun,
    "period_days": period_days,
    "planet_radius_Re": planet_radius_Re,
    "b": b,
    "a_AU": a_AU,
    "depth_ppm": depth_ppm,
    "snr": snr
})

# Datos faltantes: MCAR/MNAR
mask_mcar_teff = rng.random(n) < 0.10
df.loc[mask_mcar_teff, "star_teff"] = np.nan

mask_mnar_rp = df["snr"] < 7
drop_prob = rng.random(n) < 0.25
df.loc[mask_mnar_rp & drop_prob, "planet_radius_Re"] = np.nan

mask_mar_met = (df["star_teff"] < 4300) & (rng.random(n) < 0.20)
df.loc[mask_mar_met, "star_metallicity"] = np.nan
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
\textbf{Discusión:} Distribuciones realistas aproximadas y faltantes MCAR/MNAR para practicar imputación. La SNR depende de profundidad y número de tránsitos.
\end{frame}

\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Limpieza y datos faltantes}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Cuenta nulos por columna y porcentaje.
      \item Imputa \texttt{star\_teff} con la mediana por cuartiles de \texttt{star\_radius\_Rsun}.
      \item Imputa \texttt{planet\_radius\_Re} usando profundidad: \(\hat{R_p} \approx \sqrt{\texttt{depth\_ppm}/10^6}\cdot R_\star \cdot 109.1\).
      \item Imputa \texttt{star\_metallicity} con la mediana global.
    \end{itemize}
  \end{block}
  \textbf{Objetivo}: Practicar estrategias basadas en física y en agrupaciones.
\end{frame}

\begin{frame}[fragile]{Solución 2 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Limpieza y datos faltantes}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
# Resumen de nulos
nulls = df.isnull().sum().sort_values(ascending=False)
pct = df.isnull().mean().round(3)
print(nulls); print(pct)

# Imputación star_teff por cuartiles de radio estelar
bins = pd.qcut(df['star_radius_Rsun'], 4, duplicates='drop')
teff_med = df.groupby(bins)['star_teff'].transform(
    lambda s: s.fillna(s.median())
)
df['star_teff'] = df['star_teff'].fillna(teff_med)

# Imputación planet_radius_Re desde profundidad
# Rp[Re] ≈ sqrt(depth/1e6) * Rstar[Rsun] * 109.1
need_rp = df['planet_radius_Re'].isna() & df['depth_ppm'].notna() & df['star_radius_Rsun'].notna()
Rp_est = np.sqrt(df.loc[need_rp, 'depth_ppm'] / 1e6) * df.loc[need_rp, 'star_radius_Rsun'] * 109.1
df.loc[need_rp, 'planet_radius_Re'] = Rp_est
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=20]{python}
# Imputación metallicity con mediana global
met_med = df['star_metallicity'].median()
df['star_metallicity'] = df['star_metallicity'].fillna(met_med)

# Verificación
print(df.isnull().sum())
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
\textbf{Discusión:} La imputación de Rp usa la relación física del tránsito; el resto usa medianas robustas por grupos.
\end{frame}

\begin{frame}{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Columnas derivadas físicas}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Recalcula \texttt{a\_AU} con Kepler: \(a \propto P^{2/3} M_\star^{1/3}\).
      \item Calcula \(\texttt{teq\_K}\) con albedo 0.3: \(\mathrm{T}_{eq}\approx T_{eff}\sqrt{\frac{R_\star}{2a}}\cdot(1-0.3)^{1/4}\), usando \(R_\odot/\mathrm{AU}=0.00465047\).
      \item Clasifica \texttt{planet\_type}: \textit{Terreno}(<2 R$_\oplus$), \textit{Sub-Neptuno}(2–4), \textit{Gigante}(>4).
    \end{itemize}
  \end{block}
  \textbf{Conceptos}: Uso de unidades y umbrales astrofísicos comunes.
\end{frame}

\begin{frame}[fragile]{Solución 3 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Columnas derivadas físicas}
\begin{minted}{python}
# a_AU (recalcular por consistencia)
df['a_AU'] = (df['period_days']/365.25)**(2/3) * (df['star_mass_Msun']**(1/3))

# Temperatura de equilibrio
Rsun_AU = 0.00465047
geom = (df['star_radius_Rsun'] * Rsun_AU) / (2.0 * df['a_AU'])
albedo = 0.30
df['teq_K'] = df['star_teff'] * np.sqrt(geom) * ((1 - albedo)**0.25)

# Clasificación por tamaño
def classify_rp(r):
    if r < 2.0: return 'Terreno'
    if r <= 4.0: return 'Sub-Neptuno'
    return 'Gigante'
df['planet_type'] = df['planet_radius_Re'].apply(classify_rp)
\end{minted}
\textbf{Discusión:} La \(\mathrm{T}_{eq}\) es una cota simplificada; útil para filtros de “zona habitable aproximada”.
\end{frame}

\begin{frame}{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Tasas por metalicidad y tipo de planeta}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Discretiza \texttt{star\_metallicity} en cuartiles.
      \item Calcula proporción de \textit{Terreno}/\textit{Sub-Neptuno}/\textit{Gigante} por cuartil.
      \item Interpreta tendencia: ¿mayor metalicidad favorece gigantes?
    \end{itemize}
  \end{block}
  \textbf{Conceptos}: \texttt{pd.cut}, \texttt{groupby}, proporciones.
\end{frame}

\begin{frame}[fragile]{Solución 5 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Tasas por metalicidad y tipo}
\begin{minted}{python}
met_bins = pd.qcut(df['star_metallicity'], 4, labels=['Q1','Q2','Q3','Q4'])
tab = (df
       .groupby([met_bins,'planet_type'])
       .size()
       .groupby(level=0)
       .apply(lambda s: s/s.sum())
       .unstack(fill_value=0)
)
print(tab.round(3))
\end{minted}
\textbf{Discusión:} Suele observarse mayor fracción de gigantes a metalicidades altas (tendencia plausible).
\end{frame}
