\begin{minted}{python}
import pandas as pd
url = 'https://raw.githubusercontent.com/datasciencedojo/datasets/master/titanic.csv'
df = pd.read_csv(url)
print(df.head(10))
print(df.columns)

\begin{frame}[fragile]{Solución 4 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ POO y pandas}
\begin{minted}{python}
import math
class Particle:
    def __init__(self, mass, x, y, vx, vy):
        self.mass = mass
        self.x = x
        self.y = y
        self.vx = vx
        self.vy = vy
    def kinetic_energy(self):
        v2 = self.vx**2 + self.vy**2
        return 0.5 * self.mass * v2

# Supón que tienes un DataFrame dfp con columnas: mass, x, y, vx, vy
particles = [Particle(row['mass'], row['x'], row['y'], row['vx'], row['vy']) for idx, row in dfp.iterrows()]
dfp['E_cinetica'] = [p.kinetic_energy() for p in particles]
\end{minted}
