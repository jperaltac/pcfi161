\section{Introducción y Dataset}

\begin{frame}{Dataset del Titanic}
  \begin{block}{Información del Dataset}
    \begin{itemize}
      \item \textbf{Fuente:} \url{https://raw.githubusercontent.com/datasciencedojo/datasets/master/titanic.csv}
      \item \textbf{Contenido:} Datos de pasajeros del Titanic (891 registros)
      \item \textbf{Variables principales:}
        \begin{itemize}
          \item \texttt{Fare}: Precio del boleto
          \item \texttt{Pclass}: Clase de pasajero (1, 2, 3)
          \item \texttt{Sex}: Género (male, female)
          \item \texttt{Survived}: Supervivencia (0=No, 1=Sí)
          \item \texttt{SibSp}: Número de hermanos/esposas a bordo
          \item \texttt{Parch}: Número de padres/hijos a bordo
          \item \texttt{Ticket}: Número de ticket
          \item \texttt{Cabin}: Cabina asignada
          \item \texttt{Embarked}: Puerto de embarque (C = Cherburgo, Q = Queenstown, S = Southampton)
        \end{itemize}
    \end{itemize}
  \end{block}
  \textbf{Objetivo:} Analizar y visualizar patrones en los datos históricos del Titanic.
\end{frame}


\begin{frame}{Problema Principal - Parte 1: \hfill \textcolor{red}{$\clubsuit$} \\ Análisis con Pandas}
  \begin{block}{Paso 1: Análisis Demográfico}
    \begin{itemize}
      \item Calcular la proporción de pasajeros por género y clase
      \item Determinar la edad media por puerto de embarque
      \item Analizar la relación entre SibSp y Parch
      \item Crear una nueva columna familia\_total sumando SibSp y Parch
      \item Identificar los grupos familiares más grandes
    \end{itemize}
  \end{block}
  
  \begin{block}{Paso 2: Análisis Económico}
    \begin{itemize}
      \item Calcular la tarifa promedio por puerto de embarque
      \item Identificar outliers en las tarifas usando IQR
      \item Crear categorías de tarifas (económica, media, premium)
      \item Relacionar tarifas con tasas de supervivencia
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Parte 1: Continuación \hfill \textcolor{blue}{$\bigstar$} \\ Análisis Avanzado}
  \begin{block}{Paso 3: Correlaciones y Patrones}
    \begin{itemize}
      \item Crear matriz de correlación entre variables numéricas
      \item Analizar la relación entre clase y tamaño familiar
      \item Estudiar patrones de supervivencia por cabina
      \item Identificar títulos (Mr., Mrs., etc.) del campo Name
    \end{itemize}
  \end{block}
  
  \begin{block}{Paso 4: Análisis de Supervivencia}
    \begin{itemize}
      \item Calcular supervivencia por rango de edad
      \item Analizar impacto del tamaño familiar en supervivencia
      \item Estudiar supervivencia por título y género
      \item Crear un índice de privilegio basado en clase y tarifa
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Problema Principal - Parte 2: \hfill \textcolor{red}{$\clubsuit$} \\ Visualización Avanzada}
  \begin{block}{Paso 1: Gráficos Multivariables}
    \begin{itemize}
      \item Crear un scatter plot 3D de edad, tarifa y supervivencia
      \item Generar subplots para análisis por puerto de embarque
      \item Diseñar un heatmap de correlaciones
      \item Visualizar distribución de edades por título
    \end{itemize}
  \end{block}
  
  \begin{block}{Paso 2: Visualizaciones Especializadas}
    \begin{itemize}
      \item Crear un violin plot de tarifas por clase y supervivencia
      \item Generar un treemap de pasajeros por clase y género
      \item Diseñar gráficos de waffle para proporciones de supervivencia
      \item Visualizar patrones de cabinas en un diagrama personalizado
    \end{itemize}
  \end{block}
\end{frame}
