\section{Introducción y Dataset}

\begin{frame}{Dataset del Titanic}
  \begin{block}{Información del Dataset}
    \begin{itemize}
      \item \textbf{Fuente:} \url{https://raw.githubusercontent.com/datasciencedojo/datasets/master/titanic.csv}
      \item \textbf{Contenido:} Datos de pasajeros del Titanic (891 registros)
      \item \textbf{Variables principales:}
        \begin{itemize}
          \item \texttt{Fare}: Precio del boleto
          \item \texttt{Pclass}: Clase de pasajero (1, 2, 3)
          \item \texttt{Sex}: Género (male, female)
          \item \texttt{Survived}: Supervivencia (0=No, 1=Sí)
          \item \texttt{SibSp}: Número de hermanos/esposas a bordo
          \item \texttt{Parch}: Número de padres/hijos a bordo
          \item \texttt{Ticket}: Número de ticket
          \item \texttt{Cabin}: Cabina asignada
          \item \texttt{Embarked}: Puerto de embarque (C = Cherburgo, Q = Queenstown, S = Southampton)
        \end{itemize}
    \end{itemize}
  \end{block}
  \textbf{Objetivo:} Analizar y visualizar patrones en los datos históricos del Titanic.
\end{frame}


\begin{frame}{Problema Principal - Parte 1: \hfill \textcolor{red}{$\clubsuit$} \\ Análisis con Pandas}

  \begin{block}{Paso 2: Análisis Adicional}
