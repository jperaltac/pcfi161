\section{Fundamentos de Pandas y POO}
\begin{frame}{¿Por qué Pandas para Análisis de Datos?}
  \begin{itemize}
    \item \textbf{DataFrames}: estructuras tabulares similares a Excel/hojas de cálculo
    \item \textbf{Lectura}: archivos CSV, Excel, JSON automáticamente
    \item \textbf{Operaciones}: filtrado, agrupación, estadísticas descriptivas
    \item \textbf{Limpieza}: manejo de datos faltantes, duplicados
    \item \textbf{Integración}: funciona perfectamente con NumPy y Matplotlib
  \end{itemize}
  
  \begin{alertblock}{Para Física/Astronomía}
    Ideal para analizar datos experimentales, observaciones astronómicas, resultados de simulaciones.
  \end{alertblock}
\end{frame}

\begin{frame}{¿Qué es pandas?}
  \begin{itemize}
    \item \textbf{pandas} es una biblioteca de Python para análisis y manipulación eficiente de datos.
    \item Permite trabajar con datos tabulares (tipo Excel) de forma sencilla y poderosa.
    \item Ofrece estructuras clave: \texttt{Series} (vectores) y \texttt{DataFrame} (tablas).
    \item Esencial para ciencia de datos, física computacional y astronomía.
  \end{itemize}
  \begin{block}{Ventajas}
    \begin{itemize}
      \item Lectura y escritura de archivos (CSV, Excel, etc.)
      \item Filtrado, selección y limpieza de datos
      \item Estadísticas y agrupaciones rápidas
      \item Integración con NumPy y Matplotlib
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Integración con POO: Clase Particle}
\begin{minted}{python}
import math

class Particle:
    def __init__(self, mass, x, y, vx, vy):
        self.mass = mass
        self.x = x
        self.y = y
        self.vx = vx
        self.vy = vy
    
    def kinetic_energy(self):
        """Retorna la energía cinética de la partícula."""
        v2 = self.vx**2 + self.vy**2
        return 0.5 * self.mass * v2
    
    def distance(self, other):
        """Distancia a otra partícula."""
        dx = self.x - other.x
        dy = self.y - other.y
        return math.sqrt(dx**2 + dy**2)
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Slide: Ejemplo de uso de la clase Particle
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Ejemplo de uso: \hfill \textcolor{green}{$\checkmark$} \\ Clase Particle}
\begin{minted}{python}
# Crear dos partículas con diferentes posiciones y velocidades
p1 = Particle(mass=2.0, x=0, y=0, vx=3, vy=4)
p2 = Particle(mass=1.5, x=1, y=2, vx=0, vy=5)

# Calcular energía cinética de cada partícula
print("Energía cinética p1:", p1.kinetic_energy())
print("Energía cinética p2:", p2.kinetic_energy())

# Calcular distancia entre ambas partículas
dist = p1.distance(p2)
print("Distancia entre p1 y p2:", dist)
\end{minted}
\textbf{Discusión:} Este ejemplo muestra cómo crear objetos \texttt{Particle}, calcular su energía cinética y la distancia entre ellas, integrando conceptos de POO y física.
\end{frame}
