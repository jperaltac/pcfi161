\section{Recursividad: Fundamento de Algoritmos}

% ------------------------------------------------------------------------
% Slide 5: ¿Qué es la recursividad?
% ------------------------------------------------------------------------
\begin{frame}{¿Qué es la Recursividad?}
  \begin{block}{Definición}
    Una función es \textbf{recursiva} cuando se llama a sí misma para resolver 
    un problema dividiéndolo en subproblemas más pequeños.
  \end{block}
  
  \vspace{0.3cm}
  \textbf{Estructura básica de una función recursiva:}
  \begin{enumerate}
    \item \textbf{Caso base:} Condición que detiene la recursión
    \item \textbf{Caso recursivo:} Llamada a sí misma con un problema más simple
  \end{enumerate}
  
  \vspace{0.3cm}
  \begin{alertblock}{Importante}
    Sin un caso base, la recursión sería infinita y causaría un error de desbordamiento de pila.
  \end{alertblock}
\end{frame}

% ------------------------------------------------------------------------
% Slide 6: Ejemplo 1 - Factorial
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Ejemplo 1: Factorial Recursivo}
  \begin{block}{Definición matemática del factorial}
    \[
    n! = \begin{cases}
      1 & \text{si } n = 0 \\
      n \times (n-1)! & \text{si } n > 0
    \end{cases}
    \]
  \end{block}
  
\begin{minted}{python}
def factorial(n):
    # Caso base: factorial de 0 es 1
    if n == 0:
        return 1
    # Caso recursivo: n! = n × (n-1)!
    else:
        return n * factorial(n - 1)

# Ejemplo
print(factorial(5))  # 5! = 5×4×3×2×1 = 120
\end{minted}
\end{frame}

% Slide 7: Visualización del factorial

\begin{frame}[fragile]{Ejemplo 2: Suma Recursiva de una Lista}
\begin{minted}{python}
def suma_recursiva(lista):
    # Caso base: lista vacía suma 0
    if len(lista) == 0:
        return 0
    # Caso recursivo: primer elemento + suma del resto
    else:
        return lista[0] + suma_recursiva(lista[1:])

# Ejemplo con distancias planetarias (UA)
distancias = [1.0, 1.52, 5.20]  # Tierra, Marte, Júpiter
print(suma_recursiva(distancias))  # 7.72 UA
\end{minted}

\vspace{0.3cm}
\textbf{Conceptos clave:}
\begin{itemize}
  \item \texttt{lista[0]}: primer elemento
  \item \texttt{lista[1:]}: resto de la lista (slicing)
  \item Cada llamada trabaja con una lista más pequeña
\end{itemize}
\end{frame}

\section{Ordenamiento: Bubble Sort}

% ------------------------------------------------------------------------
% Slide 10: Introducción al ordenamiento
% ------------------------------------------------------------------------
\begin{frame}{¿Qué es el Ordenamiento?}
  \begin{block}{Definición}
    Reorganizar una colección de elementos en un orden específico 
    (ascendente o descendente).
  \end{block}
  
  \vspace{0.3cm}
  \textbf{¿Por qué es importante?}
  \begin{itemize}
    \item Facilita la búsqueda de información
    \item Permite identificar extremos (máximos/mínimos)
    \item Base para algoritmos más complejos
    \item Mejora la visualización de datos
  \end{itemize}
  
  \vspace{0.3cm}
  \textbf{Hoy estudiaremos:} \textbf{Bubble Sort} (ordenamiento por burbujeo)
\end{frame}

% ------------------------------------------------------------------------
% Slide 11: Concepto de Bubble Sort
% ------------------------------------------------------------------------
\begin{frame}{Bubble Sort: Idea Principal}
  \begin{block}{¿Cómo funciona?}
    Compara \textbf{pares de elementos adyacentes} y los intercambia si están 
    en el orden incorrecto. Repite este proceso hasta que la lista esté ordenada.
  \end{block}
  
  \vspace{0.3cm}
  \textbf{Analogía visual:}
  \begin{itemize}
    \item Imagina burbujas de aire en agua
    \item Las más ligeras "suben" naturalmente
    \item Los elementos más grandes "burbujean" hacia el final
    \item Video demostrativo: \url{https://www.youtube.com/watch?v=kPRA0W1kECg&t=242s}  
  \end{itemize}
  
  \vspace{0.3cm}
  \begin{alertblock}{Característica clave}
    Después de cada pasada completa, el elemento más grande está en su posición final.
  \end{alertblock}
\end{frame}

% Slide 12: Ejemplo paso a paso CON CAJAS
% ------------------------------------------------------------------------

% ------------------------------------------------------------------------
% Slide 13: Implementación de Bubble Sort
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Implementación de Bubble Sort}
\begin{minted}{python}
def bubble_sort(lista):
    n = len(lista)
    # Realizar n-1 pasadas
    for i in range(n - 1):
        # En cada pasada, comparar elementos adyacentes
        for j in range(n - 1 - i):
            # Si están en orden incorrecto, intercambiar
            if lista[j] > lista[j + 1]:
                lista[j], lista[j + 1] = lista[j + 1], lista[j]
    return lista
\end{minted}

\vspace{0.3cm}
\textbf{Análisis del código:}
\begin{itemize}
  \item \texttt{range(n-1)}: necesitamos n-1 pasadas
  \item \texttt{range(n-1-i)}: cada pasada hace menos comparaciones
  \item \texttt{lista[j], lista[j+1] = ...}: intercambio elegante en Python
\end{itemize}
\end{frame}

% ------------------------------------------------------------------------
% Slide 14: Complejidad de Bubble Sort
% ------------------------------------------------------------------------
\begin{frame}{Análisis de Eficiencia: Bubble Sort}
  \begin{block}{Complejidad Temporal}
    \[
    \mathcal{O}(n^2) \text{ (complejidad cuadrática)}
    \]
  \end{block}
  
	  \vspace{0.3cm}
	  \textbf{¿Qué significa esto?}
	  \[
	    \text{Comparaciones} =
	    (n-1) + (n-2) + \cdots + 1 =
	    \frac{n(n-1)}{2}
	  \]
	  \begin{itemize}
	    \item Para \textbf{10 elementos}: 45 comparaciones
	    \item Para \textbf{100 elementos}: 4,950 comparaciones
	    \item Para \textbf{1,000 elementos}: 499,500 comparaciones
	  \end{itemize}
  
  \vspace{0.3cm}
  \begin{alertblock}{Ventajas y desventajas}
    \textbf{Ventaja:} Fácil de entender e implementar\\
    \textbf{Desventaja:} Ineficiente para listas grandes
  \end{alertblock}
\end{frame}

\section{Algoritmos de Búsqueda}

% ------------------------------------------------------------------------
% Slide 16: Introducción a la búsqueda
% ------------------------------------------------------------------------
\begin{frame}{Búsqueda: ¿Qué es y por qué importa?}
  \begin{block}{Problema}
    Encontrar si un elemento específico existe en una colección de datos 
    y determinar su posición.
  \end{block}
  
  \vspace{0.3cm}
  \textbf{Dos estrategias fundamentales:}
  \begin{enumerate}
    \item \textbf{Búsqueda Lineal:} revisar elemento por elemento
    \item \textbf{Búsqueda Binaria:} aprovechar que la lista está ordenada
  \end{enumerate}
  
  \vspace{0.3cm}
  \textbf{Aplicaciones en astronomía:}
  \begin{itemize}
    \item Buscar estrella por magnitud en catálogos
    \item Encontrar exoplaneta con características específicas
    \item Localizar mediciones en grandes conjuntos de datos
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------
% Slide 17: Búsqueda Lineal
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Búsqueda Lineal}
  \begin{block}{Estrategia}
    Revisar cada elemento de la lista secuencialmente hasta encontrar el objetivo 
    o llegar al final.
  \end{block}

\begin{minted}{python}
def linear_search(lista, objetivo):
    for i, x in enumerate(lista):
        if x == objetivo:
            return i  # Retorna la posición
    return -1  # No encontrado
\end{minted}

\vspace{0.3cm}
\textbf{Características:}
\begin{itemize}
  \item \textbf{Complejidad:} $\mathcal{O}(n)$ - lineal
  \item \textbf{Ventaja:} funciona en listas no ordenadas
  \item \textbf{Desventaja:} lento para listas grandes
\end{itemize}
\end{frame}

% ------------------------------------------------------------------------
% Slide 18: Búsqueda Binaria - Concepto
% ------------------------------------------------------------------------
\begin{frame}{Búsqueda Binaria: Idea Principal}
  \begin{block}{Requisito previo}
    La lista \textbf{DEBE} estar ordenada.
  \end{block}
  
  \vspace{0.3cm}
  \textbf{Estrategia "Divide y Conquista":}
  \begin{enumerate}
    \item Comparar el objetivo con el elemento del \textbf{centro}
    \item Si es igual → encontrado
    \item Si el objetivo es menor → buscar en la mitad izquierda
    \item Si el objetivo es mayor → buscar en la mitad derecha
    \item Repetir hasta encontrar o agotar opciones
  \end{enumerate}
  
  \vspace{0.3cm}
  \begin{alertblock}{Ventaja clave}
    En cada paso, \textbf{elimina la mitad} de las opciones restantes.
  \end{alertblock}
\end{frame}

% ------------------------------------------------------------------------
% Slide 19: Ejemplo visual de búsqueda binaria CON CAJAS
% ------------------------------------------------------------------------
\begin{frame}{Búsqueda Binaria: Ejemplo Visual}
  \small
  \textbf{Buscar el} \fcolorbox{red}{red!20}{\textbf{7}} \textbf{en:}

  \end{itemize}

  \begin{center}\vspace{-0.15cm}

  \vspace{0.1cm}
  \textbf{Comparación:}
  \begin{itemize}\setlength{\itemsep}{0pt}
    \item Búsqueda lineal: hasta 7 comparaciones
    \item Búsqueda binaria: máximo 3 comparaciones
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------
% Slide 20: Implementación búsqueda binaria
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Implementación de Búsqueda Binaria}
\begin{minted}{python}
def binary_search(lista_ordenada, objetivo):
    low, high = 0, len(lista_ordenada) - 1
    
    while low <= high:
        mid = (low + high) // 2  # Índice del centro
        
        if lista_ordenada[mid] == objetivo:
            return mid  # Encontrado
        elif objetivo < lista_ordenada[mid]:
            high = mid - 1  # Buscar en mitad izquierda
        else:
            low = mid + 1   # Buscar en mitad derecha
    
    return -1  # No encontrado
\end{minted}

\textbf{Complejidad:} $\mathcal{O}(\log n)$ - logarítmica (muy eficiente)
\end{frame}

  \end{alertblock}
\end{frame}

def bubble_sort(lst):
    n = len(lst)
    for i in range(n-1):
        for j in range(n-1-i):
            if lst[j] > lst[j+1]:
                lst[j], lst[j+1] = \
                    lst[j+1], lst[j]
    return lst

\begin{solutionblock}
\begin{frame}[fragile]{Solución 2: \hfill \textcolor{green}{$\checkmark$} \\ Implementación con Contador}
\begin{minted}{python}
def binary_search(lst, target):
    low, high, checks = 0, len(lst) - 1, 0
    
    while low <= high:
        mid = (low + high) // 2
        checks += 1  # Contar cada comparación
        
        if lst[mid] == target:
            return mid, checks  # Encontrado
        elif target < lst[mid]:
            high = mid - 1
        else:
            low = mid + 1
    
    return -1, checks  # No encontrado
\end{minted}

\textbf{Discusión:} El contador \texttt{checks} nos permite verificar la eficiencia teórica $\mathcal{O}(\log n)$.
\end{frame}
\end{solutionblock}

    for checks, x in enumerate(lst, 1):
        if x == target:
            return checks
    return checks  # No encontrado

\begin{itemize}

% ------------------------------------------------------------------------
% Slide 36: Errores comunes
% ------------------------------------------------------------------------
\begin{frame}{Errores Comunes}
  \begin{block}{En Bubble Sort}
    \begin{itemize}
      \item Olvidar \texttt{.copy()} al ordenar (modifica lista original)
      \item Índices incorrectos: \texttt{range(n)} en lugar de \texttt{range(n-1)}
      \item No reducir comparaciones: \texttt{range(n-1-i)} es crucial para eficiencia
    \end{itemize}
  \end{block}
  
  \begin{block}{En Búsqueda Binaria}
    \begin{itemize}
      \item Olvidar ordenar la lista previamente
      \item Usar \texttt{/} en lugar de \texttt{//} para \texttt{mid} (debe ser entero)
      \item Actualizar incorrectamente \texttt{low} o \texttt{high}
      \item Bucle infinito si no se actualizan límites
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Consejo de depuración}
    Usar \texttt{print()} dentro de los bucles para visualizar el proceso paso a paso.
  \end{alertblock}
\end{frame}
