\section{Recursividad \& Merge Sort}
% -----------------------------------------------------------------
\begin{frame}[fragile]{¿Qué es recursividad?}
\begin{itemize}
  \item Una función que se llama a sí misma para resolver un problema más pequeño.
  \item Necesita: \textit{caso base} + \textit{llamada recursiva}.
  \item Ejemplo clásico: factorial, Fibonacci… ¡y algoritmos de ordenamiento!
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Merge Sort en Python (\(\mathcal{O}(n\log n)\))}
\begin{minted}[fontsize=\scriptsize]{python}
def merge_sort(lst):
    if len(lst) <= 1:
        return lst                    # caso base
    mid = len(lst) // 2
    left  = merge_sort(lst[:mid])     # divide
    right = merge_sort(lst[mid:])
    return merge(left, right)         # conquista

def merge(left, right):
    out, i, j = [], 0, 0                    # apuntadores
    while i < len(left) and j < len(right): # mezcla ordenada 
	        if left[i] <= right[j]:             # comparar 
            out.append(left[i])             # agregar menor
            i += 1                          # avanzar en left
        else:
            out.append(right[j])            # agregar menor
            j += 1                          # avanzar en right
    return out + left[i:] + right[j:]       # agregar restos
\end{minted}
\begin{itemize}
  \item Dos fases: división recursiva y mezcla ordenada.
	  \item Estable (\alert{preserva el orden relativo de iguales}) y predecible.
\end{itemize}
\end{frame}

\section{Merge Sort: Ordenamiento Eficiente}

\begin{frame}{Merge Sort: ¿Por qué es mejor que Bubble Sort?}
  \begin{block}{Problema con Bubble Sort}
	    Con 10,000 elementos → 49,995,000 comparaciones \\
    Tiempo: varios segundos o minutos
  \end{block}
  
  \begin{block}{Ventaja de Merge Sort}
	    Con 10,000 elementos → $\sim$133,000 comparaciones estimadas \\
    Tiempo: milisegundos
  \end{block}
  
  \textbf{Complejidad:} $\mathcal{O}(n \log n)$ vs $\mathcal{O}(n^2)$
  
  \begin{alertblock}{Diferencia práctica}
    Para 100,000 elementos: Bubble Sort tardaría \textbf{horas}, 
    Merge Sort tardaría \textbf{segundos}.
  \end{alertblock}
\end{frame}

\begin{frame}{Merge Sort: Estrategia "Divide y Conquista"}
  \begin{block}{Idea Principal}
    \begin{enumerate}
      \item \textbf{Divide:} Partir la lista en dos mitades
      \item \textbf{Conquista:} Ordenar recursivamente cada mitad
      \item \textbf{Combina:} Mezclar las dos mitades ordenadas
    \end{enumerate}
  \end{block}

\begin{frame}[fragile]{Merge Sort: Función de Mezcla (merge)}
  \textbf{Paso 1: Entender cómo mezclar dos listas ordenadas}
  
\begin{minted}{python}
def merge(left, right):
    """
    Mezcla dos listas ordenadas en una sola lista ordenada.
    Ejemplo: merge([2, 5], [1, 8]) → [1, 2, 5, 8]
    """
    resultado = []
    i, j = 0, 0
    
    # Comparar elementos de ambas listas
    while i < len(left) and j < len(right):
	        if left[i] <= right[j]:
            resultado.append(left[i])
            i += 1
        else:
            resultado.append(right[j])
            j += 1
    
    # Agregar elementos restantes
    resultado.extend(left[i:])
    resultado.extend(right[j:])
    return resultado
\end{minted}
\end{frame}

\begin{frame}[fragile]{Merge Sort: Función Principal (Recursiva)}
\begin{minted}{python}
def merge_sort(lista):
    """
    Ordena una lista usando el algoritmo Merge Sort.
    Complejidad: O(n log n)
    """
    # Caso base: lista de 0 o 1 elemento ya está ordenada
    if len(lista) <= 1:
        return lista
    
    # Dividir la lista en dos mitades
    mid = len(lista) // 2
    left = lista[:mid]
    right = lista[mid:]
    
    # Ordenar recursivamente cada mitad
    left_ordenada = merge_sort(left)
    right_ordenada = merge_sort(right)
    
    # Combinar las mitades ordenadas
    return merge(left_ordenada, right_ordenada)
\end{minted}

\textbf{Elementos clave:} Caso base, división, recursión, mezcla.
\end{frame}

\section{Medición de Rendimiento}

\begin{frame}[fragile]{Introducción al Módulo \texttt{time}}
  \textbf{¿Cómo medir cuánto tarda un algoritmo?}
  
\begin{minted}{python}
import time

# Iniciar cronómetro
inicio = time.time()

# Ejecutar código a medir
resultado = bubble_sort(lista.copy())

# Detener cronómetro
fin = time.time()

# Calcular tiempo transcurrido
tiempo_total = fin - inicio
print(f"Tiempo: {tiempo_total:.4f} segundos")
\end{minted}

\vspace{0.3cm}
\textbf{Conceptos clave:}
\begin{itemize}
  \item \texttt{time.time()}: retorna tiempo actual en segundos
  \item Restar inicio de fin da el tiempo transcurrido
  \item Usar \texttt{.copy()} para no modificar datos originales
\end{itemize}
\end{frame}

\begin{frame}{Errores Comunes con Merge Sort}
  \begin{block}{Errores frecuentes}
    \begin{itemize}
      \item \textbf{Caso base incorrecto:} olvidar \texttt{if len(lista) <= 1}
      \item \textbf{División incorrecta:} usar \texttt{/} en lugar de \texttt{//}
      \item \textbf{Índices en merge:} confundir \texttt{i} y \texttt{j}
      \item \textbf{Olvidar elementos restantes:} no usar \texttt{extend()}
    \end{itemize}
  \end{block}
  
  \begin{block}{En medición de tiempos}
    \begin{itemize}
      \item \textbf{No usar \texttt{.copy()}:} modifica datos originales
      \item \textbf{Medir solo una vez:} resultados inconsistentes
      \item \textbf{No calentar caché:} primera ejecución más lenta
    \end{itemize}
