\section{Ajuste lineal con \texttt{np.polyfit}}

\begin{frame}{Fenómeno físico: relación lineal simple}
  \begin{itemize}
    \item \textbf{Escenarios típicos:} calibración sensor \(V=aT+b\), ley de Hooke en pequeñas deformaciones \(F=kx\), ley de Ohm \(V=RI\).
    \item \textbf{¿Por qué una recta?} por \textit{aproximación lineal} (expansión de Taylor de primer orden) en un rango acotado.
    \item \textbf{Qué medimos:} pares \((x\_i, y\_i)\) con incertidumbre en \(y\) y (a veces) en \(x\).
    \item \textbf{Meta:} estimar parámetros físicos (pendiente e intercepto) y evaluar calidad del modelo.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Recta por mínimos cuadrados con \texttt{np.polyfit}}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Supongamos y = m x + b + ruido
m_est, b_est = np.polyfit(x, y_obs, deg=1) # deg (int) grado del polinomio a ajustar
y_fit = m_est * x + b_est

res = y_obs - y_fit
SS_res = np.sum(res**2)
SS_tot = np.sum((y_obs - np.mean(y_obs))**2)
R2 = 1 - SS_res/SS_tot

plt.scatter(x, y_obs, label='datos')
plt.plot(x, y_fit, 'r', label=f'fit: y={m_est:.2f}x+{b_est:.2f}')
plt.legend(); plt.grid(alpha=0.3); plt.show()
print('R^2 =', R2)
\end{minted}
	\textbf{Notas:} \texttt{np.polyfit} entrega coeficientes polinomiales por mínimos cuadrados.
\end{frame}

\begin{frame}{¿Qué hace este código? (paso a paso)}
  \begin{itemize}
    \item Genera datos \(x\) y una relación lineal \(y\_\text{true}=mx+b\) con ruido gaussiano (\(\sigma\)).
    \item Aplica \texttt{np.polyfit} para obtener \(m,b\) que minimizan \(\sum r\_i^2\).
    \item Calcula \(R^2\) para medir la fracción de varianza explicada.
    \item Grafica datos y la recta ajustada para inspección visual.
  \end{itemize}
  	\textbf{Por qué así:} es la forma estándar y eficiente de mínimos cuadrados en NumPy.
\end{frame}

\begin{frame}{¿Qué hace realmente \texttt{polyfit}? y supuestos}
  \begin{itemize}
    \item \textbf{Modelo:} asumimos una relación polinómica (aquí, recta) \(y\_i \approx m x\_i + b\).
    \item \textbf{Objetivo:} encontrar \(m,b\) que \textit{minimicen la suma de cuadrados} de los residuos: \(\sum (y\_i - y\_\text{mod})^2\).
    \item \textbf{Supuestos implícitos:}
      \begin{itemize}
        \item Errores en \(y\) son gaussianos, independientes y con varianza constante.
        \item El modelo lineal es adecuado en el rango de datos.
      \end{itemize}
    \item \textbf{Física:} estos supuestos se aproximan cuando la incertidumbre experimental es similar para todos los puntos y el fenómeno es lineal en el rango.
    \item \textbf{Resultado:} \texttt{polyfit} devuelve los coeficientes del polinomio que minimiza el error cuadrático medio.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ponderación por incertidumbres: \(w = 1/\sigma\)}
\begin{minted}{python}
# Si cada punto tiene una incertidumbre distinta sigma_i en y,
# podemos ponderar el ajuste: puntos con menor sigma pesan más.
rng = np.random.default_rng(0)
sigma_i = rng.uniform(1.0, 3.0, size=x.size)
w = 1.0 / sigma_i

m_w, b_w = np.polyfit(x, y_obs, deg=1, w=w)
print(f"sin peso:  m={m_est:.2f}, b={b_est:.2f}")
print(f"con peso:  m={m_w:.2f}, b={b_w:.2f}")
\end{minted}
	\textbf{Interpretación:} en física, mediciones más precisas deben influir más en el ajuste.
\end{frame}

\section{Ejemplo: Ley de Hubble}

\begin{frame}{Ley de Hubble: contexto físico y observacional}
  \begin{itemize}
    \item \textbf{Idea simple:} cuanto más \textbf{lejos} está una galaxia (\(d\)), más \textbf{rápido} se aleja (\(v\)). La relación es \(v = H_0\, d\).
    \item \textbf{Interpretación geométrica:} el \textit{pendiente} de la recta es \(H_0\). Si la expansión fuera perfecta, el \textit{intercepto} sería 0.
    \item \textbf{Régimen de validez local:} para \(z \ll 1\), \(v \approx c\,z\). Para \(z\) grandes se usa un modelo cosmológico completo.
    \item \textbf{Cómo se mide:}
      \begin{itemize}
        \item \(v\): a partir del \textbf{corrimiento al rojo} \(z\) medido en espectros.
        \item \(d\): con la \textbf{escalera de distancias} (cefeidas, supernovas Ia, etc.).
      \end{itemize}
    \item \textbf{Dispersión real:} las \textbf{velocidades peculiares} (\(\sim100{-}300\) km/s) añaden ruido, especialmente a distancias pequeñas.
  \end{itemize}

\end{frame}

\begin{frame}
  \begin{block}{Diccionario rápido (símbolos y unidades)}
    \begin{itemize}
      \item \(v\): velocidad de recesión, en km/s.
      \item \(d\): distancia, en Mpc (1 Mpc \(\approx\) 3.26 millones de años luz).
      \item \(H_0\): constante de Hubble, en km\,s\(^{-1}\)\,Mpc\(^{-1}\) (\(\sim 70\) en este ejemplo).
      \item \(z\): corrimiento al rojo; si \(z\ll 1\), \(v \approx c\,z\) (\(c\): velocidad de la luz).
      \item Intercepto: valor de \(v\) cuando \(d=0\); idealmente \(\approx 0\), pero puede verse afectado por sesgos/peculiares.
    \end{itemize}
  \end{block}
  \begin{alertblock}{Ejemplo numérico}
    Si \(H_0=70\) km\,s\(^{-1}\)\,Mpc\(^{-1}\) y \(d=100\) Mpc, entonces \(v\approx 7000\) km/s.
  \end{alertblock}
\end{frame}

\begin{frame}{Supuestos, validez y fuentes de dispersión}
  \begin{itemize}
    \item \textbf{Supuestos:} el universo es \textit{más o menos igual} en todas partes (homogéneo e isótropo) y se expande de forma uniforme \textit{en promedio}.
    \item \textbf{Unidades y tiempos:} \(H_0\) también puede escribirse en s\(^{-1}\). El \textit{tiempo de Hubble} \(t_H \approx 1/H_0\) da una escala de edad del universo.
    \item En el ejemplo siguiente usaremos datos \textbf{sintéticos} para practicar el ajuste y recuperar \(H_0\).
  \end{itemize}
  \begin{block}{Traducción simple}
    "Más lejos \(\Rightarrow\) más rápido". El \(H_0\) es cuántos km/s aumentan por cada Mpc de distancia.
  \end{block}
  \begin{alertblock}{Cuidado}
    A distancias pequeñas los términos peculiares pueden dominar y hacer que el intercepto aparente no sea 0.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Ley de Hubble: \(v = H_0\, d\)}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt
# Simular galaxias: velocidad vs distancia
rng = np.random.default_rng(7)
H0_true = 70.0  # km/s/Mpc
d = np.linspace(0, 200, 30)  # Mpc
v_true = H0_true * d
sigma_v = 150.0  # km/s (dispersión peculiar)
v_obs = v_true + rng.normal(0, sigma_v, size=d.size)

# Graficar datos observados
plt.scatter(d, v_obs, s=20)
plt.xlabel('Distancia (Mpc)'); plt.ylabel('Velocidad (km/s)')

# Ajustar H0 a los datos simulados
H0_hat, v0_hat = np.polyfit(d, v_obs, 1)
v_fit = H0_hat * d + v0_hat
plt.plot(d, v_fit, 'r', label=f'fit: H0={H0_hat:.1f} km/s/Mpc')

print(f"H0 estimado = {H0_hat:.1f} km/s/Mpc, intercepto = {v0_hat:.1f} km/s")
\end{minted}
	\textbf{Física:} a primer orden, la expansión se ve como una relación lineal entre \(v\) y \(d\).
\end{frame}

\begin{frame}{¿Qué hace este código de Hubble?}
  \begin{itemize}
    \item Simula distancias \(d\) y velocidades \(v\) con \(H\_0\) verdadero, agregando \textbf{dispersión peculiar} (ruido) \(\sigma\_v\).
    \item Ajusta \(v=H\_0\,d + b\) para estimar \(H\_0\) e intercepto.
    \item Muestra la recta resultante y reporta los parámetros.
    \item Posteriormente calcula residuos y \(\chi^2\_\text{red}\) para evaluar consistencia con las \(\sigma\_v\).
  \end{itemize}
  	\textbf{Conexión física:} el pendiente es una medida de la expansión local del Universo.
\end{frame}

\begin{frame}[fragile]{Ley de Hubble: residuos y \(\chi^2\) reducido}
\begin{minted}{python}
# Residuos y chi^2 reducido del ajuste lineal v = H0 d + b
res = v_obs - v_fit
chi2 = np.sum((res/sigma_v)**2)
nu = d.size - 2  # parámetros: H0 y b
chi2_red = chi2/nu
print(f"chi2_red = {chi2_red:.2f}")

# (Opcional) Inspección de residuos por distancia
# for di, ri in zip(d, res):
#     print(f"d={di:6.1f} Mpc, residuo={ri:7.1f} km/s")
\end{minted}
	\textbf{Lectura física:} residuos reflejan velocidades peculiares y errores en distancias;
\(\chi^2_\mathrm{red}\approx 1\) sugiere incertidumbres realistas.
\end{frame}

\begin{frame}{Residuos en contexto físico}
  \begin{itemize}
    \item \textbf{Residuos} \(r\_i = v\_{\text{obs},i} - v\_{\text{fit},i}\): nos dicen cuánto se desvía cada galaxia de la recta de Hubble.
    \item \textbf{Causas típicas:} velocidades peculiares (caídas al pozo de un cúmulo, interacciones), errores en distancias, selección de muestra.
    \item \textbf{Patrones:} si los residuos muestran tendencia con \(d\), el modelo lineal podría ser insuficiente o las incertidumbres están mal estimadas.
    \item \textbf{\(\chi^2\) reducido cercano a 1:} sugiere que las \(\sigma\) usadas reflejan bien la dispersión real.
  \end{itemize}
\end{frame}

\section{Ajuste lineal con \texttt{curve\_fit}}

\begin{frame}[fragile]{\texttt{scipy.optimize.curve\_fit} en un modelo lineal}
\begin{minted}{python}
from scipy.optimize import curve_fit

def modelo_lineal(x, m, b):
    return m*x + b

popt, pcov = curve_fit(modelo_lineal, d, v_obs, sigma=sigma_v,
                        absolute_sigma=True)
m_cf, b_cf = popt

# Errores estándar de los parámetros
# np.diag(pcov) extrae la varianza de cada parámetro
perr = np.sqrt(np.diag(pcov))

# Graficar resultados del ajuste
plt.scatter(d, v_obs, s=20, label='galaxias')
plt.plot(d, modelo_lineal(d, *popt), 'r', label=f'fit: m={m_cf:.2f}, b={b_cf:.1f}')

print(f"m={m_cf:.2f}±{perr[0]:.2f}, b={b_cf:.1f}±{perr[1]:.1f}")
\end{minted}
	\textbf{Ventaja:} entrega \textit{covarianzas} de parámetros para estimar errores.
\end{frame}

\section{Relación masa–luminosidad}

\begin{frame}[fragile]{Potencia: \(L = k\, M^{\alpha}\). Enfoque log–log}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
# Simular estrellas
rng = np.random.default_rng(1)
alpha_true = 3.5
k_true = 1.0
M = np.linspace(0.5, 10, 40)
L_true = k_true * M**alpha_true
sigma_rel = 0.10
L_obs = L_true * (1 + rng.normal(0, sigma_rel, size=M.size))

# Transformación log10
X = np.log10(M)
Y = np.log10(L_obs)
a, b = np.polyfit(X, Y, 1)  # Y = a X + b
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
alpha_hat = a
k_hat = 10**b
print(f"alpha≈{alpha_hat:.2f}, k≈{k_hat:.2f}")

import matplotlib.pyplot as plt
plt.scatter(X, Y, s=20)
plt.plot(X, a*X + b, 'r')
plt.xlabel('log M'); plt.ylabel('log L')
plt.title('Masa–Luminosidad (log–log)')
plt.grid(alpha=0.3); plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Discusión:} la linealización facilita el ajuste de leyes de potencia.
\end{frame}

\begin{frame}{¿Qué hace este código (log–log)?}
  \begin{itemize}
    \item Simula \(M\) y \(L\) con una ley de potencia y ruido multiplicativo (más realista en astrofísica).
    \item Transforma a \(\log\) para linealizar: \(\log L = \alpha\,\log M + \log k\).
    \item Ajusta con \texttt{polyfit} en el espacio logarítmico y recupera \(\alpha\) y \(k\).
    \item Discute diferencias con el ajuste no lineal en el espacio original.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Potencia directa con \texttt{curve\_fit} (no lineal)}
\begin{minted}{python}
from scipy.optimize import curve_fit

def power_law(M, k, alpha):
    return k * M**alpha

popt, pcov = curve_fit(power_law, M, L_obs, p0=(1.0, 3.0))
k_cf, alpha_cf = popt
perr = np.sqrt(np.diag(pcov))
print(f"k={k_cf:.2f}±{perr[0]:.2f}, alpha={alpha_cf:.2f}±{perr[1]:.2f}")
# Graficar resultados
plt.scatter(M, L_obs, s=20, label='estrellas')
M_fit = np.linspace(0.5, 10, 100)
plt.plot(M_fit, power_law(M_fit, *popt), 'r', label='fit potencia')

\end{minted}
	\textbf{Comparación:} log–log vs ajuste no lineal; considerar sesgos con ruido multiplicativo.
\end{frame}

\begin{frame}{Física detrás de \(L \propto M^{\alpha}\)}
  \begin{itemize}
    \item \(\alpha\) captura cómo cambia la \textbf{producción de energía} con la masa (tasa de fusión y estructura interna).
    \item Para estrellas de la secuencia principal, típicamente \(\alpha \approx 3{-}4\) en ciertos rangos de masa.
    \item \textbf{Escalas y unidades:} cuidado con trabajar en unidades consistentes; la forma funcional es adimensional si se normaliza.
    \item \textbf{Log–log vs no lineal:} la transformación logarítmica linealiza pero cambia la distribución de errores; el ajuste no lineal trabaja en el espacio original.
  \end{itemize}
\end{frame}

\section{Interpolación}

\begin{frame}[fragile]{Interpolación 1D con \texttt{interp1d}}
\begin{minted}{python}
import numpy as np
from scipy.interpolate import interp1d
import matplotlib.pyplot as plt

x = np.linspace(0, 10, 12)
y = np.sin(x) + 0.1*np.cos(3*x)
f_lin = interp1d(x, y, kind='linear')
f_cub = interp1d(x, y, kind='cubic')

xx = np.linspace(0, 10, 200)
plt.plot(x, y, 'o', label='muestras')
plt.plot(xx, f_lin(xx), '-', label='lineal')
plt.plot(xx, f_cub(xx), '--', label='cúbica')
plt.legend(); plt.grid(alpha=0.3); plt.show()
\end{minted}
	\textbf{Uso típico:} rellenar curvas (p. ej., espectros o curvas de luz) entre puntos.
\end{frame}

\begin{frame}{¿Qué hace este código de interpolación?}
  \begin{itemize}
    \item Construye una señal suave \(y(x)\) muestreada en pocos puntos.
    \item Define interpoladores lineal y cúbico, y evalúa en una malla densa.
    \item Compara ambas curvas para decidir según la física del fenómeno (suavidad esperable).
  \end{itemize}
\end{frame}

% Casos de sesgo/errores típicos en interpolación
\begin{frame}[fragile]{Interpolación: polinomio de alto grado (mal) vs cúbica (bien)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Función tipo Runge con ruido
rng = np.random.default_rng(0)
x = np.linspace(-1, 1, 11)
y = 1/(1 + 25*x**2) + rng.normal(0, 0.02, x.size)

# MAL: polinomio de alto grado
deg = 10
p = np.polyfit(x, y, deg)
xx = np.linspace(-1, 1, 400)
yy_poly = np.polyval(p, xx)

plt.plot(x, y, 'o', label='muestras')
plt.plot(xx, yy_poly, 'r-', label=f'pol grado {deg}')
plt.title('Oscilaciones (Runge)')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=1]{python}
# BIEN: interpolación cúbica por tramos
from scipy.interpolate import interp1d
f_cub = interp1d(x, y, kind='cubic')  # corrección clave

plt.plot(x, y, 'o', label='muestras')
plt.plot(xx, f_cub(xx), 'g-', label='interp cúbica')
plt.title('Cúbica estable (sin oscilaciones)')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Idea:} polinomios de grado alto suelen oscilar; preferir esquemas por tramos (cúbica, splines).
\end{frame}

\begin{frame}{¿Qué error vemos? Fenómeno de Runge}
  \begin{itemize}
    \item \textbf{Síntoma:} la curva ajustada con un polinomio global de alto grado \textbf{oscila} mucho, sobre todo en los extremos, pese a pasar cerca de los puntos.
    \item \textbf{Causa:} polinomios globales son rígidos y sensibles a nodos equiespaciados; pequeñas variaciones en datos generan grandes ondulaciones.
    \item \textbf{Consecuencia:} la forma física se distorsiona; el error entre nodos puede ser grande aunque el error en nodos sea pequeño.
    \item \textbf{Buenas prácticas:}
      \begin{itemize}
        \item Usar \textbf{interpolación por tramos} (cúbica, splines B) o \textbf{PCHIP} si se requiere preservar forma.
        \item Si se exige polinomio global, preferir \textbf{nodos de Chebyshev} en lugar de equiespaciados.
        \item Validar visualmente y con métricas locales, no solo con \(R^2\).
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Extrapolación peligrosa vs límites seguros}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt
from scipy.interpolate import interp1d

x = np.linspace(0, 10, 8)
y = np.exp(-0.3*x)
f_lin = interp1d(x, y, kind='linear',
                 fill_value='extrapolate')
xx = np.linspace(-2, 12, 300)

plt.plot(x, y, 'o', label='muestras')
plt.plot(xx, f_lin(xx), 'r-', label='extrapola')
plt.axvspan(-2, 0, color='red', alpha=0.05)
plt.axvspan(10, 12, color='red', alpha=0.05)
plt.title('Fuera de rango: engañoso')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=1]{python}
# Corrección: no extrapolar
f_safe = interp1d(x, y, kind='linear',
                  bounds_error=False,
                  fill_value=np.nan)  # clave

plt.plot(x, y, 'o')
plt.plot(xx, f_safe(xx), 'g-', label='sin extrapolar')
plt.title('Restringir al rango de datos')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Regla:} evitar extrapolar a menos que el modelo físico lo justifique.
\end{frame}

\begin{frame}{¿Por qué extrapolar es peligroso?}
  \begin{itemize}
    \item \textbf{Fuera del rango no hay datos:} la pendiente y curvatura son \textit{suposiciones}, no información.
    \item \textbf{Amplificación de error:} pequeños errores en el ajuste dentro del rango pueden crecer mucho fuera de él.
    \item \textbf{Modelos físicos primero:} solo extrapole si un \textbf{modelo físico} bien justificado describe el comportamiento.
    \item \textbf{Prácticas seguras:}
      \begin{itemize}
        \item Use \texttt{bounds\_error=False, fill\_value=np.nan} para no “inventar” valores.
        \item Señalice visualmente regiones extrapoladas si las muestra.
        \item Si debe extrapolar, \textbf{propague incertidumbre} y reporte intervalos amplios.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Conservar forma física: cúbica vs PCHIP (monótono)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt
from scipy.interpolate import interp1d

y = 1 - np.exp(-x) + 0.02*np.sin(6*x)
f_cub = interp1d(x, y, kind='cubic')
xx = np.linspace(0, 6, 400)

plt.plot(x, y, 'o')
plt.plot(xx, f_cub(xx), 'r-', label='cúbica')
plt.title('Overshoot (viola monotonicidad)')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=1]{python}

plt.plot(x, y, 'o')
plt.plot(xx, f_mon(xx), 'g-', label='PCHIP')
plt.title('PCHIP: sin overshoot')
plt.legend(); plt.grid(True, alpha=0.3); plt.show()

\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Uso:} cuando el fenómeno debe ser monótono o libre de oscilaciones no físicas.
\end{frame}

\begin{frame}{¿Qué es overshoot? ¿Por qué PCHIP ayuda?}
  \begin{itemize}
    \item \textbf{Overshoot/undershoot:} la interpolación \textbf{sobrepasa} (o cae por debajo de) el rango definido por puntos vecinos, creando crestas/valles \textit{no físicos}.
    \item \textbf{Cúbica estándar:} prioriza suavidad de derivadas pero puede introducir oscilaciones y overshoot cerca de cambios de curvatura.
    \item \textbf{PCHIP:} interpolador por tramos que \textbf{preserva la forma y la monotonicidad}; evita overshoot manteniendo la curva dentro del rango local.
    \item \textbf{Cuándo usar:} series que deben ser \textbf{monótonas} o sin oscilaciones artificiales (curvas de respuesta, acumulados, luminosidad vs tiempo sin rebotes).
    \item \textbf{Trade-off:} PCHIP puede ser menos suave en derivadas que un spline cúbico, pero más fiel físicamente.
  \end{itemize}
\end{frame}

\section{Ajuste de distribuciones}

\begin{frame}{¿Por qué ajustar distribuciones? (física/astro)}
  \begin{itemize}
    \item \textbf{Ruido térmico} y errores de medición: a menudo \textbf{normal} (gaussiano).
    \item \textbf{Flujos/energías multiplicativos} y \textbf{masas} positivas: suelen ser \textbf{log-normales}.
    \item \textbf{Tiempos de espera} para eventos aleatorios: \textbf{exponencial}.
    \item \textbf{Conteos de fotones}: Poisson (discreta, fuera del alcance de este ejemplo continuo).
    \item Elegir bien la distribución \textbf{mejora la modelación de incertidumbre} y la \textbf{inferencia} posterior.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Generar datos sintéticos: normal, lognormal, exponencial}
\begin{minted}{python}
import numpy as np
rng = np.random.default_rng(42)

# Normal (media=0, sigma=1)
x_norm = rng.normal(0, 1, size=2000)

# Lognormal: si Z~N(mu,s), X=exp(Z) => log X ~ N(mu,s)
mu, s = 0.0, 0.6
x_logn = rng.lognormal(mean=mu, sigma=s, size=2000)

# Exponencial (escala=tau)
tau = 2.0
x_exp = rng.exponential(scale=tau, size=2000)
\end{minted}
	\textbf{Nota:} elegir tamaños suficientes para estimación estable de parámetros.
\end{frame}

\begin{frame}[fragile]{Ajustar parámetros con \texttt{scipy.stats}}
\begin{minted}{python}
from scipy import stats

# Normal: params = (mu_hat, sigma_hat)
mu_hat, sigma_hat = stats.norm.fit(x_norm)

# Lognormal en SciPy: shape=s, loc, scale; X ~ lognorm(s, loc, scale)
s_hat, loc_hat, scale_hat = stats.lognorm.fit(x_logn, floc=0)
mu_hat_logn = np.log(scale_hat)  # equivalente a mu del log

# Exponencial: params = (loc, scale=tau)
loc_e, tau_hat = stats.expon.fit(x_exp, floc=0)

print(mu_hat, sigma_hat)
print(s_hat, mu_hat_logn)
print(tau_hat)
\end{minted}
	\textbf{Por qué:} \texttt{fit} maximiza verosimilitud para obtener estimadores eficientes bajo supuestos.
\end{frame}

\begin{frame}[fragile]{Bondad de ajuste: KS y verosimilitud (AIC)}
\begin{minted}{python}
from scipy import stats
import numpy as np

# KS para normal
D, p = stats.kstest(x_norm, 'norm', args=(mu_hat, sigma_hat))

# AIC: 2k - 2 ln L
ll_norm = np.sum(stats.norm.logpdf(x_norm, mu_hat, sigma_hat))
k = 2
AIC_norm = 2*k - 2*ll_norm
print(D, p, AIC_norm)
\end{minted}
	\textbf{Lectura:} p grande no rechaza; AIC menor suele preferirse entre modelos comparables.
\end{frame}

\begin{frame}{¿Qué hace el código y por qué?}
  \begin{itemize}
    \item \textbf{Generación:} construimos datos para escenarios típicos (normal, lognormal, exponencial).
    \item \textbf{Ajuste:} usamos \texttt{.fit} de SciPy (MLE) para recuperar parámetros.
    \item \textbf{Diagnóstico:} KS verifica distancia entre CDFs; AIC compara modelos por parsimonia.
    \item \textbf{Cuidado:} lognormal en SciPy usa parámetros \textit{shape}\,\(s\), \textit{loc}, \textit{scale} (no directamente \(\mu,\sigma\)).
  \end{itemize}
\end{frame}

\begin{frame}{Notas prácticas de interpolación en física/astro}
  \begin{itemize}
    \item \textbf{No extrapolar:} fuera del rango de datos la interpolación se vuelve especulación.
    \item \textbf{Curvas de luz:} útil para rellenar huecos antes de integrar o ajustar un período.
    \item \textbf{Espectros:} alinear rejillas (grid) antes de comparar o integrar flujo en bandas.
    \item \textbf{Suavidad vs fidelidad:} \texttt{cubic} suaviza más que \texttt{linear}; elegir según física del fenómeno.
  \end{itemize}
\end{frame}
