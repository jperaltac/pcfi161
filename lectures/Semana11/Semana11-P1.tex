\section{Fundamentos de pandas}

\begin{frame}{¿Por qué pandas?}
  \begin{itemize}
    \item En física y astronomía muchas mediciones llegan como \textbf{tablas}: columnas, unidades, categorías y valores faltantes.
    \item NumPy es excelente para arreglos numéricos homogéneos.
    \item pandas agrega una capa tabular con nombres de columnas, índices y herramientas de análisis exploratorio.
  \end{itemize}

  \begin{block}{Idea central}
    Un \texttt{DataFrame} es una tabla con columnas etiquetadas. Cada columna puede representar una variable física, observacional o experimental.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Importar pandas}
\begin{minted}{python}
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
\end{minted}

  \begin{itemize}
    \item \texttt{pd} es el alias estándar de pandas.
    \item \texttt{np} se mantiene para cálculos numéricos.
    \item \texttt{plt} se usa para visualizar resultados.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Series: una columna con índice}
\begin{columns}
\begin{column}{0.52\textwidth}
\begin{minted}{python}

    [1.5, 2.0, 0.8],
    index=["objeto1", "objeto2", "objeto3"]

\textbf{Lectura:} una \texttt{Series} funciona como un vector, pero cada elemento puede tener una etiqueta.
\end{frame}

\begin{columns}
\begin{column}{0.50\textwidth}

    \item Cada columna tiene nombre.
    \item Cada fila representa una observación u objeto.

  \end{itemize}
\end{frame}

\section{Selección y filtrado}

\begin{frame}[fragile]{Seleccionar columnas}
\begin{columns}
\begin{column}{0.44\textwidth}
\begin{minted}{python}
# Una columna
df["masa"]

# Varias columnas
subset = df[["masa", "velocidad"]]

subset
\end{minted}
\end{column}
\begin{column}{0.52\textwidth}
  \begin{figure}
    \centering
    \includegraphics[width=\linewidth]{df_subset.png}
  \end{figure}
\end{column}
\end{columns}
\vspace{0.1cm}
\textbf{Regla práctica:} una columna se selecciona con un string; varias columnas con una lista de strings.
\end{frame}

\begin{frame}[fragile]{Seleccionar filas con \texttt{loc} e \texttt{iloc}}
\begin{minted}{python}
# Por posición
print(planets.iloc[0])
print(planets.iloc[0:5])

# Por condición con loc
transito = planets.loc[planets["method"] == "Transit"]

print(transito.head())
\end{minted}

  \begin{itemize}
    \item \texttt{iloc}: usa posiciones numéricas.
    \item \texttt{loc}: usa etiquetas o condiciones lógicas.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Filtros simples}
\begin{columns}
\begin{column}{0.46\textwidth}
\begin{minted}{python}
# Filtrado con una condición
altas_temp = df[
    df["temperatura"] > 300
]

altas_temp
\end{minted}
\end{column}
\begin{column}{0.50\textwidth}
  \begin{figure}
    \centering
    \includegraphics[width=\linewidth]{df_altas_temp.png}
  \end{figure}
\end{column}
\end{columns}
\vspace{0.1cm}
\textbf{Idea:} un filtro crea una máscara \texttt{True}/\texttt{False}; pandas conserva solo las filas verdaderas.
\end{frame}

\begin{frame}[fragile]{Filtros múltiples}
\begin{minted}{python}
mask = (
    (planets["method"] == "Radial Velocity") &
    (planets["mass"] > 1) &
    (planets["distance"] < 100)
)

seleccion = planets.loc[
    mask,
    ["method", "orbital_period", "mass", "distance"]
]

print(seleccion.head())
\end{minted}

  \begin{itemize}
    \item Use \texttt{\&} para condiciones simultáneas.
    \item Cada condición debe ir entre paréntesis.
  \end{itemize}
\end{frame}

\section{Agrupación y resumen}

\begin{frame}[fragile]{Agrupar con \texttt{groupby}}
\begin{columns}
\begin{column}{0.48\textwidth}
\begin{minted}{python}
resumen = df.groupby("categoria")[
    "temperatura"
].mean()

resumen
\end{minted}
\end{column}
\begin{column}{0.48\textwidth}
  \begin{figure}
    \centering
    \includegraphics[width=\linewidth]{df_temperatura.png}
  \end{figure}
\end{column}
\end{columns}
\vspace{0.1cm}
\textbf{Interpretación:} \texttt{groupby} separa la tabla por categorías y aplica una operación a cada grupo.
\end{frame}

\begin{frame}[fragile]{Agrupar con varias estadísticas}
\begin{minted}{python}
resumen = df.groupby("categoria").agg({
    "masa": ["mean", "min", "max"],
    "velocidad": ["mean", "std"],
    "temperatura": "mean"
})

resumen
\end{minted}

  \begin{itemize}
    \item \texttt{count}: cantidad de datos no nulos.
    \item \texttt{mean}: promedio.
    \item \texttt{median}: mediana.
    \item \texttt{max}: valor máximo.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ordenar resultados}
\begin{minted}{python}
conteos = df["categoria"].value_counts()
print(conteos.head())

categorias_comunes = conteos.sort_values(ascending=False)
print(categorias_comunes.head())
\end{minted}

  \begin{block}{Uso típico}
    Ordenar ayuda a identificar categorías dominantes, valores extremos o métodos más frecuentes.
  \end{block}
\end{frame}

\section{Ejercicio integrador}

\begin{frame}{Ejercicio guiado: exploración de exoplanetas}
  Use el dataset \texttt{planets.csv} y realice:
  \begin{enumerate}
    \item Leer el CSV con \texttt{pd.read\_csv}.
    \item Mostrar \texttt{head}, \texttt{shape}, \texttt{columns} e \texttt{info}.
    \item Crear un DataFrame limpio usando columnas relevantes.
    \item Filtrar planetas detectados por velocidad radial.
    \item Calcular distancia promedio y masa promedio por método.
    \item Graficar masa versus periodo orbital.
    \item Exportar el subconjunto filtrado.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile,shrink=5]{Solución guiada: lectura, limpieza y filtro}
\begin{minted}{python}
import pandas as pd
import matplotlib.pyplot as plt

url = (
    "https://raw.githubusercontent.com/"
    "mwaskom/seaborn-data/master/planets.csv"
)

planets = pd.read_csv(url)

print(planets.head())
print(planets.shape)
print(planets.columns)
planets.info()

cols = ["method", "orbital_period", "mass", "distance", "year"]
datos = planets[cols].dropna()

rv = datos[datos["method"] == "Radial Velocity"]
print(rv.head())
\end{minted}
\end{frame}

\begin{frame}[fragile,shrink=5]{Solución guiada: resumen, gráfico y exportación}
\begin{minted}{python}
resumen = datos.groupby("method").agg({
    "distance": "mean",
    "mass": "mean",
    "orbital_period": "median"
})

print(resumen)

plt.figure(figsize=(7, 5))
plt.scatter(rv["orbital_period"], rv["mass"], alpha=0.6)
plt.xlabel("Periodo orbital [días]")
plt.ylabel("Masa [M_J]")
plt.title("Planetas detectados por velocidad radial")
plt.grid(True, alpha=0.3)
plt.show()

rv.to_csv("planetas_velocidad_radial.csv", index=False)
\end{minted}
\end{frame}

\begin{frame}{Errores comunes}
  \begin{itemize}
    \item Olvidar importar pandas como \texttt{pd}.

    \item Usar \texttt{and} en vez de \texttt{\&} para filtros de pandas.
    \item Calcular promedios sin revisar valores faltantes.
    \item Modificar una tabla sin guardar una copia cuando se quiere conservar el original.
  \end{itemize}
\end{frame}
