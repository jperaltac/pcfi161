\section{Ajustes no lineales}

\begin{frame}[fragile]{Exponencial: \(y = A\, e^{-t/\tau} + C\)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit

rng = np.random.default_rng(123)
t = np.linspace(0, 10, 60)
A_true, tau_true, C_true = 5.0, 2.0, 0.5
sigma = 0.2

def expo(t, A, tau, C):
    return A*np.exp(-t/tau) + C

y_true = expo(t, A_true, tau_true, C_true)
y_obs = y_true + rng.normal(0, sigma, size=t.size)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
# Ajuste con curva inicial razonable
p0 = (4.0, 1.0, 0.0)
popt, pcov = curve_fit(expo, t, y_obs, p0=p0,
                       sigma=np.full_like(t, sigma),
                       absolute_sigma=True)
perr = np.sqrt(np.diag(pcov))
A, tau, C = popt
print(f"A={A:.2f}±{perr[0]:.2f}, tau={tau:.2f}±{perr[1]:.2f}, C={C:.2f}±{perr[2]:.2f}")

plt.scatter(t, y_obs, s=12)
plt.plot(t, expo(t,*popt), 'r')
plt.grid(alpha=0.3); plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Tips:} las conjeturas iniciales (\texttt{p0}) ayudan a converger al mínimo correcto.
\end{frame}

\begin{frame}[fragile]{Potencia: \(y = k\, x^{\alpha}\) (no lineal vs log–log)}
\begin{minted}{python}
from scipy.optimize import curve_fit

def power(x, k, alpha):
    return k * x**alpha

x = np.linspace(1, 20, 40)
alpha_true, k_true, sigma_rel = 1.7, 2.0, 0.08
y_true = power(x, k_true, alpha_true)
rng = np.random.default_rng(5)
y_obs = y_true * (1 + rng.normal(0, sigma_rel, size=x.size))

popt, pcov = curve_fit(power, x, y_obs, p0=(1.0, 1.0))
print('k, alpha =', popt)
\end{minted}
	\textbf{Comentario:} comparar con ajuste lineal en \(\log\)–\(\log\) para entender potenciales sesgos.
\end{frame}

\section{Propagación de errores}

\begin{frame}[fragile]{Errores de parámetros desde la covarianza}
\begin{minted}{python}
# Desde curve_fit: pcov -> varianzas en la diagonal
perr = np.sqrt(np.diag(pcov))
for name, val, err in zip(['k','alpha'], popt, perr):
    print(f"{name} = {val:.3f} ± {err:.3f}")
\end{minted}
	\textbf{Interpretación:} asume errores gaussianos y modelo adecuado alrededor del óptimo.
\end{frame}

\begin{frame}[fragile]{Monte Carlo: muestreo de parámetros y banda de confianza}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Muestras de parámetros ~ N(popt, pcov)
ns = 2000
theta_s = np.random.multivariate_normal(popt, pcov, size=ns)
xx = np.linspace(x.min(), x.max(), 200)

Y = np.array([power(xx, *th) for th in theta_s])
y_med = np.median(Y, axis=0)
y_lo, y_hi = np.percentile(Y, [16,84], axis=0)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
plt.scatter(x, y_obs, s=12, label='datos')
plt.plot(xx, y_med, 'r', label='ajuste (mediana)')
plt.fill_between(xx, y_lo, y_hi, color='r', alpha=0.2,
                 label='~68% conf.')
plt.legend(); plt.grid(alpha=0.3)
plt.xlabel('x'); plt.ylabel('y')
plt.title('Banda de confianza por MC')
plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Ventaja:} no requiere linealizar; captura correlaciones entre parámetros.
\end{frame}

\begin{frame}[fragile]{Enfoques prácticos}
\begin{itemize}

\begin{frame}{Tarea Pregunta: \hfill \textcolor{blue}{$\bigstar$} \\ Ajuste de curva de luz estelar}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Descargue o genere una \textbf{curva de luz} (tiempo vs flujo) de una estrella con variabilidad simple.
      \item Proponga un \textbf{modelo} adecuado, por ejemplo:
        \begin{itemize}
          \item \(F(t) = A\,\sin(2\pi t/P + \phi) + C\) (rotación/mancha)
          \item o pulso/gaussiana para un \textit{evento} breve
        \end{itemize}
      \item Ajuste parámetros con \texttt{curve\_fit}, estime errores (\(\sqrt{\mathrm{diag}(\mathrm{cov})}\)).
      \item Visualice la \textbf{banda de confianza} (68%) y reporte \(\chi^2_\mathrm{red}\).
    \end{itemize}
  \end{block}
  	\textbf{Entrega:} notebook o script + figuras (PDF/PNG). \textbf{Relevancia:} análisis fotométrico básico en astronomía.
\end{frame}

import numpy as np
from scipy.optimize import curve_fit

def seno(t, A, P, phi, C):
    return A*np.sin(2*np.pi*t/P + phi) + C

rng = np.random.default_rng(10)
t = np.linspace(0, 10, 200)
F_true = seno(t, 0.02, 2.5, 0.3, 1.0)
sigma = 0.005
F_obs = F_true + rng.normal(0, sigma, size=t.size)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=14]{python}
p0 = (0.01, 2.0, 0.0, 1.0)
popt, pcov = curve_fit(seno, t, F_obs, p0=p0,
                       sigma=np.full_like(t, sigma),
                       absolute_sigma=True)
perr = np.sqrt(np.diag(pcov))
print('parámetros ±1σ:', popt, perr)

# Construya banda de confianza (MC) y grafique
\end{minted}
