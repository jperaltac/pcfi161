\begin{frame}{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Crear una columna física}

\begin{frame}{Ejercicio 6: \hfill \textcolor{red}{$\clubsuit$} \\ Datos faltantes}
  \begin{block}{Enunciado}

      \item Detecte valores faltantes con \texttt{df.isnull().sum()}.
      \item Cree \texttt{df\_clean} eliminando filas con faltantes en \texttt{magnitud\_V}.
      \item Cree \texttt{df\_filled} rellenando \texttt{magnitud\_V} con su promedio.
      \item Compare cuántas filas hay en cada DataFrame.
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} \texttt{dropna}, \texttt{fillna}, copias limpias.

\begin{solutionblock}
\begin{frame}[fragile,shrink=5]{Solución 6: datos faltantes}
\begin{minted}{python}
print(df.isnull().sum())

df_clean = df.dropna(subset=["magnitud_V"])

mag_media = df["magnitud_V"].mean()
df_filled = df.copy()
df_filled["magnitud_V"] = df_filled["magnitud_V"].fillna(mag_media)

print("Original:", len(df))
print("Sin nulos:", len(df_clean))
print("Rellenado:", len(df_filled))
print(df_filled[["estrella", "magnitud_V"]])
\end{minted}
\end{frame}
\end{solutionblock}

\begin{frame}{Ejercicio 7: \hfill \textcolor{red}{$\clubsuit$} \\ Agrupación por categoría}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Agrupe por \texttt{categoria}.
      \item Calcule media de \texttt{temperatura\_K}, \texttt{distancia\_pc} y \texttt{luminosidad\_rel}.
      \item Calcule también cuántas estrellas hay en cada categoría.
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} \texttt{groupby}, \texttt{agg}, conteos por grupo.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile,shrink=5]{Solución 7: agrupación por categoría}
\begin{minted}{python}
if "luminosidad_rel" not in df.columns:
    df["luminosidad_rel"] = (
        df["radio_Rsol"]**2 *
        (df["temperatura_K"] / 5778)**4
    )

resumen = df.groupby("categoria").agg({
    "estrella": "count",
    "temperatura_K": "mean",
    "distancia_pc": "mean",
    "luminosidad_rel": "mean"
})

print(resumen)
\end{minted}
\end{frame}
\end{solutionblock}
