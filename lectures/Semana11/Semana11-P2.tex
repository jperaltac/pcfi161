\begin{frame}{Ejercicio 1: \hfill \textcolor{red}{$\clubsuit$} \\ Calibración de sensor de temperatura}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Un sensor de temperatura mide voltaje \(V\) según \(V = aT + b\)
      \item Simule 12 medidas en \(T = 0, 10, 20, ..., 110\)°C
      \item Use \(a_{\text{real}} = 0.018\) V/°C, \(b_{\text{real}} = 0.15\) V
      \item Agregue ruido gaussiano \(\sigma = 0.02\) V (constante)
      \item \textbf{Tareas}:
        \begin{enumerate}
          \item Ajuste con \texttt{np.polyfit(T, V, 1)}
          \item Calcule y reporte \(R^2\)
          \item Grafique datos y recta ajustada
          \item Compare \(a_{\text{est}}\) y \(b_{\text{est}}\) con valores reales
        \end{enumerate}
    \end{itemize}
  \end{block}
  
  \textbf{Conceptos}: Ajuste lineal básico, \(R^2\)
\end{frame}

% ------------------------------------------------------------------------
% Slide: Ejercicio 2

\begin{frame}{Ejercicio 2: \hfill \textcolor{red}{$\clubsuit$} \\ Ley de Hubble (análisis completo)}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Relación de Hubble: \(v = H_0 d + b\)
      \item Simule 25 galaxias con \(d \in [10, 200]\) Mpc (espaciado uniforme)
      \item Use \(H_0 = 70.0\) km/s/Mpc, \(b = 0\) km/s
      \item Agregue ruido \(\sigma_v = 180\) km/s (velocidades peculiares)
      \item \textbf{Tareas}:
        \begin{enumerate}
          \item Ajuste con ponderación: \texttt{np.polyfit(d, v, 1, w=1/sigma\_v)}
          \item Calcule \(\chi^2_{\text{red}}\)
          \item Grafique: (a) datos + ajuste con barras de error, (b) residuos vs distancia
          \item Interprete: ¿es \(\chi^2_{\text{red}} \approx 1\)?
        \end{enumerate}
    \end{itemize}
  \end{block}
  
  \textbf{Conceptos}: Ponderación, \(\chi^2_{\text{red}}\), residuos, aplicación cosmológica
\end{frame}

% ------------------------------------------------------------------------
% Solución 1
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 1: \hfill \textcolor{green}{$\checkmark$} \\ Calibración de sensor}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Generar datos sintéticos
np.random.seed(42)
T = np.arange(0, 120, 10)
a_true, b_true = 0.018, 0.15
sigma = 0.02
V = a_true*T + b_true + \
    np.random.normal(0, sigma, 
                     len(T))

# Ajustar con polyfit
a_est, b_est = np.polyfit(T, V, 1)
V_fit = a_est*T + b_est
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
# Calcular R^2
SS_res = np.sum((V - V_fit)**2)
SS_tot = np.sum((V - np.mean(V))**2)
R2 = 1 - SS_res/SS_tot

# Graficar
plt.scatter(T, V, label='Datos')
plt.plot(T, V_fit, 'r', 
         label=f'Ajuste: '
         f'a={a_est:.4f}, '
         f'b={b_est:.3f}')
plt.xlabel('Temperatura (°C)')
plt.ylabel('Voltaje (V)')
plt.legend(); plt.grid()
plt.show()

print(f"a = {a_est:.4f} V/°C")
print(f"b = {b_est:.3f} V")
print(f"R² = {R2:.4f}")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 2: \hfill \textcolor{green}{$\checkmark$} \\ Ley de Hubble (Parte 1: Ajuste)}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Generar datos
np.random.seed(10)
H0_true, sigma_v = 70.0, 180.0
d = np.linspace(10, 200, 25)
v_obs = H0_true*d + \
        np.random.normal(0, sigma_v,
                        len(d))

# Ajuste CON ponderación
w = 1.0 / sigma_v
H0_est, b_est = np.polyfit(
    d, v_obs, 1, w=w)
v_fit = H0_est*d + b_est
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=17]{python}
# Calcular chi^2_red
res = v_obs - v_fit
chi2 = np.sum((res/sigma_v)**2)
nu = len(d) - 2
chi2_red = chi2/nu

print(f"H₀ = {H0_est:.1f} "
      f"km/s/Mpc")
print(f"b = {b_est:.1f} km/s")
print(f"χ²_red = {chi2_red:.2f}")

# Interpretación
if 0.5 < chi2_red < 2.0:
    print("✓ Ajuste consistente")
else:
    print("⚠ Revisar modelo")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Errores Comunes con \texttt{np.polyfit}}
  \begin{block}{Errores frecuentes}
    \begin{itemize}
      \item \textbf{Olvidar el grado del polinomio}: \texttt{np.polyfit(x, y, 1)} para lineal
      \item \textbf{No revisar el \(\chi^2_{\text{red}}\)}: \(R^2\) alto no garantiza consistencia con incertidumbres
      \item \textbf{Ignorar el parámetro \texttt{w}}: Esencial cuando hay ponderación
      \item \textbf{No graficar residuos}: Patrones sistemáticos indican modelo inadecuado
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Consejo}
    \textbf{Workflow}: Visualizar datos → Ajustar con \texttt{np.polyfit} → Calcular \(R^2\) y \(\chi^2_{\text{red}}\) → Graficar residuos → Interpretar
  \end{alertblock}
\end{frame}
