\begin{frame}[fragile]{Solución 1: \hfill \textcolor{green}{$\checkmark$} \\ Calibración lineal}
\begin{minted}{python}
import random, math
from scipy.optimize import curve_fit

# Generar datos sintéticos del sensor
T = [i*10 for i in range(12)]  # Temperaturas de 0 a 110°C
a_true, b_true, sigma = 0.018, 0.15, 0.02
V = [a_true*t + b_true + random.gauss(0.0, sigma) for t in T]

# Definir modelo lineal
def modelo_lineal(x, a, b):
    return a*x + b

# Ajustar con curve_fit
sig = [sigma]*len(T)
popt, _ = curve_fit(modelo_lineal, T, V, sigma=sig, absolute_sigma=True)
a_est, b_est = popt

# Calcular métricas de calidad
V_fit = [modelo_lineal(t, a_est, b_est) for t in T]
media_V = sum(V)/len(V)
SS_res = sum((vi - vfi)**2 for vi, vfi in zip(V, V_fit))
SS_tot = sum((vi - media_V)**2 for vi in V)
R2 = 1 - SS_res/SS_tot
chi2_red = sum(((vi - vfi)/sigma)**2 for vi, vfi in zip(V, V_fit))/(len(T)-2)

print(f"Sensibilidad: a={a_est:.4f} V/°C")
print(f"Offset: b={b_est:.3f} V")
print(f"R²={R2:.3f}, χ²_red={chi2_red:.2f}")
\end{minted}
\textbf{Lectura:} \(\chi^2_\text{red}\approx 1\) confirma que \(\sigma\) estimado es razonable.
\end{frame}

\begin{frame}[fragile]{Solución 2: \hfill \textcolor{green}{$\checkmark$} \\ Hubble con diagnóstico}
\begin{minted}{python}
import random, math
from scipy.optimize import curve_fit
import matplotlib.pyplot as plt

# Generar datos de galaxias
H0_true, sigma_v = 70.0, 180.0
n = 25
d = [10 + i*(190/(n-1)) for i in range(n)]  # 10 a 200 Mpc
v_obs = [H0_true*di + random.gauss(0.0, sigma_v) for di in d]

# Ajustar modelo de Hubble
def hubble(d, H0, b):
    return H0*d + b

sig = [sigma_v]*n
popt, pcov = curve_fit(hubble, d, v_obs, sigma=sig, absolute_sigma=True)
H0_est, b_est = popt
perr = [math.sqrt(pcov[i][i]) for i in range(len(popt))]

# Calcular chi2_red
v_fit = [hubble(di, H0_est, b_est) for di in d]
res = [vi - vfi for vi, vfi in zip(v_obs, v_fit)]
chi2_red = sum((ri/sigma_v)**2 for ri in res)/(n-2)

print(f"H0 = {H0_est:.1f} ± {perr[0]:.1f} km/s/Mpc")
print(f"Intercepto b = {b_est:.1f} km/s")
print(f"χ²_red = {chi2_red:.2f}")
\end{minted}
\end{frame}

\begin{minted}{python}
from scipy.interpolate import interp1d
import random, math
import matplotlib.pyplot as plt

# Generar curva de luz completa
P = 2.5  # periodo
t_full = [i*0.05 for i in range(201)]  # 0 a 10 en 201 puntos
F_full = [1.0 + 0.1*math.sin(2*math.pi*t/P) for t in t_full]

# Eliminar 20 puntos al azar
random.seed(42)
indices = random.sample(range(len(t_full)), len(t_full) - 20)
indices.sort()
t_obs = [t_full[i] for i in indices]
F_obs = [F_full[i] for i in indices]

# Crear interpoladores
f_lin = interp1d(t_obs, F_obs, kind='linear')
f_cub = interp1d(t_obs, F_obs, kind='cubic')

# Evaluar en grilla fina
t_interp = [i*0.025 for i in range(401)]  # más fino para suavidad
F_lin = [f_lin(t) for t in t_interp]
F_cub = [f_cub(t) for t in t_interp]
\end{minted}
\end{frame}

\begin{frame}[fragile]{Solución 4: \hfill \textcolor{green}{$\checkmark$} \\ Lognormal vs Normal}
\begin{minted}{python}
from scipy import stats
import random
import matplotlib.pyplot as plt
import numpy as np

# Generar datos lognormales
random.seed(42)
n = 1500
mu, sigma = 0.0, 0.6
x = [random.lognormvariate(mu, sigma) for _ in range(n)]

# Ajustar distribución Lognormal
s_log, _, scale_log = stats.lognorm.fit(x, floc=0)
ll_log = sum(stats.lognorm.logpdf(xi, s_log, 0, scale_log) for xi in x)
k_log = 2  # parámetros: s y scale
AIC_log = 2*k_log - 2*ll_log

# Ajustar distribución Normal
mu_norm, sig_norm = stats.norm.fit(x)
ll_norm = sum(stats.norm.logpdf(xi, mu_norm, sig_norm) for xi in x)
k_norm = 2  # parámetros: mu y sigma
AIC_norm = 2*k_norm - 2*ll_norm

print(f"Lognormal: AIC = {AIC_log:.1f}")
print(f"Normal: AIC = {AIC_norm:.1f}")
print(f"Mejor modelo: {'Lognormal' if AIC_log < AIC_norm else 'Normal'}")
\end{minted}
\end{frame}

         label='Histograma')

# Generar curvas de densidad
x_plot = np.linspace(0, max(x), 500)
pdf_log = stats.lognorm.pdf(x_plot, s_log, 0, scale_log)
pdf_norm = stats.norm.pdf(x_plot, mu_norm, sig_norm)

plt.plot(x_plot, pdf_log, 'r-', linewidth=2, 
         label=f'Lognormal (AIC={AIC_log:.1f})')
plt.plot(x_plot, pdf_norm, 'b--', linewidth=2, 
         label=f'Normal (AIC={AIC_norm:.1f})')

plt.xlabel('Valor')
plt.ylabel('Densidad de Probabilidad')
plt.title('Comparación de Ajustes: Lognormal vs Normal')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\textbf{Conclusión:} Lognormal tiene menor AIC y captura mejor la asimetría positiva.
\end{frame}
