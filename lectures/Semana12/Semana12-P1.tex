\section{Ajuste con \texttt{scipy.optimize.curve\_fit}}

\begin{frame}{¿Qué es \texttt{curve\_fit}?}
  \begin{block}{Definición}
    \texttt{curve\_fit} es una función de \texttt{scipy.optimize} que ajusta cualquier función arbitraria a datos mediante mínimos cuadrados no lineales.
  \end{block}
  
  \begin{itemize}
    \item \textbf{Diferencia clave con \texttt{polyfit}:}
      \begin{itemize}
        \item \texttt{polyfit}: limitado a polinomios (\(y = a_0 + a_1 x + a_2 x^2 + ...\))
        \item \texttt{curve\_fit}: \textbf{cualquier función} (exponenciales, trigonométricas, físicas personalizadas, etc.)
      \end{itemize}
    \item \textbf{Ventajas principales:}
      \begin{enumerate}
        \item Flexibilidad total en la forma funcional
        \item Calcula automáticamente incertidumbres de parámetros
        \item Proporciona matriz de covarianza completa
      \end{enumerate}
  \end{itemize}
\end{frame}

\begin{frame}{Flexibilidad de \texttt{curve\_fit}: Ejemplos de funciones}
  \begin{block}{Ejemplos físicos}
    \texttt{curve\_fit} puede ajustar modelos como:
  \end{block}
  
  \begin{itemize}
    \item \textbf{Decaimiento exponencial:} \(N(t) = N_0 e^{-\lambda t}\) (radiactividad)
    \item \textbf{Oscilador amortiguado:} \(x(t) = A e^{-\gamma t} \cos(\omega t + \phi)\)
    \item \textbf{Ley de potencia:} \(L = k M^{\alpha}\) (masa-luminosidad)
    \item \textbf{Gaussiana:} \(f(x) = A e^{-(x-\mu)^2/(2\sigma^2)}\) (perfiles espectrales)
    \item \textbf{Planck:} \(B(\lambda, T) = \frac{2hc^2}{\lambda^5} \frac{1}{e^{hc/(\lambda k_B T)} - 1}\)
    \item \textbf{Y cualquier combinación o modelo personalizado!}
  \end{itemize}
  
  \begin{alertblock}{Clave}
    Solo necesitas definir tu función en Python y \texttt{curve\_fit} encuentra los parámetros óptimos.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Estructura básica de \texttt{curve\_fit}}
\begin{minted}{python}
from scipy.optimize import curve_fit

# Paso 1: Definir tu función (modelo físico)
def mi_modelo(x, param1, param2, ...):
    """Función que describe tu fenómeno físico"""
    return ...  # fórmula que depende de x y parámetros

# Paso 2: Llamar a curve_fit
popt, pcov = curve_fit(
    mi_modelo,          # función a ajustar
    x_datos,            # variable independiente
    y_datos,            # datos observados
    p0=[...],           # valores iniciales (opcional)
    sigma=sigma_datos,  # incertidumbres (opcional)
    absolute_sigma=True # usar sigma como absolutos
)

# Paso 3: Extraer resultados
params_optimos = popt       # parámetros ajustados
incertidumbres = np.sqrt(np.diag(pcov))  # errores
\end{minted}
\end{frame}

\begin{frame}[fragile]{\texttt{curve\_fit} en un modelo lineal (ejemplo simple)}
\begin{minted}{python}
from scipy.optimize import curve_fit
import numpy as np

# Definir modelo lineal: y = m*x + b
def modelo_lineal(x, m, b):
    return m * x + b

# Datos del ejemplo de Hubble (del código anterior)
# d = distancias, v_obs = velocidades observadas
# sigma_v = incertidumbre en velocidades

# Ajustar el modelo a los datos
popt, pcov = curve_fit(
    modelo_lineal,           # nuestra función
    d,                       # x: distancias
    v_obs,                   # y: velocidades observadas
    sigma=sigma_v,           # incertidumbres
    absolute_sigma=True      # tratar sigma como valores reales
)

m_cf, b_cf = popt  # extraer parámetros optimizados
print(f"Pendiente: m = {m_cf:.2f} km/s/Mpc")
print(f"Intercepto: b = {b_cf:.2f} km/s")
\end{minted}
\end{frame}

\begin{frame}{Matriz de covarianza: ¿Qué es \texttt{pcov}?}
  \begin{block}{Definición}
    \texttt{pcov} es la matriz de covarianza de los parámetros ajustados. Contiene información sobre las incertidumbres y correlaciones entre parámetros.
  \end{block}
  
  \begin{itemize}
    \item \textbf{Para 2 parámetros} (ej: \(m, b\)):
    \[
    \text{pcov} = \begin{pmatrix}
      \sigma_m^2 & \text{cov}(m,b) \\
      \text{cov}(m,b) & \sigma_b^2
    \end{pmatrix}
    \]
    \item \textbf{Diagonal} (\texttt{np.diag(pcov)}): varianzas \(\sigma_i^2\) de cada parámetro
    \item \textbf{Fuera de diagonal}: covarianzas entre parámetros
    \item \textbf{Incertidumbres:} \(\sigma_i = \sqrt{\text{pcov}[i,i]}\)
  \end{itemize}
  
  \begin{alertblock}{Importante}
    Si \texttt{absolute\_sigma=True}, las incertidumbres son reales. Si es \texttt{False} (por defecto), son relativas y deben escalarse.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Extrayendo incertidumbres de \texttt{pcov}}
\begin{minted}{python}
# pcov es una matriz 2x2 para el modelo lineal (m, b)
print("Matriz de covarianza:")
print(pcov)
# Ejemplo de salida:
# [[0.0234  -0.1102]
#  [-0.1102  1.4567]]

# Extraer incertidumbres (errores estándar)
perr = np.sqrt(np.diag(pcov))
sigma_m = perr[0]  # incertidumbre de la pendiente
sigma_b = perr[1]  # incertidumbre del intercepto

print(f"\nParámetros con incertidumbres:")
print(f"m = {m_cf:.3f} ± {sigma_m:.3f} km/s/Mpc")
print(f"b = {b_cf:.3f} ± {sigma_b:.3f} km/s")

# Interpretación física para Hubble:
# H0 = 70.5 ± 2.3 km/s/Mpc significa que con ~68% de confianza,
# el valor real está entre 68.2 y 72.8 km/s/Mpc
\end{minted}
\end{frame}

\begin{frame}{¿Qué representa \texttt{sigma} (\(\sigma\))?}
  \begin{block}{Definición simple}
    \texttt{sigma} (\(\sigma_i\)) es la \textbf{incertidumbre} o \textbf{error} asociado a cada medición. Nos dice qué tan confiable es cada dato.
  \end{block}
  
  \begin{itemize}
    \item \textbf{Interpretación física:}
      \begin{itemize}
        \item \(\sigma\) pequeño: medición precisa, alta confianza
        \item \(\sigma\) grande: medición imprecisa, baja confianza
      \end{itemize}
    \item \textbf{¿De dónde viene \(\sigma\)?}
      \begin{itemize}
        \item Especificaciones del instrumento de medición
        \item Repetición de mediciones (desviación estándar)
        \item Análisis de fuentes de error experimental
      \end{itemize}
    \item \textbf{Ejemplo concreto:} Si medimos una temperatura como \(T = 25.3 \pm 0.5\,°\text{C}\), entonces \(\sigma = 0.5\,°\text{C}\)
  \end{itemize}
\end{frame}

\begin{frame}{Importancia de \(\sigma\) en los ajustes}
  \begin{itemize}
    \item \textbf{En \texttt{curve\_fit}:} el parámetro \texttt{sigma} permite ponderar los datos
      \begin{itemize}
        \item Puntos con \(\sigma\) pequeño → mayor peso en el ajuste
        \item Puntos con \(\sigma\) grande → menor peso en el ajuste
      \end{itemize}
    \item \textbf{En \(\chi^2\) reducido:} \(\sigma\) es esencial para evaluar consistencia
      \[
      \chi^2_{\text{red}} = \frac{1}{\nu} \sum_{i=1}^{N} \frac{(y_i - y_{\text{fit},i})^2}{\sigma_i^2}
      \]
    \item \textbf{Resultado:} Ajustes más realistas que reflejan la calidad de cada medición
  \end{itemize}
  
  \begin{alertblock}{Consejo práctico}
    Si no conoces \(\sigma_i\) individual, puedes usar el mismo valor para todos los puntos (incertidumbre uniforme).
  \end{alertblock}
\end{frame}

\begin{frame}{Comparación: \texttt{curve\_fit} vs \texttt{polyfit}}
  \begin{center}
  \begin{tabular}{|l|c|c|}
    \hline
    \textbf{Característica} & \textbf{polyfit} & \textbf{curve\_fit} \\
    \hline
    Tipos de función & Solo polinomios & \textbf{Cualquiera} \\
    \hline
    Incertidumbres (\(\sigma_i\)) & Pesos \(w\) & Directo \\
    \hline
    Errores de parámetros & No (manual) & \textbf{Sí (automático)} \\
    \hline
    Matriz covarianza & No & \textbf{Sí} \\
    \hline
    Valores iniciales & No necesita & A veces necesario \\
    \hline
    Velocidad & Rápido & Más lento \\
    \hline
    Simplicidad & Muy simple & Moderada \\
    \hline
  \end{tabular}
  \end{center}
  
  \vspace{0.3cm}
  \begin{block}{Cuándo usar cada uno}
    \begin{itemize}
      \item \textbf{polyfit}: ajustes polinomiales simples, exploración rápida
      \item \textbf{curve\_fit}: modelos físicos complejos, necesitas errores rigurosos
    \end{itemize}
  \end{block}
\end{frame}

\section{Relación masa–luminosidad}

\begin{frame}[fragile]{Potencia: \(L = k\, M^{\alpha}\). Enfoque log–log}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
# Simular estrellas
rng = np.random.default_rng(1)
alpha_true = 3.5
k_true = 1.0
M = np.linspace(0.5, 10, 40)
L_true = k_true * M**alpha_true
sigma_rel = 0.10
L_obs = L_true * (1 + rng.normal(0, sigma_rel, size=M.size))

# Transformación log10
X = np.log10(M)
Y = np.log10(L_obs)
a, b = np.polyfit(X, Y, 1)  # Y = a X + b
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=16]{python}
alpha_hat = a
k_hat = 10**b
print(f"alpha≈{alpha_hat:.2f}, k≈{k_hat:.2f}")

import matplotlib.pyplot as plt
plt.scatter(X, Y, s=20)
plt.plot(X, a*X + b, 'r')
plt.xlabel('log M'); plt.ylabel('log L')
plt.title('Masa–Luminosidad (log–log)')
plt.grid(alpha=0.3); plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
	\textbf{Discusión:} la linealización facilita el ajuste de leyes de potencia.
\end{frame}

\begin{frame}{¿Qué hace este código (log–log)?}
  \begin{itemize}
    \item Simula \(M\) y \(L\) con una ley de potencia y ruido multiplicativo (más realista en astrofísica).
    \item Transforma a \(\log\) para linealizar: \(\log L = \alpha\,\log M + \log k\).
    \item Ajusta con \texttt{polyfit} en el espacio logarítmico y recupera \(\alpha\) y \(k\).
    \item Discute diferencias con el ajuste no lineal en el espacio original.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Potencia directa con \texttt{curve\_fit} (no lineal)}
\begin{minted}{python}
from scipy.optimize import curve_fit

def power_law(M, k, alpha):
    return k * M**alpha

popt, pcov = curve_fit(power_law, M, L_obs, p0=(1.0, 3.0))
k_cf, alpha_cf = popt
perr = np.sqrt(np.diag(pcov))
print(f"k={k_cf:.2f}±{perr[0]:.2f}, alpha={alpha_cf:.2f}±{perr[1]:.2f}")
# Graficar resultados
plt.scatter(M, L_obs, s=20, label='estrellas')
M_fit = np.linspace(0.5, 10, 100)
plt.plot(M_fit, power_law(M_fit, *popt), 'r', label='fit potencia')

\end{minted}
	\textbf{Comparación:} log–log vs ajuste no lineal; considerar sesgos con ruido multiplicativo.
\end{frame}

\begin{frame}{Física detrás de \(L \propto M^{\alpha}\)}
  \begin{itemize}
    \item \(\alpha\) captura cómo cambia la \textbf{producción de energía} con la masa (tasa de fusión y estructura interna).
    \item Para estrellas de la secuencia principal, típicamente \(\alpha \approx 3{-}4\) en ciertos rangos de masa.
    \item \textbf{Escalas y unidades:} cuidado con trabajar en unidades consistentes; la forma funcional es adimensional si se normaliza.
    \item \textbf{Log–log vs no lineal:} la transformación logarítmica linealiza pero cambia la distribución de errores; el ajuste no lineal trabaja en el espacio original.
  \end{itemize}
\end{frame}
