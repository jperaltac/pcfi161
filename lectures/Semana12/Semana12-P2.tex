\section{Resumen estadístico}

\begin{frame}{Ejercicio 3: \hfill \textcolor{red}{$\clubsuit$} \\ Estadística descriptiva}
  \begin{block}{Enunciado}
    Para la columna \texttt{temperatura\_K}, calcule:
    \begin{itemize}
      \item media;
      \item mediana;
      \item desviación estándar;
      \item mínimo y máximo;
      \item cuartiles 25\%, 50\% y 75\%.
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} tendencia central, dispersión y cuantiles.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile,shrink=5]{Solución 3: estadística descriptiva}
\begin{minted}{python}
temp = df["temperatura_K"]

resumen_temp = pd.Series({
    "media": temp.mean(),
    "mediana": temp.median(),
    "desviacion": temp.std(),
    "minimo": temp.min(),
    "maximo": temp.max()
})

print(resumen_temp)
print(temp.quantile([0.25, 0.50, 0.75]))
\end{minted}
\end{frame}
\end{solutionblock}

\section{Comparación por grupos}

\begin{frame}{Ejercicio 5: \hfill \textcolor{red}{$\clubsuit$} \\ Agrupar por tipo}
  \begin{block}{Enunciado}
    \begin{itemize}
      \item Agrupe el DataFrame por \texttt{tipo}.
      \item Calcule promedio y mediana de \texttt{temperatura\_K}.
      \item Calcule promedio y desviación estándar de \texttt{radio\_Rsol}.
      \item Ordene el resumen por temperatura promedio.
    \end{itemize}
  \end{block}
  \textbf{Conceptos:} \texttt{groupby}, \texttt{agg}, comparación entre categorías.
\end{frame}

\begin{solutionblock}
\begin{frame}[fragile,shrink=5]{Solución 5: agrupar por tipo}
\begin{minted}{python}
resumen_tipo = df.groupby("tipo").agg({
    "temperatura_K": ["mean", "median"],
    "radio_Rsol": ["mean", "std"],
    "objeto": "count"
})

resumen_tipo = resumen_tipo.sort_values(
    ("temperatura_K", "mean")
)

print(resumen_tipo)
\end{minted}
\end{frame}
\end{solutionblock}
