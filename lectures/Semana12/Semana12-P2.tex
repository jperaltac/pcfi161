\begin{frame}[fragile]{Solución 5: \hfill \textcolor{green}{$\checkmark$} \\ Interpolación de curva de luz}
\begin{minted}{python}
from scipy.interpolate import interp1d
import random, math
import matplotlib.pyplot as plt

# Generar curva de luz completa
P = 2.5  # periodo
t_full = [i*0.05 for i in range(201)]  # 0 a 10 en 201 puntos
F_full = [1.0 + 0.1*math.sin(2*math.pi*t/P) for t in t_full]

# Eliminar 20 puntos al azar
random.seed(42)
indices = random.sample(range(len(t_full)), len(t_full) - 20)
indices.sort()
t_obs = [t_full[i] for i in indices]
F_obs = [F_full[i] for i in indices]

# Crear interpoladores
f_lin = interp1d(t_obs, F_obs, kind='linear')
f_cub = interp1d(t_obs, F_obs, kind='cubic')

# Evaluar en grilla fina
t_interp = [i*0.025 for i in range(401)]  # más fino para suavidad
F_lin = [f_lin(t) for t in t_interp]
F_cub = [f_cub(t) for t in t_interp]
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Solución 5 (continuación - gráficos)
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 5: \hfill \textcolor{green}{$\checkmark$} \\ Interpolación (gráficos)}
\begin{minted}[firstnumber=23]{python}
# Graficar comparación
plt.figure(figsize=(10, 6))
plt.plot(t_full, F_full, 'k-', alpha=0.3, linewidth=2, label='Original')
plt.scatter(t_obs, F_obs, s=20, c='black', zorder=5, label='Observaciones')
plt.plot(t_interp, F_lin, 'b--', linewidth=1.5, label='Lineal')
plt.plot(t_interp, F_cub, 'r-', linewidth=1.5, label='Cúbica')
plt.xlabel('Tiempo')
plt.ylabel('Flujo F(t)')
plt.title('Interpolación de Curva de Luz')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

# Comentario: La interpolación cúbica sigue mejor la forma suave
# de la señal original, mientras que la lineal produce segmentos
# rectos entre puntos. Ambas son exactas en los puntos observados.
\end{minted}
\textbf{Nota:} cúbica captura mejor la variación suave; lineal es más robusta al ruido.
\end{frame}

% ------------------------------------------------------------------------
% Interpretación Solución 5
% ------------------------------------------------------------------------
\begin{frame}{Interpretación Solución 5: Interpolación de Curva de Luz}
  \begin{block}{Interpolación lineal vs cúbica}
    \begin{itemize}
      \item \textbf{Lineal}: Conecta puntos con rectas → simple, robusta, no suave
      \item \textbf{Cúbica}: Usa splines cúbicos → suave, captura curvaturas, puede oscilar
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{¿Cuándo usar cada una?}
    \begin{itemize}
      \item \textbf{Lineal}: Datos ruidosos, pocos puntos, necesita robustez
      \item \textbf{Cúbica}: Señal suave subyacente, alta densidad de puntos
      \item \textbf{Nunca extrapolar}: Interpolación solo válida dentro del rango de datos
    \end{itemize}
  \end{alertblock}
  
  \textbf{Error RMS}: \(\text{RMS} = \sqrt{\frac{1}{N}\sum (F_{\text{interp}} - F_{\text{real}})^2}\)
  
  Un RMS bajo indica que la interpolación reconstruye bien la señal original.
  
  \vspace{0.3cm}
  \textbf{Aplicación en astronomía}: Rellenar huecos en curvas de luz, tránsitos de exoplanetas, variabilidad estelar.
\end{frame}

\begin{frame}[fragile]{Solución 6: \hfill \textcolor{green}{$\checkmark$} \\ Lognormal vs Normal}
\begin{minted}{python}
from scipy import stats
import random
import matplotlib.pyplot as plt
import numpy as np

# Generar datos lognormales
random.seed(42)
n = 1500
mu, sigma = 0.0, 0.6
x = [random.lognormvariate(mu, sigma) for _ in range(n)]

# Ajustar distribución Lognormal
s_log, _, scale_log = stats.lognorm.fit(x, floc=0)
ll_log = sum(stats.lognorm.logpdf(xi, s_log, 0, scale_log) for xi in x)
k_log = 2  # parámetros: s y scale
AIC_log = 2*k_log - 2*ll_log

# Ajustar distribución Normal
mu_norm, sig_norm = stats.norm.fit(x)
ll_norm = sum(stats.norm.logpdf(xi, mu_norm, sig_norm) for xi in x)
k_norm = 2  # parámetros: mu y sigma
AIC_norm = 2*k_norm - 2*ll_norm

print(f"Lognormal: AIC = {AIC_log:.1f}")
print(f"Normal: AIC = {AIC_norm:.1f}")
print(f"Mejor modelo: {'Lognormal' if AIC_log < AIC_norm else 'Normal'}")
\end{minted}
\end{frame}

% ------------------------------------------------------------------------
% Solución 6 (continuación - gráficos)
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 6: \hfill \textcolor{green}{$\checkmark$} \\ Lognormal vs Normal (gráficos)}
\begin{minted}[firstnumber=24]{python}
# Graficar histograma y PDFs ajustadas
plt.figure(figsize=(10, 6))
plt.hist(x, bins=40, density=True, alpha=0.5, edgecolor='black', 
         label='Histograma')

# Generar curvas de densidad
x_plot = np.linspace(0, max(x), 500)
pdf_log = stats.lognorm.pdf(x_plot, s_log, 0, scale_log)
pdf_norm = stats.norm.pdf(x_plot, mu_norm, sig_norm)

plt.plot(x_plot, pdf_log, 'r-', linewidth=2, 
         label=f'Lognormal (AIC={AIC_log:.1f})')
plt.plot(x_plot, pdf_norm, 'b--', linewidth=2, 
         label=f'Normal (AIC={AIC_norm:.1f})')

plt.xlabel('Valor')
plt.ylabel('Densidad de Probabilidad')
plt.title('Comparación de Ajustes: Lognormal vs Normal')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\textbf{Conclusión:} Lognormal tiene menor AIC y captura mejor la asimetría positiva.
\end{frame}

% ------------------------------------------------------------------------
% Interpretación Solución 6
% ------------------------------------------------------------------------
\begin{frame}{Interpretación Solución 6: Lognormal vs Normal}
  \begin{block}{Criterio de Información de Akaike (AIC)}
    \begin{itemize}
      \item \(\text{AIC} = 2k - 2\ln(L)\) donde \(k\) = número de parámetros, \(L\) = verosimilitud
      \item Penaliza modelos complejos (muchos parámetros)
      \item \textbf{Regla}: El modelo con \textbf{menor AIC} es preferible
      \item Diferencia \(\Delta\text{AIC} > 10\): Evidencia fuerte a favor del mejor modelo
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{¿Por qué la Normal falla?}
    \begin{itemize}
      \item Datos Lognormales son \textbf{asimétricos} (cola larga hacia valores altos)
      \item La Normal es \textbf{simétrica} → no puede capturar la asimetría
      \item Forzar una Normal produce residuos sistemáticos
    \end{itemize}
  \end{alertblock}
  
  \textbf{Interpretación física}: Muchas magnitudes astrofísicas son lognormales (masas, luminosidades, tiempos) debido a procesos multiplicativos. Usar Normal lleva a conclusiones erróneas.
  
  \vspace{0.3cm}
  \textbf{Conclusión}: Siempre visualizar datos antes de elegir modelo. El AIC cuantifica lo que el ojo ve.
\end{frame}

\begin{frame}[fragile]{Extra: Prueba KS (Kolmogorov-Smirnov)}
\begin{minted}{python}
from scipy import stats

# Comparar datos con distribución teórica
D, p_value = stats.kstest(x, 'norm', args=(mu, sigma))

# D: estadístico KS (distancia máxima entre CDFs)
# p_value > 0.05: no rechazamos la hipótesis (datos ~ Normal)
# p_value < 0.05: rechazamos (datos NO son Normal)

print(f"KS: D={D:.3f}, p={p_value:.3f}")
\end{minted}
\textbf{Nota:} Complementa al AIC para validación de modelos.
