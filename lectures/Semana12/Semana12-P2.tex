      \item Datos simulados: 15 mediciones de actividad (\(N\)) vs tiempo (\(t\)) en años
      \item Valores reales: \(N_0 = 1000\) Bq, \(T_{1/2} = 5730\) años

          \item Use `curve\_fit` para ajustar los parámetros
          \item Extraiga incertidumbres desde `pcov`
          \item Grafique datos con barras de error + curva ajustada
          \item Calcule \(T_{1/2}\) ajustado y compare con el real
        \end{enumerate}
    \end{itemize}
  \end{block}
  
  \textbf{Conceptos}: Decaimiento exponencial, vida media
  \\
  \textbf{Física}: Datación por radiocarbono

% Solución 1: Decaimiento Radiactivo
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución 1: \hfill \textcolor{green}{$\checkmark$} \\ Decaimiento Radiactivo}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
from scipy.optimize import curve_fit
import numpy as np
import matplotlib.pyplot as plt

# Modelo de decaimiento exponencial
def exponencial_decay(t, N0, lambda_val):
    return N0 * np.exp(-lambda_val * t)

# Parámetros reales
T_half_real = 5730  # años
lambda_real = np.log(2) / T_half_real
N0_real = 1000  # Bq

# Generar datos
np.random.seed(42)
t = np.linspace(0, 15000, 15)
N_real = exponencial_decay(t, N0_real, lambda_real)
sigma_N = 0.05 * N_real  # 5% de ruido
N_obs = N_real + np.random.normal(0, sigma_N)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=18]{python}
# Ajustar con curve_fit
popt, pcov = curve_fit(
    exponencial_decay, t, N_obs, 
    sigma=sigma_N, absolute_sigma=True
)
N0_fit, lambda_fit = popt
perr = np.sqrt(np.diag(pcov))

# Calcular T_1/2 ajustado
T_half_fit = np.log(2) / lambda_fit
sigma_T_half = T_half_fit * (perr[1]/lambda_fit)

print(f"N0 = {N0_fit:.1f} ± {perr[0]:.1f} Bq")
print(f"T_1/2 = {T_half_fit:.0f} ± {sigma_T_half:.0f} años")
print(f"Real: {T_half_real} años")
print(f"Error relativo: {abs(T_half_fit-T_half_real)/T_half_real*100:.1f}%")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 1 (cont.): \hfill \textcolor{green}{$\checkmark$} \\ Visualización}
\begin{minted}{python}
# Graficar resultados
t_fit = np.linspace(0, 15000, 200)
N_fit = exponencial_decay(t_fit, *popt)

plt.figure(figsize=(10, 6))
plt.errorbar(t, N_obs, yerr=sigma_N, fmt='o', capsize=3, 
             label='Mediciones', alpha=0.7)
plt.plot(t_fit, N_fit, 'r-', lw=2, 
         label=f'Ajuste: T₁/₂={T_half_fit:.0f}±{sigma_T_half:.0f} años')
plt.axhline(N0_real/2, color='gray', linestyle='--', alpha=0.5, 
            label=f'N₀/2 (t={T_half_real} años)')
plt.xlabel('Tiempo (años)')
plt.ylabel('Actividad N (Bq)')
plt.title('Decaimiento Radiactivo de C-14')
plt.legend()
plt.grid(alpha=0.3)
plt.show()
\end{minted}
\textbf{Discusión}: La incertidumbre en \(\lambda\) se propaga a \(T_{1/2}\) mediante \(\sigma_{T_{1/2}} = T_{1/2} \cdot (\sigma_\lambda / \lambda)\).
\end{frame}

# Modelo oscilador amortiguado
def oscilador(t, A, gamma, omega, phi):
    return A * np.exp(-gamma*t) * np.cos(omega*t + phi)

# Parámetros reales
A_real, gamma_real = 10, 0.1  # cm, s^-1
omega_real, phi_real = 2*np.pi, 0  # rad/s

# Generar datos
np.random.seed(42)
t = np.linspace(0, 20, 50)
x_real = oscilador(t, A_real, gamma_real, omega_real, phi_real)
sigma_x = np.full_like(x_real, 0.3)  # 0.3 cm
x_obs = x_real + np.random.normal(0, sigma_x)

# Ajustar (necesitamos valores iniciales)
p0 = [8, 0.05, 6, 0]  # estimaciones
popt, pcov = curve_fit(
    oscilador, t, x_obs, p0=p0,
    sigma=sigma_x, absolute_sigma=True
)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=20]{python}
A_fit, gamma_fit, omega_fit, phi_fit = popt
perr = np.sqrt(np.diag(pcov))

# Magnitudes derivadas
T_periodo = 2*np.pi / omega_fit
tau_decay = 1 / gamma_fit

print(f"Amplitud A = {A_fit:.2f} ± {perr[0]:.2f} cm")
print(f"Amortiguamiento γ = {gamma_fit:.3f} ± {perr[1]:.3f} s⁻¹")
print(f"Frecuencia ω = {omega_fit:.2f} ± {perr[2]:.2f} rad/s")
print(f"Fase φ = {phi_fit:.2f} ± {perr[3]:.2f} rad")
print(f"\nPeriodo T = {T_periodo:.2f} s")
print(f"Tiempo de decaimiento τ = {tau_decay:.1f} s")
\end{minted}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Solución 2 (cont.): \hfill \textcolor{green}{$\checkmark$} \\ Banda de Confianza}

# Graficar con banda de confianza
t_fit = np.linspace(0, 20, 300)
x_fit = oscilador(t_fit, *popt)

# Banda de confianza (simplificada: ±1 sigma en amplitud)
x_upper = oscilador(t_fit, A_fit+perr[0], gamma_fit, omega_fit, phi_fit)
x_lower = oscilador(t_fit, A_fit-perr[0], gamma_fit, omega_fit, phi_fit)

plt.figure(figsize=(12, 6))
plt.errorbar(t, x_obs, yerr=sigma_x, fmt='o', capsize=2, 
             label='Mediciones', alpha=0.6)
plt.plot(t_fit, x_fit, 'r-', lw=2, label='Ajuste')
plt.fill_between(t_fit, x_lower, x_upper, color='red', alpha=0.2, 
                 label='Banda ±1σ')
plt.xlabel('Tiempo (s)')
plt.ylabel('Posición x (cm)')
plt.title('Oscilador Amortiguado')
plt.legend()
plt.grid(alpha=0.3)
plt.show()
\end{minted}
\textbf{Nota}: Para banda completa, propagar todas las incertidumbres con Monte Carlo.

def logistica(t, K, r, t0):

popt, pcov = curve_fit(

)

K_fit, r_fit, t0_fit = popt

print(f"Punto medio t₀ = {t0_fit:.1f} ± {perr[2]:.1f} Myr")

def t_90(K, r, t0):

\begin{frame}[fragile]{Solución 4: \hfill \textcolor{green}{$\checkmark$} \\ Ley de Enfriamiento}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{minted}{python}
# Modelo de enfriamiento
def enfriamiento(t, T_amb, T0, k):
    return T_amb + (T0 - T_amb) * np.exp(-k*t)

# Parámetros reales
T0_real, T_amb_real = 25000, 2.7  # K
k_real = 0.05  # Gyr^-1

# Generar datos
np.random.seed(42)
t = np.linspace(0, 50, 20)
T_real = enfriamiento(t, T_amb_real, T0_real, k_real)
sigma_T = np.full_like(T_real, 200)  # 200 K
T_obs = T_real + np.random.normal(0, sigma_T)

# Ajustar
popt, pcov = curve_fit(
    enfriamiento, t, T_obs,
    sigma=sigma_T, absolute_sigma=True
)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=18]{python}
T_amb_fit, T0_fit, k_fit = popt
perr = np.sqrt(np.diag(pcov))

# Tiempo de semi-enfriamiento
t_half = np.log(2) / k_fit
sigma_t_half = t_half * (perr[2]/k_fit)

print(f"T_amb = {T_amb_fit:.1f} ± {perr[0]:.1f} K")
print(f"T₀ = {T0_fit:.0f} ± {perr[1]:.0f} K")
print(f"k = {k_fit:.4f} ± {perr[2]:.4f} Gyr⁻¹")
print(f"\nt_1/2 = {t_half:.1f} ± {sigma_t_half:.1f} Gyr")

# Calcular chi^2 reducido
residuos = T_obs - enfriamiento(t, *popt)
chi2_red = np.sum((residuos/sigma_T)**2) / (len(t) - 3)
print(f"χ²_red = {chi2_red:.2f}")
\end{minted}
\end{column}
\end{columns}
\end{frame}

         label=f'Ajuste: t₁/₂={t_half:.1f}±{sigma_t_half:.1f} Gyr')

\end{frame}

\section{Consolidación y Retroalimentación}

\begin{frame}{Puesta en Común - ¿Cómo les fue?}
  \begin{block}{Reflexión sobre los ejercicios}
    \begin{itemize}
      \item ¿Qué modelo resultó más difícil de ajustar? ¿Por qué?
      \item ¿Encontraron problemas con valores iniciales en oscilador/logística?
      \item ¿Cómo interpretaron las incertidumbres en el contexto físico?
      \item ¿Lograron calcular magnitudes derivadas (T_{1/2}, periodo, etc.)?
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Objetivo}
    Desarrollar intuición para elegir modelos apropiados y validar ajustes con métricas como \(\chi^2_{\text{red}}\).
  \end{alertblock}
\end{frame}

\begin{frame}{Errores Comunes con \texttt{curve\_fit}}
  \begin{block}{Errores frecuentes}
    \begin{itemize}
      \item \textbf{Olvidar valores iniciales `p0`}: Modelos no lineales necesitan buenas estimaciones
      \item \textbf{No usar `absolute\_sigma=True`}: Las incertidumbres quedan en escala relativa
      \item \textbf{Extrapolar fuera del rango de datos}: `curve\_fit` ajusta, no predice
      \item \textbf{Ignorar $(\chi^2_{\text{red}})$}: Un valor lejano de 1 indica problemas
      \item \textbf{No propagar errores correctamente}: Usar fórmulas de propagación para magnitudes derivadas
    \end{itemize}
  \end{block}
  
  \begin{alertblock}{Consejo}
    \textbf{Siempre visualiza residuos}: `plt.plot(t, T\_obs - T\_fit)` revela patrones sistemáticos.
  \end{alertblock}
\end{frame}

# Para T_1/2 = ln(2) / lambda:

    deriv = -np.log(2) / lambda_fit**2
    var_T = deriv**2 * pcov_lambda

# Criterio de Información de Akaike (introducción simple)
def calcular_AIC(residuos, sigma, n_params):
    chi2 = np.sum((residuos/sigma)**2)
    log_likelihood = -0.5 * chi2
    AIC = 2*n_params - 2*log_likelihood
    return AIC

# Ejemplo: comparar modelo lineal vs cuadrático
residuos_lin = y_obs - modelo_lineal(x, *popt_lin)
residuos_quad = y_obs - modelo_cuadratico(x, *popt_quad)

AIC_quad = calcular_AIC(residuos_quad, sigma, 3) # 3 parámetros

print(f"AIC lineal: {AIC_lin:.1f}")
print(f"AIC cuadrático: {AIC_quad:.1f}")
print("Mejor modelo:", "lineal" if AIC_lin < AIC_quad else "cuadrático")
