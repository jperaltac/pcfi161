% ------------------------------------------------------------------------
% Slide 5: ¿Qué es Matplotlib?
% ------------------------------------------------------------------------
\begin{frame}{¿Qué es Matplotlib?}
  \begin{itemize}
    \item \textbf{Librería de visualización} más popular en Python para crear gráficos estáticos, animados e interactivos.
    \item \textbf{Inspirada en MATLAB}: sintaxis similar y familiar.
    \item \textbf{Dos interfaces principales}:
      \begin{itemize}
        \item \texttt{pyplot} (estilo MATLAB, más simple)
        \item Orientada a objetos (más control y flexibilidad)
      \end{itemize}
    \item \textbf{Integración perfecta} con NumPy y otras librerías científicas.
  \end{itemize}
  
  \begin{alertblock}{Importación estándar}
    \texttt{import matplotlib.pyplot as plt}
  \end{alertblock}
\end{frame}

\begin{frame}{Ejercicio Guiado: \hfill \textcolor{red}{$\clubsuit$} \\ Gráfica de Caída Libre}
  \begin{block}{Enunciado}
    Crear una gráfica que muestre la posición vs tiempo para un objeto en caída libre:
    \begin{itemize}
      \item Usar la fórmula: \(y(t) = y_0 - \frac{1}{2}gt^2\)
      \item \(y_0 = 100\) m (altura inicial)
      \item \(g = 9.81\) m/s² (gravedad)
      \item Tiempo desde 0 hasta cuando el objeto toca el suelo
    \end{itemize}
  \end{block}
  
  \textbf{Pasos a seguir:}
  \begin{enumerate}
    \item Crear array de tiempo
    \item Calcular posiciones
    \item Crear gráfica con etiquetas apropiadas
    \item Agregar título y grilla
    \item Marcar el punto donde toca el suelo
  \end{enumerate}
\end{frame}

% ------------------------------------------------------------------------
% Slide 13: Solución del Ejercicio Guiado
% ------------------------------------------------------------------------
\begin{frame}[fragile]{Solución del Ejercicio Guiado: \hfill \textcolor{green}{$\checkmark$} \\ Gráfica de Caída Libre}
\begin{minted}[fontsize=\tiny]{python}
# Paso 1: Definir parámetros
y0 = 100  # altura inicial en metros
g = 9.81  # aceleración de gravedad en m/s²

# Calcular tiempo cuando toca el suelo: y = 0
# 0 = y0 - (1/2)*g*t²  →  t = sqrt(2*y0/g)
t_final = np.sqrt(2 * y0 / g)
print(f"Tiempo de caída: {t_final:.2f} segundos")
# Paso 2: Crear array de tiempo
t = np.linspace(0, t_final, 100)
# Paso 3: Calcular posiciones
y = y0 - 0.5 * g * t**2
# Paso 4: Crear la gráfica
plt.figure(figsize=(8, 6))
plt.plot(t, y, 'b-', linewidth=2, label='Posición vs Tiempo')
# Paso 5: Marcar punto de impacto
plt.plot(t_final, 0, 'ro', markersize=8, label='Impacto con el suelo')
# Paso 6: Personalizar la gráfica
plt.xlabel('Tiempo (s)')
plt.ylabel('Altura (m)')
plt.title('Caída Libre desde 100 metros')
plt.grid(True, alpha=0.3)
plt.legend()
# Agregar anotación
plt.annotate(f't = {t_final:.2f} s', 
             xy=(t_final, 0), xytext=(t_final-1, 20),
             arrowprops=dict(arrowstyle='->', color='red'))
plt.show()
\end{minted}
\end{frame}

# Generar array x en [0, 2π]
import numpy as np
import matplotlib.pyplot as plt

x = np.linspace(0, 2*np.pi, 100)

# Calcular funciones trigonométricas
y_sin , y_cos , y_tan = np.sin(x), np.cos(x), np.tan(x)

# Limitar valores extremos de tangente para mejor visualización
y_tan = np.where(np.abs(y_tan) > 10, np.nan, y_tan)

# Crear subplots en una columna
fig, axs = plt.subplots(3, 1, figsize=(8, 10))

# Subplot 1: Seno
axs[0].plot(x, y_sin, 'r-', linewidth=2, label='sin(x)')
axs[0].set_title('Función Seno')
axs[0].set_ylabel('sin(x)')
axs[0].legend()
axs[0].grid(True, alpha=0.3)
\end{minted}

axs[1].grid(True, alpha=0.3)

# Subplot 3: Tangente
axs[2].plot(x, y_tan, 'g:', linewidth=2, label='tan(x)')
axs[2].set_title('Función Tangente')
axs[2].set_xlabel('x (radianes)')
axs[2].set_ylabel('tan(x)')
axs[2].set_ylim(-5, 5)  # Limitar rango y para mejor visualización
axs[2].legend()
axs[2].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
\textbf{Discusión:} El manejo de la tangente requiere cuidado especial en sus discontinuidades usando \texttt{np.where} y \texttt{set\_ylim}.

import numpy as np

datos = np.random.normal(5, 1, 100)  # media=5, desv_std=1, 100 valores

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

\begin{column}{0.5\textwidth}
\begin{minted}{python}
from mpl_toolkits.mplot3d import Axes3D
import matplotlib.pyplot as plt
import numpy as np

# Definir dominio para x e y
x = np.linspace(-2, 2, 50)
y = np.linspace(-2, 2, 50)

# Crear meshgrid
X, Y = np.meshgrid(x, y)

# Calcular la función f(x,y) = e^(-x²-y²)
Z = np.exp(-(X**2 + Y**2))

# Crear figura 3D
fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')

# Plot de superficie
surface = ax.plot_surface(X, Y, Z, cmap='plasma', alpha=0.9)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=21]{python}
# Configurar etiquetas y título
ax.set_xlabel('X')
ax.set_ylabel('Y')
ax.set_zlabel('f(x,y) = e^(-x²-y²)')
ax.set_title('Superficie 3D: Función Gaussiana 2D')

# Agregar barra de colores
fig.colorbar(surface, shrink=0.5, aspect=20)

# Ajustar límites del eje z
ax.set_zlim(0, 1)

# Ajustar ángulo de visualización
ax.view_init(elev=30, azim=45)
plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
\textbf{Discusión:} La función gaussiana 2D crea una "campana" perfecta. El colormap \texttt{plasma} resalta bien las variaciones de altura.
