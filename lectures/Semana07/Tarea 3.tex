\section{Problema de Evaluación}

% ------------------------------------------------------------------------
% Slide 4: Problema Principal - Parte 1: Análisis Matricial
% ------------------------------------------------------------------------
\begin{frame}{Problema Principal - Parte 1: \hfill \textcolor{red}{$\clubsuit$} \\ Análisis de Matriz Aleatoria}
  \begin{block}{Enunciado - Generación y Análisis Matricial}
    \begin{enumerate}
      \item \textbf{Crear una matriz} $M$ de tamaño $5 \times 5$ con valores enteros aleatorios en el rango $[1, 10)$ usando \texttt{np.random.randint()}.
      \item \textbf{Configurar semilla aleatoria} usando \texttt{np.random.seed()} con el último dígito de su cédula para reproducibilidad.
      \item \textbf{Calcular las siguientes propiedades matriciales:}
        \begin{itemize}
          \item Determinante: $\det(M)$ usando \texttt{np.linalg.det()}
          \item Traza: $\text{tr}(M) = \sum_{i=1}^{5} M_{ii}$ usando \texttt{np.trace()}
          \item Norma Euclidiana: $\|M\|_F = \sqrt{\sum_{i,j} M_{ij}^2}$ usando \texttt{np.linalg.norm()}
        \end{itemize}
      \item \textbf{Imprimir resultados} con formato profesional incluyendo 2 decimales.
    \end{enumerate}
  \end{block}
  
  \textbf{Conceptos aplicados:} Álgebra lineal computacional, generación de datos aleatorios, propiedades matriciales.
\end{frame}

% ------------------------------------------------------------------------
% Slide 5: Problema Principal - Parte 2: Visualización Científica
% ------------------------------------------------------------------------
\begin{frame}{Problema Principal - Parte 2: \hfill \textcolor{red}{$\clubsuit$} \\ Visualización Científica Simple}
  \begin{block}{Enunciado - Gráfico de Funciones Superpuestas}
    \begin{enumerate}
      \item \textbf{Generar datos para el gráfico}:
        \begin{itemize}
          \item Crear array $x$ usando \texttt{np.linspace(0, 2*np.pi, 100)}
          \item Calcular dos funciones: $f_1(x) = \sin(x)$ y $f_2(x) = \cos(x)$
        \end{itemize}
      \item \textbf{Crear una figura simple}:
      \begin{itemize}
        \item Usando \texttt{plt.figure(figsize=(10, 6))}.
      \end{itemize}
      \item \textbf{Configurar el gráfico}:
        \begin{itemize}
          \item Título: "Funciones Trigonométricas"
          \item Etiquetas: "x (radianes)" y "f(x)"
          \item Agregar leyenda y grilla
          \item Configurar límites del eje y: $[-1.2, 1.2]$
        \end{itemize}
      \item \textbf{Añadir línea vertical} en $x = \pi/2$ con color verde y estilo discontinuo.
    \end{enumerate}
  \end{block}
  
  \textbf{Conceptos aplicados:} Gráficos superpuestos, personalización de estilos, funciones trigonométricas.
\end{frame}

      \item \textbf{Configuración de subplots 3D:} Olvidar \texttt{projection='3d'}

    \end{itemize}
  \end{block}
