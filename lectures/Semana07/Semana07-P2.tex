\begin{frame}[fragile]{Solución 1 de Referencia: \hfill \textcolor{green}{$\checkmark$} \\ Primer Gráfico con NumPy}
\begin{minted}[fontsize=\tiny]{python}
import numpy as np
import matplotlib.pyplot as plt

# Crear array de 20 puntos entre 0 y 10
x = np.linspace(0, 10, 20)
y = x**2

# Crear la gráfica
plt.plot(x, y, 'bo-', markersize=6)
plt.xlabel('x')
plt.ylabel('x²')
plt.title('Función Cuadrática')
plt.grid(True, alpha=0.3)
plt.show()

# Resultado: Gráfica clara de la parábola y = x²
print(f"Rango de x: [{np.min(x):.1f}, {np.max(x):.1f}]")
print(f"Rango de y: [{np.min(y):.1f}, {np.max(y):.1f}]")
\end{minted}
\textbf{Discusión:} NumPy permite crear espaciados uniformes y aplicar operaciones a todo el array simultáneamente.
\end{frame}

ax2.set_ylabel('cos(x)')
ax2.legend()
ax2.grid(True, alpha=0.3)

plt.show()

\begin{columns}[t]
\begin{column}{0.5\textwidth}
\begin{minted}[fontsize=\scriptsize]{python}
import numpy as np
import matplotlib.pyplot as plt

# Parámetros del proyectil
v0, theta, g = 30, 45, 9.81
theta_rad = np.radians(theta)

# Tiempo de vuelo total
t_max = 2 * v0 * np.sin(theta_rad) / g
t = np.linspace(0, t_max, 100)

# Ecuaciones cinemáticas
x = v0 * np.cos(theta_rad) * t
y = v0 * np.sin(theta_rad) * t - 0.5 * g * t**2

# Crear subplots
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[fontsize=\scriptsize,firstnumber=18]{python}
# Altura vs tiempo
ax1.plot(t, y, 'b-', linewidth=2)
ax1.set_xlabel('Tiempo (s)')
ax1.set_ylabel('Altura (m)')
ax1.set_title('Altura vs Tiempo')
ax1.grid(True, alpha=0.3)

# Trayectoria 2D
ax2.plot(x, y, 'r-', linewidth=2)
ax2.set_xlabel('Distancia (m)')
ax2.set_ylabel('Altura (m)')
ax2.set_title('Trayectoria del Proyectil')
ax2.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
print(f"Alcance máximo: {np.max(x):.2f} m")
print(f"Altura máxima: {np.max(y):.2f} m")
\end{minted}
\end{column}
\end{columns}
\vspace{0.15cm}
\textbf{Discusión:} Las ecuaciones cinemáticas vectorizadas permiten calcular toda la trayectoria eficientemente.
\end{frame}

import numpy as np
import matplotlib.pyplot as plt

# Crear matriz 4x4 con números 1-16
matriz = np.arange(1, 17).reshape(4, 4)
diagonal = np.diag(matriz)
suma_diag = np.sum(diagonal)

print(f"Matriz 4x4:\n{matriz}")
print(f"Diagonal principal: {diagonal}")
print(f"Suma de la diagonal: {suma_diag}")

# Crear visualizaciones
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(10, 4))

# Visualizar matriz como mapa de calor
im = ax1.imshow(matriz, cmap='viridis', interpolation='nearest')
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=18]{python}
ax1.set_title('Matriz 4×4')
for i in range(4):
    for j in range(4):
        ax1.text(j, i, matriz[i, j], ha='center', 
                 va='center', color='white', fontweight='bold')
plt.colorbar(im, ax=ax1)

# Gráfico de barras de la diagonal
ax2.bar(range(len(diagonal)), diagonal, 
        color='orange', alpha=0.7)
ax2.set_title('Diagonal Principal')
ax2.set_xlabel('Posición')
ax2.set_ylabel('Valor')
ax2.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
\end{minted}
\end{column}
\end{columns}

\begin{minted}{python}
import numpy as np
import matplotlib.pyplot as plt

# Configurar semilla para reproducibilidad
np.random.seed(42)

# Generar tres conjuntos de datos
datos1 = np.random.normal(0, 1, 1000)      # Normal estándar
datos2 = np.random.normal(2, 0.5, 1000)    # Normal desplazada
datos3 = np.random.uniform(0, 4, 1000)     # Uniforme

# Calcular medias
media1, media2, media3 = np.mean(datos1), np.mean(datos2), np.mean(datos3)

# Crear histogramas superpuestos
plt.figure(figsize=(10, 6))
plt.hist(datos1, bins=30, alpha=0.6, label='N(0,1)', color='blue', density=True)
\end{minted}
\end{column}
\begin{column}{0.5\textwidth}
\begin{minted}[firstnumber=18]{python}
plt.hist(datos2, bins=30, alpha=0.6, label='N(2,0.5)', color='red', density=True)
plt.hist(datos3, bins=30, alpha=0.6, label='U(0,4)', color='green', density=True)

# Líneas verticales para las medias
plt.axvline(media1, color='blue', linestyle='--', 
            linewidth=2, label=f'Media N(0,1): {media1:.2f}')
plt.axvline(media2, color='red', linestyle='--', 
            linewidth=2, label=f'Media N(2,0.5): {media2:.2f}')
plt.axvline(media3, color='green', linestyle='--', 
            linewidth=2, label=f'Media U(0,4): {media3:.2f}')

plt.legend()
plt.title('Comparación de Distribuciones Estadísticas')
plt.xlabel('Valor')
plt.ylabel('Densidad')
plt.grid(True, alpha=0.3)
plt.show()
\end{minted}
\end{column}
\end{columns}
\vspace{0.3cm}
